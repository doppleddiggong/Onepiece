18FReply UChatInputBox::NativeOnPreviewKeyDown(const FGeometry& InGeometry, const FKeyEvent& InKeyEvent)
19{
20 // Enter 키 감지 & Shift+Enter는 줄바꿈 허용
21 if (InKeyEvent.GetKey() == EKeys::Enter && !InKeyEvent.IsShiftDown())

23 HandleSendClicked();
24 return FReply::Handled();
25 }
26

30FText UChatInputBox::FlushMessage()

39void UChatInputBox::HandleSendClicked()
40{
41 // 텍스트가 비어있지 않다면 Broadcast
42 FText message = FlushMessage();
43 if (!message.IsEmpty())
44 {
45 if (auto* PC = Cast<APlayerControl>(GetWorld()->GetFirstPlayerController<APlayerController>()))
46 {
47 PC->ServerRPC_SendChat(message);
48 }
49 }
50}
