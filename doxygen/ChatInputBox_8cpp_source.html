21void UChatInputBox::NativeTick(const FGeometry& MyGeometry, float InDeltaTime)
22{
23 Super::NativeTick(MyGeometry, InDeltaTime);
24
25 // 현재 포커스 상태 확인
26 bool bIsFocused = MultiLineEditableTextBox_Input && MultiLineEditableTextBox_Input->HasKeyboardFocus();
27
28 // 포커스 상태 변화 감지
29 if (bWasFocused != bIsFocused)
30 {
31 PRINTLOG(TEXT("[ChatInputBox] Focus changed: %d -> %d"), bWasFocused, bIsFocused);
32
33 // 포커스를 잃었을 때만 처리 (화면 클릭 등)
34 if (bWasFocused && !bIsFocused)
35 {
36 PRINTLOG(TEXT("[ChatInputBox] Focus lost - switching to GameOnly mode"));
37
38 // GameOnly 모드로 전환
39 if (APlayerControl* PC = Cast<APlayerControl>(GetOwningPlayer()))
40 {
41 FInputModeGameOnly InputMode;
42 PC->SetInputMode(InputMode);
43 PC->SetShowMouseCursor(false);
44 FSlateApplication::Get().SetAllUserFocusToGameViewport();
45 }
46
47 // ChatWidget에 포커스 해제 알림
48 if (OwningChatWidget)
49 {
50 PRINTLOG(TEXT("[ChatInputBox] Notifying ChatWidget of focus loss"));
51 OwningChatWidget->OnInputFocusChanged(false);
52 }
53 else
54 {
55 PRINTLOG(TEXT("[ChatInputBox] ERROR: OwningChatWidget is null!"));
56 }
57 }
58 }
59
60 bWasFocused = bIsFocused;
61}

63FReply UChatInputBox::NativeOnPreviewKeyDown(const FGeometry& InGeometry, const FKeyEvent& InKeyEvent)
64{
65 // Enter 키 감지 & Shift+Enter는 줄바꿈 허용
66 if (InKeyEvent.GetKey() == EKeys::Enter && !InKeyEvent.IsShiftDown())
67 {
68 HandleSendClicked();
69 return FReply::Handled();
70 }
71
72 // ESC 키로 포커스 해제
73 if (InKeyEvent.GetKey() == EKeys::Escape)
74 {
75 SetInputFocus(false);
76 return FReply::Handled();
77 }
78
79 return Super::NativeOnPreviewKeyDown(InGeometry, InKeyEvent);
80}

92void UChatInputBox::SetInputFocus(bool bFocus)
93{
94 if ( bFocus )
95 {
96 // 먼저 포커스 설정
97 MultiLineEditableTextBox_Input->SetKeyboardFocus();
98
99 // UI 입력 모드로 전환
100 if (APlayerControl* PC = Cast<APlayerControl>(GetOwningPlayer()))
101 {
102 FInputModeUIOnly InputMode;
103 InputMode.SetWidgetToFocus(MultiLineEditableTextBox_Input->TakeWidget());
104 PC->SetInputMode(InputMode);
105 PC->SetShowMouseCursor(true);
106 }
107
108 // ChatWidget에 포커스 획득 알림
109 if (OwningChatWidget)
110 {
111 OwningChatWidget->OnInputFocusChanged(true);
112 }
113
114 bWasFocused = true;
115 }
116 else
117 {
118 // GameOnly 모드로 전환 및 포커스 해제
119 if (APlayerControl* PC = Cast<APlayerControl>(GetOwningPlayer()))
120 {
121 FInputModeGameOnly InputMode;
122 PC->SetInputMode(InputMode);
123 PC->SetShowMouseCursor(false);
124
125 // 뷰포트로 포커스 이동 (자동으로 입력창 포커스 해제됨)
126 FSlateApplication::Get().SetAllUserFocusToGameViewport();
127 }
128
129 // ChatWidget에 포커스 해제 알림
130 if (OwningChatWidget)
131 {
132 OwningChatWidget->OnInputFocusChanged(false);
133 }
134
135 bWasFocused = false;
136 }
137}

139bool UChatInputBox::IsAIAsk(const FString& InMessage, FString& OutQuestion) const
140{
141 FString LeftPart, RightPart;
142
143 // 콜론(:)을 기준으로 분리
144 if (InMessage.Split(TEXT(":"), &LeftPart, &RightPart))
145 {
146 // 왼쪽 파트의 공백을 제거하고 "AI"와 일치하는지 확인 (대소문자 무시)
147 if (LeftPart.TrimStartAndEnd().Equals(GameName::AI, ESearchCase::IgnoreCase))
148 {
149 OutQuestion = RightPart.TrimStart();
150 return !OutQuestion.IsEmpty(); // 내용이 비어있지 않아야 true
151 }
152 }
153 return false;
154}

156void UChatInputBox::HandleSendClicked()
157{
158 // 텍스트가 비어있지 않다면 처리
159 FText Message = FlushMessage();
160 if (Message.IsEmpty())
161 {
162 // 빈 메시지면 포커스만 해제
163 SetInputFocus(false);
164 return;
165 }
166
167 auto* PC = Cast<APlayerControl>(GetWorld()->GetFirstPlayerController());
168 if (!PC)
169 return;
170
171 const FString MessageStr = Message.ToString();
172 FString CleanQuestion;
173
174 if (IsAIAsk(MessageStr, CleanQuestion))
175 {
176 // 사용자가 입력한 전체 메시지 표시 (로그성)
177 PC->ServerRPC_SendChat(Message);
178
179 // AI에게 정제된 질문 전송
180 PC->ServerRPC_SendAIQuestion(CleanQuestion);
181
182 PRINTLOG(TEXT("[AI Chat] User question: %s"), *CleanQuestion);
183 }
184 else
185 {
186 // 2. 일반 채팅 메시지 전송
187 PC->ServerRPC_SendChat(Message);
188 }
189
190 // 메시지 전송 후 포커스 해제 및 게임 모드로 복원
191 SetInputFocus(false);
192
193}
