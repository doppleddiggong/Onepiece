57bool UChatInputBox::IsAIAsk(const FString& InMessage, FString& OutQuestion) const
58{
59 FString LeftPart, RightPart;
60
61 // 콜론(:)을 기준으로 분리
62 if (InMessage.Split(TEXT(":"), &LeftPart, &RightPart))
63 {
64 // 왼쪽 파트의 공백을 제거하고 "AI"와 일치하는지 확인 (대소문자 무시)
65 if (LeftPart.TrimStartAndEnd().Equals(GameName::AI, ESearchCase::IgnoreCase))
66 {
67 OutQuestion = RightPart.TrimStart();
68 return !OutQuestion.IsEmpty(); // 내용이 비어있지 않아야 true
69 }
70 }
71 return false;
72}

74void UChatInputBox::HandleSendClicked()
75{
76 // 텍스트가 비어있지 않다면 처리
77 FText Message = FlushMessage();
78 if (Message.IsEmpty())
79 return;
80
81 auto* PC = Cast<APlayerControl>(GetWorld()->GetFirstPlayerController());
82 if (!PC)
83 return;
84
85 const FString MessageStr = Message.ToString();
86 FString CleanQuestion;
87
88 if (IsAIAsk(MessageStr, CleanQuestion))
89 {
90 // 사용자가 입력한 전체 메시지 표시 (로그성)
91 PC->ServerRPC_SendChat(Message);
92
93 // AI에게 정제된 질문 전송
94 PC->ServerRPC_SendAIQuestion(CleanQuestion);
95
96 PRINTLOG(TEXT("[AI Chat] User question: %s"), *CleanQuestion);
97 }
98 else
99 {
100 // 2. 일반 채팅 메시지 전송
101 PC->ServerRPC_SendChat(Message);
102 }
103}
