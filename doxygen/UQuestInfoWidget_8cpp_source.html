58void UQuestInfoWidget::InitQuestInfo(EQuestRole InQuestRole)
59{
60 const auto GS = ULingoGameHelper::GetLingoGameState(GetWorld());
61 if (!GS || !GS->IsQuestIng())
62 {
63 SetVisibility(ESlateVisibility::Collapsed);
64 return;
65 }
66
67 const EQuestType QuestType = GS->GetCurrentQuestType();
68 this->QuestRole = InQuestRole;
69
70 SetVisibility(ESlateVisibility::Visible);
71
72 if (QuestType == EQuestType::Read)
73 {
74 const FResponseReadScenario& ReadScenario = GS->GetReadScenarioData();
75
76 if (QuestRole == EQuestRole::Both)
77 SetQuestText(ReadScenario.full_data);
78 else if (QuestRole == EQuestRole::OnlyQuestion1)
79 SetQuestText(ReadScenario.word_data1);
80 else if (QuestRole == EQuestRole::OnlyQuestion2)
81 SetQuestText(ReadScenario.word_data2);
82 }
83 else if (QuestType == EQuestType::Listen)
84 {
85 const FResponseListenScenario& ListenScenario = GS->GetListenScenarioData();
86
87 if (QuestRole == EQuestRole::Both)
88 SetQuestText(ListenScenario.full_data);
89 else if (QuestRole == EQuestRole::OnlyQuestion1)
90 SetQuestText(ListenScenario.word_data1);
91 else if (QuestRole == EQuestRole::OnlyQuestion2)
92 SetQuestText(ListenScenario.word_data2);
93 }
94}
