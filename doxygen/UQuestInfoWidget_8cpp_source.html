13void UQuestInfoWidget::InitQuestInfo()
14{
15 const auto GS = ULingoGameHelper::GetLingoGameState(GetWorld());
16 if (!GS || !GS->IsQuestIng())
17 {
18 SetVisibility(ESlateVisibility::Collapsed);
19 return;
20 }
21
22 const EQuestType QuestType = GS->GetCurrentQuestType();
23 auto QuestRole = ULingoGameHelper::GetLingoPlayerState(GetWorld())->QuestRole;
24
25 SetVisibility(ESlateVisibility::Visible);
26
27 if (QuestType == EQuestType::Read)
28 {
29 const FResponseReadScenario& ReadScenario = GS->GetReadScenarioData();
30
31 if (QuestRole == EQuestRole::Both)
32 SetQuestText(ReadScenario.full_data);
33 else if (QuestRole == EQuestRole::OnlyQuestion1)
34 SetQuestText(ReadScenario.word_data1);
35 else if (QuestRole == EQuestRole::OnlyQuestion2)
36 SetQuestText(ReadScenario.word_data2);
37 }
38 else if (QuestType == EQuestType::Listen)
39 {
40 const FResponseListenScenario& ListenScenario = GS->GetListenScenarioData();
41
42 if (QuestRole == EQuestRole::Both)
43 SetQuestText(ListenScenario.full_data);
44 else if (QuestRole == EQuestRole::OnlyQuestion1)
45 SetQuestText(ListenScenario.word_data1);
46 else if (QuestRole == EQuestRole::OnlyQuestion2)
47 SetQuestText(ListenScenario.word_data2);
48 }
49}
