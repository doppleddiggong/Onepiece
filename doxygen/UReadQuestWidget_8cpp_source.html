36void UReadQuestWidget::CacheReferences()
37{
38 // GameState 가져오기
39 if (!CachedGameState)
40 {
41 CachedGameState = Cast<ALingoGameState>(UGameplayStatics::GetGameState(GetWorld()));
42 if (!CachedGameState)
43 {
44 PRINTLOG(TEXT("[ReadQuestWidget] Failed to get GameState"));
45 }
46 }
47
48 // PlayerState 가져오기
49 if (!CachedPlayerState)
50 {
51 APlayerController* PC = GetOwningPlayer();
52 if (PC)
53 {
54 CachedPlayerState = PC->GetPlayerState<ALingoPlayerState>();
55 if (!CachedPlayerState)
56 {
57 PRINTLOG(TEXT("[ReadQuestWidget] Failed to get PlayerState"));
58 }
59 }
60 }
61}

63void UReadQuestWidget::InitializeQuest()
64{
65 PRINTLOG(TEXT("[ReadQuestWidget] Initializing quest"));
66
67 // 참조 확인
68 CacheReferences();
69
70 if (!CachedGameState || !CachedPlayerState)
71 {
72 PRINTLOG(TEXT("[ReadQuestWidget] Cannot initialize - missing references"));
73 return;
74 }
75
76 // 미션 설명 텍스트 설정
77 if (MissionDescriptionText && CachedGameState->CurScenarioData.word_data1.Eng.Len() > 0)
78 {
79 FString Description = FString::Printf(TEXT("Find: %s %s"),
80 *CachedGameState->CurScenarioData.word_data1.Eng,
81 *CachedGameState->CurScenarioData.word_data2.Eng);
82 MissionDescriptionText->SetText(FText::FromString(Description));
83 }
84
85 // 선택지 리스트 빌드
86 BuildSymbolList();
87 BuildColorList();
88
89 PRINTLOG(TEXT("[ReadQuestWidget] Quest initialized successfully"));
90}

92TArray<FString> UReadQuestWidget::GetUniqueSymbols() const
93{
94 TSet<FString> UniqueSet;
95 TArray<FString> Result;
96
97 if (!CachedGameState)
98 return Result;
99
100 // target_data에서 symbol 필드만 추출하여 중복 제거
101 for (const FScenarioTargetData& TargetData : CachedGameState->CurScenarioData.target_data)
102 {
103 if (!TargetData.symbol.IsEmpty())
104 {
105 UniqueSet.Add(TargetData.symbol);
106 }
107 }
108
109 Result = UniqueSet.Array();
110 return Result;
111}

134void UReadQuestWidget::BuildSymbolList()
135{
136 if (!SymbolScrollBox || !EntryWidgetClass)
137 {
138 PRINTLOG(TEXT("[ReadQuestWidget] Cannot build symbol list - missing components"));
139 return;
140 }
141
142 // 기존 엔트리 제거
143 SymbolScrollBox->ClearChildren();
144 SymbolEntries.Empty();
145
146 // 중복 제거된 심볼 리스트 가져오기
147 TArray<FString> Symbols = GetUniqueSymbols();
148
149 PRINTLOG(TEXT("[ReadQuestWidget] Building symbol list with %d items"), Symbols.Num());
150
151 // 플레이어 역할 확인 (OnlyQuestion2는 심볼 선택 불가)
152 bool bCanSelectSymbol = (CachedPlayerState && CachedPlayerState->QuestRole != EReadQuestRole::OnlyQuestion2);
153
154 // 엔트리 위젯 생성
155 for (const FString& Symbol : Symbols)
156 {
157 UReadQuestEntryWidget* EntryWidget = CreateWidget<UReadQuestEntryWidget>(GetWorld(), EntryWidgetClass);
158 if (EntryWidget)
159 {
160 EntryWidget->InitializeEntry(Symbol, bCanSelectSymbol);
161 EntryWidget->OnEntrySelected.AddUObject(this, &UReadQuestWidget::OnSymbolSelected);
162
163 SymbolScrollBox->AddChild(EntryWidget);
164 SymbolEntries.Add(EntryWidget);
165 }
166 }
167
168 PRINTLOG(TEXT("[ReadQuestWidget] Symbol list built successfully"));
169}

250void UReadQuestWidget::UpdateTimer()
251{
252 if (!TimerText || !CachedGameState)
253 return;
254
255 // 남은 시간 표시
256 float RemainTime = CachedGameState->GetRemainMissionTime();
257 int32 Minutes = FMath::FloorToInt(RemainTime / 60.f);
258 int32 Seconds = FMath::FloorToInt(RemainTime) % 60;
259
260 FString TimeString = FString::Printf(TEXT("%02d:%02d"), Minutes, Seconds);
261 TimerText->SetText(FText::FromString(TimeString));
262
263 // 시간이 30초 이하면 빨간색으로 표시
264 if (RemainTime <= 30.f)
265 {
266 TimerText->SetColorAndOpacity(FLinearColor::Red);
267 }
268 else
269 {
270 TimerText->SetColorAndOpacity(FLinearColor::White);
271 }
272}

274void UReadQuestWidget::OnPlayerStateUpdated()
275{
276 if (!CachedPlayerState)
277 return;
278
279 PRINTLOG(TEXT("[ReadQuestWidget] PlayerState updated - checking for wrong answers"));
280
281 // 심볼이 틀렸다면 엔트리 업데이트
282 if (CachedPlayerState->bSymbolWrong)
283 {
284 for (UReadQuestEntryWidget* Entry : SymbolEntries)
285 {
286 if (Entry && Entry->GetChoiceValue() == CachedPlayerState->SelectedSymbol)
287 {
288 Entry->SetWrong(true);
289 Entry->SetSelected(false);
290 SelectedSymbolEntry = nullptr;
291 break;
292 }
293 }
294 }
295
296 // 색상이 틀렸다면 엔트리 업데이트
297 if (CachedPlayerState->bColorWrong)
298 {
299 for (UReadQuestEntryWidget* Entry : ColorEntries)
300 {
301 if (Entry && Entry->GetChoiceValue() == CachedPlayerState->SelectedColor)
302 {
303 Entry->SetWrong(true);
304 Entry->SetSelected(false);
305 SelectedColorEntry = nullptr;
306 break;
307 }
308 }
309 }
