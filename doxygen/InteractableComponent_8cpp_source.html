18UInteractableComponent::UInteractableComponent()
19{
20 PrimaryComponentTick.bCanEverTick = true;
21
22 bIsPickedUp = false;
23 bOriginalSimulatePhysics = false;
24 OriginalCollisionType = ECollisionEnabled::NoCollision;
25
26 // 생성자에서 생성해서 자동 복제
27 DetectionRange = CreateDefaultSubobject<UBoxComponent>(TEXT("DetectionRange"));
28 if (DetectionRange)
29 {
30 DetectionRange->SetBoxExtent(FVector(250.f)); // DetectionDistance 기본값
31 DetectionRange->SetCollisionEnabled(ECollisionEnabled::QueryOnly);
32 DetectionRange->SetCollisionResponseToAllChannels(ECR_Ignore);
33 DetectionRange->SetCollisionResponseToChannel(ECC_Pawn, ECR_Overlap);
34 }
35
36 SetIsReplicatedByDefault(true);
37}

53void UInteractableComponent::InitDetectionRange()
54{
55 // 1. 컴포넌트 등록
56 if (!DetectionRange->IsRegistered())
57 {
58 DetectionRange->RegisterComponent();
59 }
60
61 AActor* Owner = GetOwner();
62
63 // 2. Owner의 RootComponent에 부착
64 if (!DetectionRange->IsAttachedTo(Owner->GetRootComponent()))
65 {
66 DetectionRange->AttachToComponent(
67 Owner->GetRootComponent(),
68 FAttachmentTransformRules::KeepRelativeTransform
69 );
70 }
71
72 // BoxExtent 설정 업데이트
73 FVector BoxExtent(DetectionDistance, DetectionDistance, DetectionDistance);
74 DetectionRange->SetBoxExtent(BoxExtent);
75
76 // Overlap 콜백 바인딩
77 DetectionRange->OnComponentBeginOverlap.AddDynamic(this, &UInteractableComponent::OnDetectionBeginOverlap);
78 DetectionRange->OnComponentEndOverlap.AddDynamic(this, &UInteractableComponent::OnDetectionEndOverlap);
79}

167void UInteractableComponent::Server_PickUp_Implementation(AActor* NewHoldingOwner)
168{
169 if (bIsPickedUp) return;
170
171 HoldingOwner = NewHoldingOwner;
172 if (!HoldingOwner)
173 {
174 PRINTLOG( TEXT("InteractableComponent::PickUp - HoldingOwner is null"));
175 return;
176 }
177
178 // Owner actor의 PrimitiveComponent 찾기
179 UPrimitiveComponent* PrimComp = GetOwnerPrimitiveComponent();
180 if (!PrimComp)
181 return;
182
183 // 원래 상태 저장
184 // Drop 시 원래 상태로 되돌리기 위함
185 bOriginalSimulatePhysics = PrimComp->IsSimulatingPhysics();
186 OriginalCollisionType = PrimComp->GetCollisionEnabled();
187
188 // pick up
189 PrimComp->SetSimulatePhysics(false);
190
191 OnRep_HoldingOwner();
192
193 bIsPickedUp = true;
194 // outline, infowidget 끄기
195 Aluggage* luggage = Cast<Aluggage>(GetOwner());
196 if (luggage)
197 luggage->OutlineOff();
198
199 HideInteractWidget();
200
201 PRINTLOG( TEXT("InteractableComponent::PickUp - %s picked up"), *GetOwner()->GetName());
202}

209void UInteractableComponent::Server_Drop_Implementation()
210{
211 if (!bIsPickedUp) return;
212
213 // outline, infowidget 켜기
214 Aluggage* luggage = Cast<Aluggage>(GetOwner());
215 if (luggage)
216 {
217 luggage->OutlineOn();
218 // luggage->InfoWidgetOn();
219 }
220
221 // ShowInteractWidget();
222
223 // detach
224 GetOwner()->DetachFromActor(FDetachmentTransformRules::KeepWorldTransform);
225
226 UPrimitiveComponent* PrimComp = GetOwnerPrimitiveComponent();
227 if (PrimComp)
228 {
229 PrimComp->SetSimulatePhysics(bOriginalSimulatePhysics);
230 PrimComp->SetCollisionEnabled(OriginalCollisionType);
231 }
232
233 bIsPickedUp = false;
234
235 PRINTLOG( TEXT("InteractableComponent::Drop - %s dropped"), *GetOwner()->GetName());
236}

239UPrimitiveComponent* UInteractableComponent::GetOwnerPrimitiveComponent() const
240{
241 AActor* Owner = GetOwner();
242 if (!Owner)
243 return nullptr;
244
245 UPrimitiveComponent* PrimComp = Cast<UPrimitiveComponent>(Owner->GetRootComponent());
246
247 // RootComponent가 PrimitiveComponent가 아니면, 자식 중에서 찾기
248 if (!PrimComp)
249 {
250 TArray<UActorComponent*> Components;
251 Owner->GetComponents(UPrimitiveComponent::StaticClass(), Components);
252
253 if (Components.Num() > 0)
254 {
255 // 첫 번쩨 primitive component 반환
256 PrimComp = Cast<UPrimitiveComponent>(Components[0]);
257 }
258 }
259
260 return PrimComp;
261}

297void UInteractableComponent::OnDetectionBeginOverlap(
298 UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
299 UPrimitiveComponent* OtherComp, int32 OtherBodyIndex,
300 bool bFromSweep, const FHitResult& SweepResult)
301{
302 if (!OtherActor)
303 return;
304
305 // Player 캐릭터인지 확인
306 ACharacter* Character = Cast<ACharacter>(OtherActor);
307 if (Character && Character->IsPlayerControlled())
308 {
309 // InteractionSystem 찾기 및 등록
310 UInteractionSystem* InteractionSystem = Character->FindComponentByClass<UInteractionSystem>();
311 if (InteractionSystem)
312 {
313 InteractionSystem->RegisterInteractable(this);
314 }
315
316 ShowInteractWidget();
317
318 // PickUp 타입이면 Luggage의 custom widget 사용
319 Aluggage* luggage = Cast<Aluggage>(GetOwner());
320 if (luggage && !bIsPickedUp)
321 {
322 luggage->OutlineOn();
323 // luggage->InfoWidgetOn();
324 }
325
326 PRINTLOG( TEXT("InteractableComponent: Player entered detection range - %s"), *GetOwner()->GetName());
327 }
328}

330void UInteractableComponent::OnDetectionEndOverlap(
331 UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
332 UPrimitiveComponent* OtherComp, int32 OtherBodyIndex)
333{
334 if (!OtherActor)
335 return;
336
337 // Player 캐릭터인지 확인
338 ACharacter* Character = Cast<ACharacter>(OtherActor);
339 if (Character && Character->IsPlayerControlled())
340 {
341 // InteractionSystem에서 등록 해제
342 UInteractionSystem* InteractionSystem = Character->FindComponentByClass<UInteractionSystem>();
343 if (InteractionSystem)
344 {
345 InteractionSystem->UnregisterInteractable(this);
346 }
347
348 PRINTLOG( TEXT("InteractableComponent: Player left detection range - %s"), *GetOwner()->GetName());
349
350 // Widget 끄기
351 FTimerHandle TimerHandle;
352 GetWorld()->GetTimerManager().SetTimer(TimerHandle, FTimerDelegate::CreateLambda([this]
353 {
354 HideInteractWidget();
355
356 Aluggage* luggage = Cast<Aluggage>(GetOwner());
357 if (luggage && !bIsPickedUp)
358 {
359 luggage->OutlineOff();
360 // luggage->InfoWidgetOff();
361 }
362 }), 0.1f, false);
363 }
364}

399void UInteractableComponent::BillboardInteractWidget()
400{
401 // 위젯이 없으면 빌보드화 안 함
402 if (!WidgetComp)
403 return;
404
405 // Visibility 체크 - 보이지 않으면 빌보드화 안 함
406 if (!WidgetComp->IsVisible())
407 return;
408
409 // 카메라 가져오기
410 AActor* Camera = UGameplayStatics::GetPlayerCameraManager(GetWorld(), 0);
411 if (!Camera)
412 return;
413
414 // 카메라를 향하도록 회전 계산
415 FRotator Rotation = UKismetMathLibrary::MakeRotFromXZ( -Camera->GetActorForwardVector(), Camera->GetActorUpVector() );
416 Rotation.Pitch = 0;
417
418 // 위젯 회전 설정
419 WidgetComp->SetWorldRotation(Rotation);
420}
