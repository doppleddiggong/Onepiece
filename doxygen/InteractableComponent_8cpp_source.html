19UInteractableComponent::UInteractableComponent()
20{

54void UInteractableComponent::InitDetectionRange()
55{
56 // 1. 컴포넌트 등록
57 if (!DetectionRange->IsRegistered())
58 {
59 DetectionRange->RegisterComponent();
60 }
61
62 AActor* Owner = GetOwner();
63
64 // 2. Owner의 RootComponent에 부착
65 if (!DetectionRange->IsAttachedTo(Owner->GetRootComponent()))
66 {
67 DetectionRange->AttachToComponent(
68 Owner->GetRootComponent(),
69 FAttachmentTransformRules::KeepRelativeTransform
70 );
71 }
72
73 // BoxExtent 설정 업데이트
74 FVector BoxExtent(DetectionDistance, DetectionDistance, DetectionDistance);
75 DetectionRange->SetBoxExtent(BoxExtent);
76
77 // Overlap 콜백 바인딩
78 DetectionRange->OnComponentBeginOverlap.AddDynamic(this, &UInteractableComponent::OnDetectionBeginOverlap);
79 DetectionRange->OnComponentEndOverlap.AddDynamic(this, &UInteractableComponent::OnDetectionEndOverlap);
80}

220bool UInteractableComponent::Server_PickUp_Validate(AActor* NewHoldingOwner)
221{
222 // 1. NewHoldingOwner가 유효한지
223 if (!NewHoldingOwner)
224 {
225 PRINTLOG(TEXT("Server_PickUp_Validate: NewHoldingOwner is null"));
226 return false;
227 }
228
229 // 2. NewHoldingOwner가 실제 PlayerController에 의해 제어되는지
230 APlayerActor* PlayerActor = Cast<APlayerActor>(NewHoldingOwner);
231 if (!PlayerActor || !PlayerActor->GetController())
232 {
233 PRINTLOG(TEXT("Server_PickUp_Validate: Not a valid PlayerActor or no Controller"));
234 return false;
235 }
236
237 // 3. 이미 픽업된 상태인지
238 if (bIsPickedUp)
239 {
240 PRINTLOG(TEXT("Server_PickUp_Validate: Already picked up"));
241 return false;
242 }
243
244 // 4. HoldingOwner가 이미 있는지 (중복 소유 방지)
245 if (HoldingOwner && HoldingOwner != NewHoldingOwner)
246 {
247 PRINTLOG(TEXT("Server_PickUp_Validate: Already held by %s"), *HoldingOwner->GetName());
248 return false;
249 }
250
251 // 5. 거리 검증 (서버 기준)
252 AActor* Owner = GetOwner();
253 if (!Owner)
254 {
255 PRINTLOG(TEXT("Server_PickUp_Validate: Owner is null"));
256 return false;
257 }
258
259 float Distance = FVector::Dist(Owner->GetActorLocation(), NewHoldingOwner->GetActorLocation());
260 float MaxDistance = DetectionDistance * 1.5f; // 네트워크 지연 고려 1.5배 여유
261
262 if (Distance > MaxDistance)
263 {
264 PRINTLOG(TEXT("Server_PickUp_Validate: Too far! Distance: %.2f, Max: %.2f"), Distance, MaxDistance);
265 return false;
266 }
267
268 return true;
269}

275void UInteractableComponent::Server_PickUp_Implementation(AActor* NewHoldingOwner)
276{
277 // Validation 통과 후 실행
278 HoldingOwner = NewHoldingOwner;
279
280 // Owner actor의 PrimitiveComponent 찾기
281 UPrimitiveComponent* PrimComp = GetOwnerPrimitiveComponent();
282 if (!PrimComp)
283 return;
284
285 // 원래 상태 저장 (드롭 시 복원용)
286 bOriginalSimulatePhysics = PrimComp->IsSimulatingPhysics();
287 OriginalCollisionType = PrimComp->GetCollisionEnabled();
288
289 // 물리 끄기
290 PrimComp->SetSimulatePhysics(false);
291
292 // Attach (OnRep_HoldingOwner에서 처리)
293 OnRep_HoldingOwner();
294
295 // 상태 변경 (복제됨 → OnRep_IsPickedUp 자동 호출)
296 bIsPickedUp = true;
297
298 PRINTLOG(TEXT("Server_PickUp: %s picked up by %s"), *GetOwner()->GetName(), *NewHoldingOwner->GetName());
299}

342void UInteractableComponent::Server_Drop_Implementation()
343{
344 // Detach (OnRep_HoldingOwner에서 처리됨)
345 GetOwner()->DetachFromActor(FDetachmentTransformRules::KeepWorldTransform);
346
347 // 물리 복원
348 UPrimitiveComponent* PrimComp = GetOwnerPrimitiveComponent();
349 if (PrimComp)
350 {
351 PrimComp->SetSimulatePhysics(bOriginalSimulatePhysics);
352 PrimComp->SetCollisionEnabled(OriginalCollisionType);
353 }
354
355 // 상태 변경 (복제됨 → OnRep_IsPickedUp, OnRep_HoldingOwner 자동 호출)
356 bIsPickedUp = false;
357 HoldingOwner = nullptr;
358
359 PRINTLOG(TEXT("Server_Drop: %s dropped"), *GetOwner()->GetName());
360}

363UPrimitiveComponent* UInteractableComponent::GetOwnerPrimitiveComponent() const
364{
365 AActor* Owner = GetOwner();
366 if (!Owner)
367 return nullptr;
368
369 UPrimitiveComponent* PrimComp = Cast<UPrimitiveComponent>(Owner->GetRootComponent());
370
371 // RootComponent가 PrimitiveComponent가 아니면, 자식 중에서 찾기
372 if (!PrimComp)
373 {
374 TArray<UActorComponent*> Components;
375 Owner->GetComponents(UPrimitiveComponent::StaticClass(), Components);
376
377 if (Components.Num() > 0)
378 {
379 // 첫 번쩨 primitive component 반환
380 PrimComp = Cast<UPrimitiveComponent>(Components[0]);
381 }
382 }
383
384 return PrimComp;
385}

427void UInteractableComponent::OnDetectionBeginOverlap(
428 UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
429 UPrimitiveComponent* OtherComp, int32 OtherBodyIndex,
430 bool bFromSweep, const FHitResult& SweepResult)
431{
432 if (!OtherActor)
433 return;
434
435 // Player 캐릭터인지 확인
436 if ( auto Character = Cast<ACharacter>(OtherActor) )
437 {
438 if ( Character->IsPlayerControlled())
439 {
440 // InteractionSystem 찾기 및 등록
441 if (auto InteractionSystem = Character->FindComponentByClass<UInteractionSystem>())
442 InteractionSystem->RegisterInteractable(this);
443
444 bool ShowState = IsWidgetShowEnable(Character);
445 if ( ShowState)
446 ShowInteractWidget();
447
448 // 픽업되지 않은 상태면 Outline 켜기
449 if (!bIsPickedUp)
450 SetOutlineState(true);
451
452 PRINTLOG( TEXT("InteractableComponent: Player entered detection range - %s"), *GetOwner()->GetName());
453 }
454 }
455}

463void UInteractableComponent::OnDetectionEndOverlap(
464 UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
465 UPrimitiveComponent* OtherComp, int32 OtherBodyIndex)
466{
467 if (!OtherActor)
468 return;
469
470 // Player 캐릭터인지 확인
471 ACharacter* Character = Cast<ACharacter>(OtherActor);
472 if (Character && Character->IsPlayerControlled())
473 {
474 // InteractionSystem에서 등록 해제
475 UInteractionSystem* InteractionSystem = Character->FindComponentByClass<UInteractionSystem>();
476 if (InteractionSystem)
477 {
478 InteractionSystem->UnregisterInteractable(this);
479 }
480
481 PRINTLOG( TEXT("InteractableComponent: Player left detection range - %s"), *GetOwner()->GetName());
482
483 // Widget 끄기
484 FTimerHandle TimerHandle;
485 GetWorld()->GetTimerManager().SetTimer(TimerHandle, FTimerDelegate::CreateLambda([this]
486 {
487 HideInteractWidget();
488
489 // 픽업되지 않은 상태면 Outline 끄기
490 if (!bIsPickedUp)
491 SetOutlineState(false);
492 }), 0.1f, false);
493 }
494}

529void UInteractableComponent::BillboardInteractWidget()
530{
531 // 위젯이 없으면 빌보드화 안 함
532 if (!WidgetComp)
533 return;
534
535 // Visibility 체크 - 보이지 않으면 빌보드화 안 함
536 if (!WidgetComp->IsVisible())
537 return;
538
539 // 카메라 가져오기
540 AActor* Camera = UGameplayStatics::GetPlayerCameraManager(GetWorld(), 0);
541 if (!Camera)
542 return;
543
544 // 카메라를 향하도록 회전 계산
545 FRotator Rotation = UKismetMathLibrary::MakeRotFromXZ( -Camera->GetActorForwardVector(), Camera->GetActorUpVector() );
546 Rotation.Pitch = 0;
547
548 // 위젯 회전 설정
549 WidgetComp->SetWorldRotation(Rotation);
550}
