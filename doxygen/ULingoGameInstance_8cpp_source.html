19void ULingoGameInstance::Init()
20{
21 Super::Init();
22
23 // 현재 사용하는 서브시스템을 가져오자
24 IOnlineSubsystem* subsys = Online::GetSubsystem(GetWorld());
25 if (subsys)
26 {
27 // 서브시스템의 인터페이스를 가져오자.
28 sessionInterface = subsys->GetSessionInterface();
29 // 세션 생성 성공시 호출되는 함수 등록
30 sessionInterface->OnCreateSessionCompleteDelegates.AddUObject(
31 this, &ULingoGameInstance::OnCreateSessionComplete);
32 // 세션 조회 성공시 호출되는 함수 등록
33 sessionInterface->OnFindSessionsCompleteDelegates.AddUObject(
34 this, &ULingoGameInstance::OnFindSessionComplete);
35 // 세션 참여 성공시 호출되는 함수 등록
36 sessionInterface->OnJoinSessionCompleteDelegates.AddUObject(
37 this, &ULingoGameInstance::OnJoinSessionComplete);
38 }
39}

41void ULingoGameInstance::Shutdown()
42{
43 // 델리게이트 정리
44 if (sessionInterface.IsValid())
45 {
46 // 세션 델리게이트 바인딩 해제
47 sessionInterface->OnCreateSessionCompleteDelegates.RemoveAll(this);
48 sessionInterface->OnFindSessionsCompleteDelegates.RemoveAll(this);
49 sessionInterface->OnJoinSessionCompleteDelegates.RemoveAll(this);
50
51 PRINTLOG(TEXT("[LingoGameInstance] Session delegates cleared"));
52 }
53
54 // sessionSearch 정리
55 if (sessionSearch.IsValid())
56 {
57 sessionSearch.Reset();
58 }
59
60 // sessionInterface 정리
61 sessionInterface.Reset();
62
63 // selectCharacter 맵 정리
64 selectCharacter.Empty();
65
66 PRINTLOG(TEXT("[LingoGameInstance] Shutdown complete"));
67
68 Super::Shutdown();
69}

169void ULingoGameInstance::CreateMySession(FString displayName)
170{
171 // 세션을 만들기 위한 옵션 담을 변수
172 FOnlineSessionSettings sessionSettings;
173 // 현재 사용중인 서브시스템 이름 가져오자.
174 FName subsysName = Online::GetSubsystem(GetWorld())->GetSubsystemName();
175 // 만약에 서브시스템이 이름이 NULL 이면 Lan 을 이용하게 설정
176 sessionSettings.bIsLANMatch = subsysName.IsEqual(FName(TEXT("NULL")));
177 UE_LOG(LogTemp, Warning, TEXT("서브시스템 : %s"), *subsysName.ToString());
178
179 // Steam 에선 필수 (bUseLobbiesIfAvailable, bUsesPresence)
180 // Lobby 사용 여부 설정
181 sessionSettings.bUseLobbiesIfAvailable = true;
182 // 친구 상태를 확인할 수 있는 여부
183 sessionSettings.bUsesPresence = true;
184
185 // 세션 검색 허용 여부
186 sessionSettings.bShouldAdvertise = true;
187 // 세션 최대 참여 인원 설정 (임시로 2명)
188 sessionSettings.NumPublicConnections = 2;
189 // 커스텀 정보
190 // displayName 을 Base64 로 변환
191 displayName = StringBase64Encode(displayName);
192 sessionSettings.Set(FName("DP_NAME"), displayName, EOnlineDataAdvertisementType::ViaOnlineServiceAndPing);
193 //sessionSettings.Set(FName("MAP_IDX"), 1, EOnlineDataAdvertisementType::ViaOnlineServiceAndPing);
194
195 // sessionSettings 이용해서 세션 생성
196 FUniqueNetIdPtr netId =
197 GetWorld()->GetFirstLocalPlayerFromController()->GetUniqueNetIdForPlatformUser().GetUniqueNetId();
198 sessionInterface->CreateSession(*netId, FName(displayName), sessionSettings);
199 //sessionInterface->CreateSession(0, FName(displayName), sessionSettings);
200
201}

217void ULingoGameInstance::FindOtherSession()
218{
219 UE_LOG(LogTemp, Warning, TEXT("세션 조회 시작"));
220 // sessionSearch 만들자.
221 sessionSearch = MakeShared<FOnlineSessionSearch>();
222 // 현재 사용중인 서브시스템 이름 가져오자.
223 FName subsysName = Online::GetSubsystem(GetWorld())->GetSubsystemName();
224 // 만약에 서브시스템이 이름이 NULL 이면 Lan 을 이용하게 설정
225 sessionSearch->bIsLanQuery = subsysName.IsEqual(FName(TEXT("NULL")));
226 // 어떤 옵션을 기준으로 검색
227 sessionSearch->QuerySettings.Set(SEARCH_LOBBIES, true, EOnlineComparisonOp::Equals);
228 //sessionSearch->QuerySettings.Set(FName("DP_NAME"), FName("Wanted"), EOnlineComparisonOp::Equals);
229 // 검색 갯수
230 sessionSearch->MaxSearchResults = 100;
231 // 위 설정을 가지고 세션 검색
232 sessionInterface->FindSessions(0, sessionSearch.ToSharedRef());
233}

235void ULingoGameInstance::OnFindSessionComplete(bool bWasSuccessful)
236{
237 UE_LOG(LogTemp, Warning, TEXT("세션 조회 끝"));
238 if (bWasSuccessful)
239 {
240 // 검색된 세션 결과들
241 auto results = sessionSearch->SearchResults;
242 for (int32 i = 0; i < results.Num(); i++)
243 {
244 // 방 제목 이름 담을 변수
245 FString displayName;
246 results[i].Session.SessionSettings.Get(FName(TEXT("DP_NAME")), displayName);
247 // displayName 을 UTF8 string 으로 변환
248 displayName = StringBase64Decode(displayName);
249 UE_LOG(LogTemp, Warning, TEXT("세션 - %i, 이름 : %s"), i, *displayName);
250 // onFindComplete 에 들어있는 함수 실행
251 onFindComplete.ExecuteIfBound(i, displayName);
252 }
253 }
254 else
255 {
256 UE_LOG(LogTemp, Warning, TEXT("세션 조회 실패"));
257 }
258
259 onFindComplete.ExecuteIfBound(-1, FString());
260}

293FString ULingoGameInstance::StringBase64Encode(FString str)
294{
295 // str 을 std::string 로 변환
296 std::string utf8String = TCHAR_TO_UTF8(*str);
297 // utf8String 을 uint8 의 Array 변환
298 TArray<uint8> arrayData = TArray<uint8>((uint8*)utf8String.c_str(), utf8String.length());
299
300 return FBase64::Encode(arrayData);
301}

303FString ULingoGameInstance::StringBase64Decode(FString str)
304{
305 TArray<uint8> arrayData;
306 FBase64::Decode(str, arrayData);
307 std::string utf8String((char*)arrayData.GetData(), arrayData.Num());
308 return UTF8_TO_TCHAR(utf8String.c_str());
309}
