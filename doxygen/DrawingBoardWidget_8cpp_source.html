34FReply UDrawingBoardWidget::NativeOnMouseButtonDown(const FGeometry& InGeometry, const FPointerEvent& InMouseEvent)
35{
36 // Check Mouse is in Canvas
37 const FGeometry CanvasGeometry = Image_Canvas->GetCachedGeometry();
38 if (!CanvasGeometry.IsUnderLocation(InMouseEvent.GetScreenSpacePosition()))
39 {
40 return FReply::Unhandled();
41 }
42
43 // Get Mouse Position in Local Image Coordinate System
44 // Save Current MousePos to prevMousePos
45 prevMousePos = GetLocalMousePos(InMouseEvent.GetScreenSpacePosition());
46 bIsDrawing = true;
47
48 // Draw Point Once
49 // PRINT_STRING(TEXT("%s"), *InMouseEvent.GetEffectingButton().GetFName().ToString());
50 if (InMouseEvent.IsMouseButtonDown(EKeys::LeftMouseButton)) // Draw
51 {
52 DrawPoint(GetLocalMousePos(InMouseEvent.GetScreenSpacePosition()), FLinearColor::Black);
53 }
54 else if (InMouseEvent.IsMouseButtonDown(EKeys::RightMouseButton)) // Erase
55 {
56 DrawPoint(GetLocalMousePos(InMouseEvent.GetScreenSpacePosition()), FLinearColor::White);
57 }
58
59 return FReply::Handled();
60}

89void UDrawingBoardWidget::DrawPoint(FVector2D mousePos, FLinearColor drawColor)
90{
91 // Begin Draw Canvas To Render Target
92 UCanvas* canvas = nullptr;
93 FVector2D size;
94 FDrawToRenderTargetContext context;
95 UKismetRenderingLibrary::BeginDrawCanvasToRenderTarget(this, RT_Canvas, canvas, size, context);
96
97 // Set thickness Whether now in Draw or Erase
98 float thickness = (drawColor == FLinearColor::Black) ? 10 : 30;
99 // Draw Box
100 canvas->K2_DrawBox(mousePos, FVector2D(1, 1), thickness, drawColor);
101
102 UKismetRenderingLibrary::EndDrawCanvasToRenderTarget(this, context);
103}

105void UDrawingBoardWidget::DrawLines(FVector2D mousePos, FLinearColor drawColor)
106{
107 // Begin Draw Canvas To Render Target
108 UCanvas* canvas = nullptr;
109 FVector2D size;
110 FDrawToRenderTargetContext context;
111 UKismetRenderingLibrary::BeginDrawCanvasToRenderTarget(this, RT_Canvas, canvas, size, context);
112
113 // Calculate Draw Positions
114 FVector2D currPos = prevMousePos;
115 int32 div = 64;
116 FVector2D drawOffset = (mousePos - currPos) / div;
117
118 // Draw
119 for (int32 i = 1; i < div; ++i)
120 {
121 currPos = prevMousePos + drawOffset * i;
122
123 // Set thickness Whether now in Draw or Erase
124 float thickness = (drawColor == FLinearColor::Black) ? 10 : 30;
125 // Draw Line
126 canvas->K2_DrawLine(prevMousePos, currPos, thickness, drawColor);
127 }
128 prevMousePos = mousePos;
129
130 UKismetRenderingLibrary::EndDrawCanvasToRenderTarget(this, context);
131}

133FVector2D UDrawingBoardWidget::GetLocalMousePos(FVector2D mousePos)
134{
135 // Get Absolute Local Pos
136 const FGeometry& geometry = Image_Canvas->GetCachedGeometry();
137 FVector2D localPos = geometry.AbsoluteToLocal(mousePos);
138
139 // Get Canvas Size
140 const FVector2D canvasSize = geometry.GetLocalSize();
141 // Transform localPos(in Image_Canvas Coord) to RT_Canvas Coord && Clamp upto RT_Canvas' Border
142 localPos.X = FMath::Clamp((localPos.X / canvasSize.X * RT_Canvas->SizeX), 0.f, RT_Canvas->SizeX);
143 localPos.Y = FMath::Clamp((localPos.Y / canvasSize.Y * RT_Canvas->SizeY), 0.f, RT_Canvas->SizeY);
144 return localPos;
145}

167bool UDrawingBoardWidget::SaveRenderTargetToPNG(UTextureRenderTarget2D* RenderTarget, const FString& FullFilePath)
168{
169 FTextureRenderTargetResource* RTResource = RenderTarget->GameThread_GetRenderTargetResource();
170 if (!RTResource)
171 {
172 return false;
173 }
174
175 // Get Width & Height
176 const int32 Width = RenderTarget->SizeX;
177 const int32 Height = RenderTarget->SizeY;
178
179 // Set Bitmap array
180 TArray<FColor> Bitmap;
181 Bitmap.AddUninitialized(Width * Height);
182
183 // Read BGRA8 pixels in RenderTarget
184 RTResource->ReadPixels(Bitmap);
185
186 // PNG Encoder
187 IImageWrapperModule& ImageWrapperModule =
188 FModuleManager::LoadModuleChecked<IImageWrapperModule>("ImageWrapper");
189
190 TSharedPtr<IImageWrapper> ImageWrapper =
191 ImageWrapperModule.CreateImageWrapper(EImageFormat::PNG);
192
193 if (!ImageWrapper.IsValid())
194 {
195 return false;
196 }
197
198 ImageWrapper->SetRaw(
199 Bitmap.GetData(),
200 Bitmap.GetAllocatedSize(),
201 Width,
202 Height,
203 ERGBFormat::BGRA,
204 8
205 );
206
207 // Compress to PNG
208 const TArray64<uint8>& PNGData = ImageWrapper->GetCompressed(100);
209
210 // Make Directory
211 const FString Directory = FPaths::GetPath(FullFilePath);
212 IFileManager::Get().MakeDirectory(*Directory, true);
213
214 return FFileHelper::SaveArrayToFile(PNGData, *FullFilePath);
215}
