62void UDrawingBoardWidget::DrawPoint(FVector2D mousePos)
63{
64 // Begin Draw Canvas To Render Target
65 UCanvas* canvas = nullptr;
66 FVector2D size;
67 FDrawToRenderTargetContext context;
68 UKismetRenderingLibrary::BeginDrawCanvasToRenderTarget(this, RT_Canvas, canvas, size, context);
69
70 // Calculate Draw Positions
71 // 10 : 적당, 20 : 너무 많음(렉)
72 int32 div = 15;
73 FVector2D currPos = prevMousePos;
74 FVector2D drawOffset = (mousePos - currPos) / div;
75
76 // Draw
77 for (int32 i = 1; i < div; ++i)
78 {
79 currPos = prevMousePos + drawOffset * i;
80
81 // Draw Material
82 canvas->K2_DrawMaterial(M_Brush, currPos, FVector2D(20, 20), FVector2D::ZeroVector);
83 }
84 prevMousePos = mousePos;
85
86 UKismetRenderingLibrary::EndDrawCanvasToRenderTarget(this, context);
87}

117bool UDrawingBoardWidget::SaveRenderTargetToPNG(UTextureRenderTarget2D* RenderTarget, const FString& FullFilePath)
118{
119 FTextureRenderTargetResource* RTResource = RenderTarget->GameThread_GetRenderTargetResource();
120 if (!RTResource)
121 {
122 return false;
123 }
124
125 const int32 Width = RenderTarget->SizeX;
126 const int32 Height = RenderTarget->SizeY;
127
128 TArray<FColor> Bitmap;
129 Bitmap.AddUninitialized(Width * Height);
130
131 // Read BGRA8 pixels in RenderTarget
132 RTResource->ReadPixels(Bitmap);
133
134 // PNG Encoder
135 IImageWrapperModule& ImageWrapperModule =
136 FModuleManager::LoadModuleChecked<IImageWrapperModule>("ImageWrapper");
137
138 TSharedPtr<IImageWrapper> ImageWrapper =
139 ImageWrapperModule.CreateImageWrapper(EImageFormat::PNG);
140
141 if (!ImageWrapper.IsValid())
142 {
143 return false;
144 }
145
146 ImageWrapper->SetRaw(
147 Bitmap.GetData(),
148 Bitmap.GetAllocatedSize(),
149 Width,
150 Height,
151 ERGBFormat::BGRA,
152 8
153 );
154
155 const TArray64<uint8>& PNGData = ImageWrapper->GetCompressed(100);
156
157 // Make Directory
158 const FString Directory = FPaths::GetPath(FullFilePath);
159 IFileManager::Get().MakeDirectory(*Directory, true);
160
161 return FFileHelper::SaveArrayToFile(PNGData, *FullFilePath);
162}
