24void UToastWidget::InitTutorMessage()
25{
26 if (TutorMessage)
27 {
28 TutorMessage->SetVisibility(ESlateVisibility::HitTestInvisible);

30 FWidgetAnimationDynamicEvent HideDelegate;

41void UToastWidget::OnTutorMessage(const FText& NewMessage)
42{
43 if (!TutorMessage)
44 return;
45
46 // 기존 타이머가 있으면 클리어
47 if (TutorHideTimerHandle.IsValid())
48 {
49 GetWorld()->GetTimerManager().ClearTimer(TutorHideTimerHandle);
50 }
51
52 if (bIsTutorVisible)
53 {
54 // 표시 중이면 Hide → SetText → Show
55 PendingMessage = NewMessage;
56 bHasPendingMessage = true;
57
58 PlayAnimation(TutorHideAnim);
59 }
60 else
61 {
62 // 숨김 상태면 바로 SetText → Show
63 TutorMessage->SetMessageText(NewMessage);
64
65 if (TutorShowAnim)
66 {
67 PlayAnimation(TutorShowAnim);
68 bIsTutorVisible = true;
69
70 // 자동 Hide 타이머 시작
71 StartTutorHideTimer();
72 }
73 }
74}

76void UToastWidget::OnTutorHideComplete()
77{

80 if (bHasPendingMessage && TutorMessage)
81 {
82 bHasPendingMessage = false;
83 TutorMessage->SetMessageText(PendingMessage);

85 if (TutorShowAnim)
86 {
87 PlayAnimation(TutorShowAnim);
88 bIsTutorVisible = true;
89
90 // 자동 Hide 타이머 시작

96void UToastWidget::StartTutorHideTimer()
97{
98 // 기존 타이머 클리어 (안전성을 위해)
99 if (TutorHideTimerHandle.IsValid())
100 {
101 GetWorld()->GetTimerManager().ClearTimer(TutorHideTimerHandle);
102 }
103
104 // 지정된 시간 후 자동으로 Hide 애니메이션 재생
105 GetWorld()->GetTimerManager().SetTimer(
106 TutorHideTimerHandle,
107 [this]()
108 {
109 if (bIsTutorVisible)
110 {
111 PlayAnimation(TutorHideAnim);
112 }
113 },
114 TutorMessageDisplayDuration,
115 false
116 );
117}

119void UToastWidget::AddItemToBoxList(const TArray<FResultStatData>& InDataList)
120{
121 if (InDataList.Num() == 0)
122 return;
123
124 // 기존 타이머가 있으면 클리어
125 if (ItemAddTimerHandle.IsValid())
126 {
127 GetWorld()->GetTimerManager().ClearTimer(ItemAddTimerHandle);
128 }
129
130 // 대기 리스트에 데이터 복사 및 인덱스 초기화
131 PendingItemDataList = InDataList;
132 CurItemIndex = 0;
133
134 // 첫 번째 아이템은 즉시 추가
135 ProcessNextItem();
136}

153void UToastWidget::ProcessNextItem()
154{
155 if (CurItemIndex >= PendingItemDataList.Num())
156 {
157 // 모든 아이템 추가 완료
158 PendingItemDataList.Empty();
159 CurItemIndex = 0;
160 return;
161 }
162
163 // 현재 인덱스의 아이템 추가
164 AddItemToBoxItem(PendingItemDataList[CurItemIndex]);
165 CurItemIndex++;
166
167 // 다음 아이템이 있으면 타이머 설정
168 if (CurItemIndex < PendingItemDataList.Num())
169 {
170 GetWorld()->GetTimerManager().SetTimer(
171 ItemAddTimerHandle,
172 this,
173 &UToastWidget::ProcessNextItem,
174 ItemAddInterval,
175 false
176 );
177 }
178 else
179 {
180 // 모든 아이템 추가 완료
181 PendingItemDataList.Empty();
182 CurItemIndex = 0;
183 }
184}
