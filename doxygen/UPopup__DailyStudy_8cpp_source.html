33void UPopup_DailyStudy::NativeConstruct()
34{
35 bAllowPlayerControl = true;
36
37 Super::NativeConstruct();
38
39 // 미디어 플레이어 초기화
40 if (!MediaPlayer)
41 {
42 MediaPlayer = NewObject<UMediaPlayer>(this);
43 MediaPlayer->SetLooping(false);
44 MediaPlayer->OnEndReached.AddDynamic(this, &UPopup_DailyStudy::OnVideoFinished);
45 }
46
47 // 미디어 텍스처 생성
48 if (!MediaTexture)
49 {
50 MediaTexture = NewObject<UMediaTexture>(this);
51 MediaTexture->SetMediaPlayer(MediaPlayer);
52 MediaTexture->UpdateResource();
53 }
54
55 // Img_Correct에 미디어 텍스처 바인딩
56 if (Img_Correct && MediaTexture)
57 {
58 FSlateBrush Brush;
59 Brush.SetResourceObject(MediaTexture);
60 Brush.ImageSize = FVector2D(1920, 1080); // 영상 해상도에 맞게 조정
61 Img_Correct->SetBrush(Brush);
62 }
63}

81void UPopup_DailyStudy::InitPopup(const TArray<FWordData>& WordDataArray)
82{
83 // 기본 초기화 (델리게이트 바인딩 등)
84 if (Btn_Close)
85 {
86 Btn_Close->OnButtonClickedEvent.RemoveDynamic(this, &UPopup_DailyStudy::OnClickClose);
87 Btn_Close->OnButtonClickedEvent.AddDynamic(this, &UPopup_DailyStudy::OnClickClose);
88 }
89
90 if (auto DM = UBroadcastManager::Get(GetWorld()))
91 {
92 DM->OnAudioCapture.RemoveDynamic(this, &UPopup_DailyStudy::OnAudioCapture);
93 DM->OnAudioCapture.AddDynamic(this, &UPopup_DailyStudy::OnAudioCapture);
94 }
95
96 if (CountDown_Widget)
97 {
98 CountDown_Widget->OnCountDownFinished.RemoveDynamic(this, &UPopup_DailyStudy::OnCountDownFinished);
99 CountDown_Widget->OnCountDownFinished.AddDynamic(this, &UPopup_DailyStudy::OnCountDownFinished);
100 }
101
102 // 초기화
103 CurIndex = 0;
104 CurrentScore = 0;
105 CorrectAnswerCount = 0;
106
107 QuestionList.Empty();
108 QuestionList.Append(WordDataArray);
109
110 AnswerList.Empty();
111
112 Canvas_Correct->SetVisibility(ESlateVisibility::Hidden);
113 Canvas_Question->SetVisibility(ESlateVisibility::Hidden);
114
115 Txt_CurScore->SetText(FText::FromString(FString::Printf(TEXT("Score : %d"), CurrentScore)));
116
117 // 카운트다운 시작
118 UGameSoundManager::Get(GetWorld())->PlaySound2D(EGameSoundType::Speak_the_Word_in_korean);
119 CountDown_Widget->StartCountDown(3);

158void UPopup_DailyStudy::OnResponseSpeakingsJudges(const FResponseSpeakingJudes& JudgeResult)
159{
160 if (!QuestionList.IsValidIndex(CurIndex))
161 return;
162
163 // 답변 리스트에 추가
164 AnswerList.Add(JudgeResult);
165
166 CurrentScore += JudgeResult.final_overall_score;
167 Txt_CurScore->SetText(FText::FromString(FString::Printf(TEXT("Score : %d"), CurrentScore)));
168
169 // 정답/오답 판별 (50점 미만이면 오답)
170 bLastAnswerCorrect = (JudgeResult.final_overall_score >= 50);
171
172 if ( bLastAnswerCorrect )
173 CorrectAnswerCount++;
174
175 // 정답/오답 화면 표시 (영상 + TTS)
176 ShowCorrectData(bLastAnswerCorrect);
177}

189void UPopup_DailyStudy::UpdateThinkTimer()
190{
191 RemainingThinkTime -= 0.1f;
192 if (RemainingThinkTime <= 0.f)
193 {
194 RemainingThinkTime = 0.f;
195 OnThinkTimeFinished();
196 }
197
198 ProgressBar_RemainTime->SetPercent(RemainingThinkTime / DailyStudyConfig::THINK_TIME);
199 Txt_RemainTime->SetText(FText::FromString(FString::Printf(TEXT("%.1f"), RemainingThinkTime)));
200}

202void UPopup_DailyStudy::OnThinkTimeFinished()
203{
204 GetWorld()->GetTimerManager().ClearTimer(ThinkingTimerHandle);
205
206 // 타임업 = 오답 처리
207 bLastAnswerCorrect = false;
208
209 // 빈 답변 추가 (건너뜀 처리)
210 if (QuestionList.IsValidIndex(CurIndex))
211 {
212 FResponseSpeakingJudes EmptyAnswer;
213 EmptyAnswer.final_overall_score = 0;
214 EmptyAnswer.grammar_score = 0;
215 EmptyAnswer.context_score = 0;
216 EmptyAnswer.final_feedback = TEXT("Time Over");
217 AnswerList.Add(EmptyAnswer);
218 }
219
220 // 오답 화면 표시 (영상 + TTS)
221 ShowCorrectData(false);
222}

259void UPopup_DailyStudy::ShowCorrectData(bool bIsCorrect)
260{
261 if ( auto DM = UDialogManager::Get(GetWorld()))
262 DM->HideToastImmediately();
263
264 Canvas_Correct->SetVisibility(ESlateVisibility::Visible);
265
266 // FWordData에서 한글 단어 표시
267 Txt_Correct->SetText(FText::FromString(QuestionList[CurIndex].Kor));
268
269 Border_Question->SetVisibility(ESlateVisibility::Hidden);
270 Txt_Infomation->SetVisibility(ESlateVisibility::Hidden);
271
272 if (UKLingoNetworkSystem* NetworkSystem = UKLingoNetworkSystem::Get(GetWorld()))

275 NetworkSystem->RequestListenAudio(
276 QuestionList[CurIndex].Kor,

278 );
279 }
280
281 PlayVideo(bIsCorrect);

291void UPopup_DailyStudy::MoveToNextQuestion()
292{
293 Canvas_Correct->SetVisibility(ESlateVisibility::Hidden);
294
295 CurIndex++;
296
297 if (CurIndex < QuestionList.Num())
298 {
299 LoadCurQuestion();
300 }
301 else
302 {
303 // 현재 팝업 닫기
304 OnClickClose();
305
306 // 모든 문제 완료 - DailyResult 팝업 표시
307 if (auto DailyResultPopup = UPopupManager::Get(GetWorld())->ShowPopupAs<UPopup_DailyResult>(EPopupType::DailyResult))
308 {
309 // FDailyStudyResult 생성
310 FDailyStudyResult Result;
311 Result.CurrentScore = CurrentScore;
312 Result.QuestionList = QuestionList;
313 Result.AnswerList = AnswerList;
314
315 DailyResultPopup->InitPopup(Result);
316 }
317 }

320void UPopup_DailyStudy::PlayVideo(bool bIsCorrect)
321{
322 if (!MediaPlayer || !Img_Correct)
323 {
324 // 영상 재생 실패 시 바로 다음 문제로
325 OnVideoFinished();
326 return;
327 }
328
329 // 재생할 영상 소스 선택
330 UMediaSource* SourceToPlay = bIsCorrect ? CorrectVideoSource : WrongVideoSource;
331
332 if (!SourceToPlay)
333 {
334 // 영상이 없으면 바로 다음 문제로 이동
335 OnVideoFinished();
336 return;
337 }
338
339 // Img_Correct 표시
340 Img_Correct->SetVisibility(ESlateVisibility::Visible);
341
342 // 영상 열기 및 재생
343 MediaPlayer->OpenSource(SourceToPlay);
344 MediaPlayer->Play();
345
346 // 시작 위치로 Seek
347 MediaPlayer->Seek(FTimespan::FromSeconds(VideoStartTime));
348
349 // 재생 시간 체크 타이머 시작 (0.1초마다 체크)
350 if (GetWorld())
351 {
352 GetWorld()->GetTimerManager().SetTimer(
353 VideoCheckTimerHandle,
354 this, &UPopup_DailyStudy::CheckVideoPlayback,
355 0.1f,
356 true // 반복
357 );
358 }
359}

361void UPopup_DailyStudy::CheckVideoPlayback()
362{
363 if (!MediaPlayer)
364 return;
365
366 // 현재 재생 시간 확인
367 FTimespan CurrentTime = MediaPlayer->GetTime();
368 float CurrentSeconds = CurrentTime.GetTotalSeconds();
369
370 // 종료 시간 도달 체크
371 if (CurrentSeconds >= VideoEndTime)
372 {
373 // 타이머 중지
374 GetWorld()->GetTimerManager().ClearTimer(VideoCheckTimerHandle);
375
376 // 영상 종료 처리
377 OnVideoFinished();
378 }
379}
