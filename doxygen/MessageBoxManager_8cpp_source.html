19void UMessageBoxManager::ShowMessageBox(EMessageBoxType Type, const FString& Title, const FString& Description)
20{
21 if (Title.IsEmpty() || Description.IsEmpty())
22 return;
23
24 UWorld* World = GetWorld();
25 if (!World)
26 return;
27
28 EnsureWidgetForWorld(World);
29 // 텍스트 설정
30 if (MessageBox)
31 {
32 MessageBox->SetTitle(Title);
33 MessageBox->SetDescription(Description);
34
35 // 타입 별로 다르게 처리
36 switch (Type)
37 {
38 case EMessageBoxType::Info:
39 {
40 // 유저이름 칸 삭제
41 MessageBox->SetNameFieldVisibility(false);
42 }
43 break;
44 case EMessageBoxType::Register:
45 {
46 MessageBox->SetNameFieldVisibility(true);
47
48 MessageBox->OnUserNameRegister.AddDynamic(this, &UMessageBoxManager::RegisterUserName);
49 }
50 break;
51 case EMessageBoxType::LogIn:
52 {
53 MessageBox->SetNameFieldVisibility(true);
54
55 MessageBox->OnUserNameRegister.AddDynamic(this, &UMessageBoxManager::GetUserToken);
56 }
57 break;
58 }
59 }
60}

62void UMessageBoxManager::HideMessageBox()

77void UMessageBoxManager::EnsureWidgetForWorld(UWorld* World)
78{
79 if (World == nullptr || !World->IsGameWorld())
80 return;
81
82 // 위젯이 유효하고 같은 월드이며, 뷰포트에 추가되어 있는지 확인
83 if (IsValid(MessageBox) &&
84 MessageBox->GetWorld() == World &&
85 MessageBox->IsInViewport())
86 {
87 return;
88 }
89
90 // 기존 위젯이 있으면 정리
91 if (MessageBox)
92 {
93 if (MessageBox->IsInViewport())
94 MessageBox->RemoveFromParent();
95
96 MessageBox = nullptr;
97 }
98
99 ULocalPlayer* LocalPlayer = GetLocalPlayer();
100 if (LocalPlayer == nullptr)
101 return;
102
103 APlayerController* PC = LocalPlayer->GetPlayerController(World);
104 if (PC == nullptr)
105 return;
106
107 if (!MessageBoxClass)
108 return;
109
110 UMessageBox* NewWidget = CreateWidget<UMessageBox>(PC, MessageBoxClass);
111 if (!NewWidget)
112 return;
113
114 NewWidget->AddToViewport(GameLayer::ToastManager);
115
116 MessageBox = NewWidget;
117}
