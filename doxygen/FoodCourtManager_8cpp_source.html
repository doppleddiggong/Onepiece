60void AFoodCourtManager::SpawnListenAnswer()
61{
62 // 서버에서만 실행
63 if (!HasAuthority()) return;
64
65 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
66 if (!GS) return;
67
68 const TArray<FScenarioTargetData>& ScenarioData = GS->GetListenScenarioData().target_data;
69
70 // 선택지 중복 안되도록 추리기
71 TSet<FWordInfo> FoodInfos;
72 TSet<FWordInfo> CityInfos;
73
74 for (int32 i=0; i<ScenarioData.Num(); i++)
75 {
76 auto SD = ScenarioData[i];
77
78 CityInfos.Add(SD.word1);
79 FoodInfos.Add(SD.word2);
80 }
81
82 // 음식 선택지 스폰 및 데이터 전달
83 int32 Index = 0;
84 for (const FWordInfo& FoodInfo : FoodInfos)
85 {
86 // 스폰하기
87 FVector SpawnLocation;
88 GetCurrentSpawnLocation(Index, SpawnLocation);
89
90 AListenAnswer* NewActor = GetWorld()->SpawnActor<AListenAnswer>(ListenAnswerClass, SpawnLocation, FRotator::ZeroRotator);
91
92 // 데이터 전달
93 FTimerHandle TimerHandle;
94 GetWorldTimerManager().SetTimer(TimerHandle, FTimerDelegate::CreateLambda([this, NewActor, FoodInfo]
95 {
96 NewActor->AnswerData.AnswerType = EAnswerType::Food;
97 NewActor->AnswerData.word1 = FoodInfo;
98
99 // 로컬 위젯 업데이트
100 NewActor->UpdateMesh();
101 NewActor->UpdateNameWidget();
102
103 }), 1.f, false);
104
105 Index++;
106 }
107
108 // 도시 이름 선택지 스폰 및 데이터 전달
109 for (const FWordInfo& CityInfo : CityInfos)
110 {
111 // 스폰하기
112 FVector SpawnLocation;
113 GetCurrentSpawnLocation(Index, SpawnLocation);
114
115 AListenAnswer* NewActor = GetWorld()->SpawnActor<AListenAnswer>(ListenAnswerClass, SpawnLocation, FRotator::ZeroRotator);
116
117 // 데이터 전달
118 FTimerHandle TimerHandle;
119 GetWorldTimerManager().SetTimer(TimerHandle, FTimerDelegate::CreateLambda([this, NewActor, CityInfo]
120 {
121 NewActor->AnswerData.AnswerType = EAnswerType::City;
122 NewActor->AnswerData.word1 = CityInfo;
123
124 // 로컬 위젯 업데이트
125 NewActor->UpdateMesh();
126 NewActor->UpdateNameWidget();
127
128 }), 1.f, false);
129
130 Index++;
131 }
132}

134void AFoodCourtManager::GetCurrentSpawnLocation(int32 Index, FVector& OutSpawnLocation)
135{
136 // 2열로 배치
137 int32 Row = Index / 2;
138 int32 Col = Index % 2;
139
140 FVector Offset = FVector(Row*SpawnDistance, Col*SpawnDistance, 0.0f);
141
142 OutSpawnLocation = AnswerSpawnLocation + Offset;
143}
