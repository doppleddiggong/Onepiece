56void AFoodCourtManager::SpawnListenAnswer()
57{
58 // 서버에서만 실행
59 if (!HasAuthority()) return;
60
61 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
62 if (!GS) return;
63
64 const TArray<FScenarioTargetData>& ScenarioData = GS->GetListenScenarioData().target_data;
65
66 // 선택지 중복 안되도록 추리기
67 TSet<FWordInfo> FoodInfos;
68 TSet<FWordInfo> CityInfos;
69
70 for (int32 i=0; i<ScenarioData.Num(); i++)
71 {
72 auto SD = ScenarioData[i];
73
74 CityInfos.Add(SD.word1);
75 FoodInfos.Add(SD.word2);
76 }
77
78 // 음식 선택지 스폰 및 데이터 전달
79 int32 Index = 0;
80 for (const FWordInfo& FoodInfo : FoodInfos)
81 {
82 // 스폰하기
83 FVector SpawnLocation;
84 GetCurrentSpawnLocation(Index, FoodSpawnLocation, 0, SpawnLocation);
85
86 AListenAnswer* NewActor = GetWorld()->SpawnActor<AListenAnswer>(ListenAnswerClass, SpawnLocation, FRotator::ZeroRotator);
87 SpawnedListenAnswers.Add(NewActor);
88
89 // 데이터 전달
90 FTimerHandle TimerHandle;
91 GetWorldTimerManager().SetTimer(TimerHandle, FTimerDelegate::CreateLambda([this, NewActor, FoodInfo]
92 {
93 NewActor->AnswerData.AnswerType = EAnswerType::Food;
94 NewActor->AnswerData.word1 = FoodInfo;
95
96 // 로컬 위젯 업데이트
97 NewActor->UpdateMesh();
98 NewActor->UpdateNameWidget();
99
100 }), 1.f, false);
101
102 Index++;
103 }
104
105 Index = 0;
106 // 도시 이름 선택지 스폰 및 데이터 전달
107 for (const FWordInfo& CityInfo : CityInfos)
108 {
109 // 스폰하기
110 FVector SpawnLocation;
111 GetCurrentSpawnLocation(Index, CitySpawnLocation, 3, SpawnLocation);
112
113 AListenAnswer* NewActor = GetWorld()->SpawnActor<AListenAnswer>(ListenAnswerClass, SpawnLocation, FRotator::ZeroRotator);
114 SpawnedListenAnswers.Add(NewActor);
115
116 // 데이터 전달
117 FTimerHandle TimerHandle;
118 GetWorldTimerManager().SetTimer(TimerHandle, FTimerDelegate::CreateLambda([this, NewActor, CityInfo]
119 {
120 NewActor->AnswerData.AnswerType = EAnswerType::City;
121 NewActor->AnswerData.word1 = CityInfo;
122
123 // 로컬 위젯 업데이트
124 NewActor->UpdateMesh();
125 NewActor->UpdateNameWidget();
126
127 }), 1.f, false);
128
129 Index++;
130 }
131}

133void AFoodCourtManager::GetCurrentSpawnLocation(int32 Index, FVector InitialLocation, int32 Dir, FVector& OutSpawnLocation)
134{
135 // 2열로 배치
136 int32 Row = Index / 2;
137 int32 Col = Index % 2;
138
139 // Dir 값에 따라 스폰 방향 변경
140 // 0: 전방-우측, 1: 전방-좌측, 2: 후방-우측, 3: 후방-좌측
141 FVector Offset;
142
143 switch (Dir)
144 {
145 case 0:
146 Offset = FVector(Row*SpawnDistance, Col*SpawnDistance, 0.0f);
147 break;
148 case 1:
149 Offset = FVector(-Col*SpawnDistance, Row*SpawnDistance, 0.0f);
150 break;
151 case 2:
152 Offset = FVector(-Row*SpawnDistance, -Col*SpawnDistance, 0.0f);
153 break;
154 case 3:
155 Offset = FVector(Col*SpawnDistance, -Row*SpawnDistance, 0.0f);
156 break;
157 default:
158 Offset = FVector(Row*SpawnDistance, Col*SpawnDistance, 0.0f);
159 break;
160 }
161
162 OutSpawnLocation = InitialLocation + Offset;
163}
