60void AFoodCourtManager::SetFoodCourtInfo()
61{
62 // 서버에서만 실행
63 if (!HasAuthority()) return;
64
65 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
66 if (!GS) return;
67
68 const TArray<FScenarioTargetData>& ScenarioData = GS->GetListenScenarioData().target_data;
69
70 for (int32 i=0; i<ScenarioData.Num(); i++)
71 {
72 auto SD = ScenarioData[i];
73
74 // 1초 타이머 - 액터들이 클라이언트에 리플리케이트되길 대기
75 FTimerHandle TimerHandle;
76 GetWorldTimerManager().SetTimer(TimerHandle, FTimerDelegate::CreateLambda([this, i, SD]
77 {
78 // 푸드코트 식당 이름 지정
79 ACityName* CityName = FindCityNameByIdx(i);
80 if (CityName)
81 {
82 CityName->SetCityName(SD.word2.name);
83 // 서버는 OnRep이 호출되지 않으므로 직접 위젯 업데이트
84 CityName->OnRep_FoodCourtInfo();
85 }
86
87 // 키오스크 지정
88 AOrderKiosk* CurrentKiosk = FindKioskNameByIdx(i);
89 if (CurrentKiosk)
90 {
91 CurrentKiosk->FoodCourtIdx = i;
92 CurrentKiosk->FoodData.word1 = SD.word1;
93 CurrentKiosk->FoodData.word2 = SD.word2;
94
95 // 서버 로컬 위젯 업데이트
96 CurrentKiosk->UpdateInteractableWidget(SD.word2.name);
97 }
98
99 }), 1.f, false);
100 }
101}
