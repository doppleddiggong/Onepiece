33void AFoodCourtManager::SetFoodCourtInfo()
34{
35 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
36 if (!GS) return;
37
38 const TArray<FScenarioTargetData>& ScenarioData = GS->GetListenScenarioData().target_data;
39
40 for (int32 i=0; i<ScenarioData.Num(); i++)
41 {
42 auto SD = ScenarioData[i];
43
44 // 푸드코트 식당 이름 지정
45 ACityName* CityName = FindCityName(i);
46 if (!CityName)
47 {
48 UE_LOG(LogTemp, Warning, TEXT("FindCityName failed for index %d"), i);
49 continue;
50 }
51
52 if (!CityName->WidgetComp)
53 {
54 UE_LOG(LogTemp, Warning, TEXT("CityName->WidgetComp is null for index %d"), i);
55 continue;
56 }
57
58 UUserWidget* CityNameWidget = CityName->WidgetComp->GetWidget();
59 if (UCityNameWidget* CNW = Cast<UCityNameWidget>(CityNameWidget))
60 {
61 CNW->SetCityName(SD.word1.name);
62 }
63 }
64}

66ACityName* AFoodCourtManager::FindCityName(int32 InIdx)
67{
68 TArray<AActor*> CityNames;
69 UGameplayStatics::GetAllActorsOfClass(GetWorld(), ACityName::StaticClass(), CityNames);
70
71 for (auto CityName : CityNames)
72 {
73 if (ACityName* CN = Cast<ACityName>(CityName))
74 {
75 if (CN->Index == InIdx)
76 {
77 return CN;
78 }
79 }
80 }
81 return nullptr;
82}
