82void ALingoGameMode::HandleLuggageSelection(APlayerState* Player, Aluggage* luggage)
83{
84 if (!HasAuthority())
85 return;
86
87 if (!Player || !luggage)
88 return;
89
90 ALingoPlayerState* LingoPlayerState = Cast<ALingoPlayerState>(Player);
91 if (!LingoPlayerState)
92 return;
93
94 // 정답 판정
95 if ( ValidateAnswer(LingoPlayerState, luggage) )
96 {
97 HandleCorrectAnswer(LingoPlayerState);
98 }
99 else
100 {
101 // 틀린 항목 확인
102 bool bSymbolCorrect = (LingoPlayerState->SelectedWord1 == luggage->Target1);
103 bool bColorCorrect = (LingoPlayerState->SelectedWord2 == luggage->Target2);
104
105 HandleWrongAnswer(LingoPlayerState, bSymbolCorrect, bColorCorrect);
106 }
107}

109bool ALingoGameMode::ValidateAnswer(ALingoPlayerState* Player, Aluggage* Luggage)
110{
111 if (!Player || !Luggage)
112 return false;
113
114 const bool bSymbolCorrect = (Player->SelectedWord1 == Luggage->Target1);
115 const bool bColorCorrect = (Player->SelectedWord2 == Luggage->Target2);
116
117 PRINTLOG(TEXT("[GameMode] ValidateAnswer - Symbol: %s vs %s (%s), Color: %s vs %s (%s)"),
118 *Player->SelectedWord1, *Luggage->Target1, bSymbolCorrect ? TEXT("Correct") : TEXT("Wrong"),
119 *Player->SelectedWord2, *Luggage->Target2, bColorCorrect ? TEXT("Correct") : TEXT("Wrong"));
120
121 return bSymbolCorrect && bColorCorrect;
122}

140void ALingoGameMode::HandleWrongAnswer(ALingoPlayerState* Player, bool bSymbolCorrect, bool bColorCorrect)
141{
142 if (!HasAuthority())
143 return;
144
145 if (!Player)
146 return;
147
148 ALingoGameState* LingoGameState = GetGameState<ALingoGameState>();
149 if (!LingoGameState)
150 return;
151
152 // 타이머 패널티
153 constexpr float Penalty = 30.f;
154 LingoGameState->DecreaseMissionTimer(Penalty);
155
156 // 틀린 항목 판정
157 if (!bSymbolCorrect)
158 {
159 Player->SelectedWord1 = TEXT("");
160 Player->bWrongWord1 = true;
161 PRINTLOG(TEXT("[GameMode] Symbol was wrong - Cleared selection"));
162 }
163
164 if (!bColorCorrect)
165 {
166 Player->SelectedWord2 = TEXT("");
167 Player->bWrongWord2 = true;
168 PRINTLOG(TEXT("[GameMode] Color was wrong - Cleared selection"));
169 }
170
171 // 시도 횟수 증가
172 Player->AttemptCount++;
173
174 PRINTLOG(TEXT("[GameMode] HandleWrongAnswer - Attempt count: %d"), Player->AttemptCount);
175}
