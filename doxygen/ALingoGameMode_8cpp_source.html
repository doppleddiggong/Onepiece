31void ALingoGameMode::BeginReadQuest(int32 InStageIndex, const FResponseScenario& InResponseData)
32{
33 if (!HasAuthority())
34 return;
35
36 // --- 1. 역할 할당 로직 ---
37 auto PSList = ULingoGameHelper::GetLingoPlayerStateList(GetWorld());
38 const int32 PlayerCount = PSList.Num();
39 PRINTLOG(TEXT("[GameMode] BeginReadQuest - Player count: %d"), PlayerCount);
40
41 if (PlayerCount == 1)
42 {
43 // 싱글 플레이: Both
44 if (PSList.IsValidIndex(0))
45 {
46 PSList[0]->QuestRole = EQuestRole::Both;
47 PRINTLOG(TEXT("[GameMode] Player 0 assigned role: Both"));
48 }
49 }
50 else if (PlayerCount >= 2)
51 {
52 // 멀티 플레이: OnlyQuestion1, OnlyQuestion2
53 if (PSList.IsValidIndex(0))
54 {
55 PSList[0]->QuestRole = EQuestRole::OnlyQuestion1;
56 PRINTLOG(TEXT("[GameMode] Player 0 assigned role: OnlyQuestion1"));
57 }
58 if (PSList.IsValidIndex(1))
59 {
60 PSList[1]->QuestRole = EQuestRole::OnlyQuestion2;
61 PRINTLOG(TEXT("[GameMode] Player 1 assigned role: OnlyQuestion2"));
62 }
63 }
64
65 // --- 2. GameState에 데이터 설정 ---
66 if (auto GS = GetGameState<ALingoGameState>())
67 {
68 GS->SetStageData(InStageIndex, InResponseData);
69 PRINTLOG(TEXT("[GameMode] SetStageData called on GameState."));
70 }
71 else
72 {
73 PRINTLOG(TEXT("[GameMode] BeginReadQuest - GameState is null"));
74 }
75}

77void ALingoGameMode::HandleCarrierSelection(APlayerState* Player, Aluggage* Carrier)
78{
79 if (!HasAuthority())
80 return;
81
82 if (!Player || !Carrier)
83 {
84 PRINTLOG(TEXT("[GameMode] HandleCarrierSelection - Invalid parameters"));
85 return;
86 }
87
88 ALingoPlayerState* LingoPlayerState = Cast<ALingoPlayerState>(Player);
89 if (!LingoPlayerState)
90 {
91 PRINTLOG(TEXT("[GameMode] HandleCarrierSelection - PlayerState is not ALingoPlayerState"));
92 return;
93 }
94
95 // 정답 판정
96 bool bIsCorrect = ValidateAnswer(LingoPlayerState, Carrier);
97
98 if (bIsCorrect)
99 {
100 HandleCorrectAnswer(LingoPlayerState);
101 }
102 else
103 {
104 // 틀린 항목 확인
105 bool bSymbolCorrect = (LingoPlayerState->SelectedWord1 == Carrier->Target1);
106 bool bColorCorrect = (LingoPlayerState->SelectedWord2 == Carrier->Target2);
107
108 HandleWrongAnswer(LingoPlayerState, bSymbolCorrect, bColorCorrect);
109 }
110}

112bool ALingoGameMode::ValidateAnswer(ALingoPlayerState* Player, Aluggage* Carrier)
113{
114 if (!Player || !Carrier)
115 return false;
116
117 bool bSymbolCorrect = (Player->SelectedWord1 == Carrier->Target1);
118 bool bColorCorrect = (Player->SelectedWord2 == Carrier->Target2);
119
120 PRINTLOG(TEXT("[GameMode] ValidateAnswer - Symbol: %s vs %s (%s), Color: %s vs %s (%s)"),
121 *Player->SelectedWord1, *Carrier->Target1, bSymbolCorrect ? TEXT("Correct") : TEXT("Wrong"),
122 *Player->SelectedWord2, *Carrier->Target2, bColorCorrect ? TEXT("Correct") : TEXT("Wrong"));
123
124 return bSymbolCorrect && bColorCorrect;
125}

127void ALingoGameMode::HandleCorrectAnswer(ALingoPlayerState* Player)
128{
129 if (!HasAuthority() || !Player)
130 return;
131
132 ALingoGameState* LingoGameState = GetGameState<ALingoGameState>();
133 if (!LingoGameState)
134 return;
135
136 // QuestResult 업데이트
137 LingoGameState->QuestResult.bSuccess = true;
138 LingoGameState->QuestResult.RemainTime = LingoGameState->GetRemainMissionTime();
139 LingoGameState->QuestResult.AttemptCount = Player->AttemptCount;
140 LingoGameState->QuestResult.SelectedSymbol = Player->SelectedWord1;
141 LingoGameState->QuestResult.SelectedColor = Player->SelectedWord2;
142
143 // 성공 플래그 설정
144 LingoGameState->bQuestSuccess = true;
145
146 // 타이머 중지
147 LingoGameState->StopMissionTimer();
148
149 PRINTLOG(TEXT("[GameMode] HandleCorrectAnswer - Quest completed successfully"));
150
151 // BroadcastManager를 통해 성공 이벤트 브로드캐스트
152 if (UBroadcastManager* BroadcastManager = UBroadcastManager::Get(GetWorld()))
153 {
154 // 추후 BroadcastManager에 퀘스트 성공 이벤트 추가 필요
155 }
156}

158void ALingoGameMode::HandleWrongAnswer(ALingoPlayerState* Player, bool bSymbolCorrect, bool bColorCorrect)
159{
160 if (!HasAuthority() || !Player)
161 return;
162
163 ALingoGameState* LingoGameState = GetGameState<ALingoGameState>();
164 if (!LingoGameState)
165 return;
166
167 // 타이머 패널티 (30초 감소)
168 const float Penalty = 30.f;
169 LingoGameState->RemainMissionTime = FMath::Max(0.f, LingoGameState->RemainMissionTime - Penalty);
170
171 PRINTLOG(TEXT("[GameMode] HandleWrongAnswer - Penalty applied: %.0f seconds, Remaining: %.0f seconds"),
172 Penalty, LingoGameState->RemainMissionTime);
173
174 // 틀린 항목 판정
175 if (!bSymbolCorrect)
176 {
177 Player->SelectedWord1 = TEXT("");
178 Player->bWrongWord1 = true;
179 PRINTLOG(TEXT("[GameMode] Symbol was wrong - Cleared selection"));
180 }
181
182 if (!bColorCorrect)
183 {
184 Player->SelectedWord2 = TEXT("");
185 Player->bWrongWord2 = true;
186 PRINTLOG(TEXT("[GameMode] Color was wrong - Cleared selection"));
187 }
188
189 // 시도 횟수 증가
190 Player->AttemptCount++;
191
192 PRINTLOG(TEXT("[GameMode] HandleWrongAnswer - Attempt count: %d"), Player->AttemptCount);
193}
