30UClass* ALingoGameMode::GetDefaultPawnClassForController_Implementation(AController* InController)
31{
32 // GameInstance에서 PlayerController의 Role 조회
33 APlayerController* PC = Cast<APlayerController>(InController);
34 if (!PC)
35 {
36 PRINTLOG( TEXT("[HouseGameMode] Controller is not a PlayerController, using default"));
37 return Super::GetDefaultPawnClassForController(InController);
38 }
39
40 ULingoGameInstance* GameInstance = GetGameInstance<ULingoGameInstance>();
41 if (!GameInstance)
42 {
43 PRINTLOG( TEXT("[HouseGameMode] FireGameInstance not found! Using default pawn."));
44 return Super::GetDefaultPawnClassForController(InController);
45 }
46
47 // Role 조회
48 EPlayerRole CurrentRole = GameInstance->GetPlayerRole(PC);
49
50 // Role에 따라 Pawn Class 결정
51 UClass* SelectedPawnClass = nullptr;
52
53 switch (CurrentRole)
54 {
55 case EPlayerRole::Firefighter:
56 SelectedPawnClass = FirefighterPawnClass;
57 PRINTLOG( TEXT("[HouseGameMode] Spawning Firefighter pawn for %s"), *PC->GetName());
58 break;
59
60 case EPlayerRole::Citizen:
61 SelectedPawnClass = CitizenPawnClass;
62 PRINTLOG( TEXT("[HouseGameMode] Spawning Citizen pawn for %s"), *PC->GetName());
63 break;
64
65 default:
66 PRINTLOG( TEXT("[HouseGameMode] No role assigned for %s, using Citizen as default"), *PC->GetName());
67 SelectedPawnClass = CitizenPawnClass;
68 break;
69 }
70
71 // Pawn Class가 설정되지 않았으면 기본값 사용
72 if (!SelectedPawnClass)
73 {
74 PRINTLOG( TEXT("[HouseGameMode] Selected pawn class is null! Using default."));
75 return Super::GetDefaultPawnClassForController(InController);
76 }
77
78 // PlayerState에도 Role 저장 (동기화)
79 ALingoPlayerState* HousePS = PC->GetPlayerState<ALingoPlayerState>();
80 if (HousePS && CurrentRole != EPlayerRole::None)
81 {
82 HousePS->PlayerRole = CurrentRole;
83 PRINTLOG( TEXT("[HouseGameMode] Set PlayerState role to %s"),
84 *UEnum::GetValueAsString(CurrentRole));
85 }
86
87 return SelectedPawnClass;
88}

173void ALingoGameMode::StartMission(int32 MissionIndex)
174{
175 ALingoGameState* HouseGS = GetGameState<ALingoGameState>();
176 if (!HouseGS) return;
177
178 if (MissionIndex >= TotalMissions) return;
179
180 //ChangeGamePhase(EGamePhase::MissionStart);
181
182 // 미션 시작
183 HouseGS->CurrentMissionIndex = MissionIndex;
184
185 // 모든 플레이어 미션 상태 리셋
186 for (APlayerState* PS : HouseGS->PlayerArray)
187 {
188 ALingoPlayerState* HousePS = Cast<ALingoPlayerState>(PS);
189 if (HousePS)
190 {
191 HousePS->ResetForNextMission();
192 }
193 }
194
195 ChangeGamePhase(EGamePhase::MissionStart);
196}

213void ALingoGameMode::AdvanceToNextMission()
214{
215 ALingoGameState* HouseGS = GetGameState<ALingoGameState>();
216 if (!HouseGS) return;
217
218 int32 NextMissionIndex = HouseGS->CurrentMissionIndex + 1;
219
220 if (NextMissionIndex >= TotalMissions)
221 {
222 // 승리!
223 ChangeGamePhase(EGamePhase::Outro);
224
225 GetWorldTimerManager().ClearTimer(MissionTimerHandle);
226 return;
227 }
228
229 // 다음 미션 시작
230 ChangeGamePhase(EGamePhase::MissionComplete);
231
232 FTimerHandle TimerHandle;
233 GetWorldTimerManager().SetTimer(TimerHandle, FTimerDelegate::CreateLambda([this, NextMissionIndex]
234 {
235 StartMission(NextMissionIndex);
236
237 }),1.f, false);
238}

269void ALingoGameMode::SetAllPlayersStart()
270{
271 for (FConstPlayerControllerIterator It = GetWorld()->GetPlayerControllerIterator(); It; ++It)
272 {
273 APlayerController* PC = It->Get();
274 if (!PC || !PC->GetPawn()) continue;
275
276 ALingoPlayerState* HousePS = PC->GetPlayerState<ALingoPlayerState>();
277 if (!HousePS) continue;
278
279 EPlayerRole CurrentRole = HousePS->PlayerRole;
280 FName PlayerStartTag;
281
282 // Role에 따라 찾을 Tag 결정
283 switch (CurrentRole)
284 {
285 case EPlayerRole::Firefighter:
286 PlayerStartTag = FName("Firefighter");
287 break;
288 case EPlayerRole::Citizen:
289 PlayerStartTag = FName("Citizen");
290 break;
291 default:
292 PRINTLOG(TEXT("[HouseGameMode] No role assigned, skipping spawn"));
293 continue;
294 }
295
296 // Tag로 PlayerStart 찾기
297 TArray<AActor*> FoundStarts;
298 UGameplayStatics::GetAllActorsOfClassWithTag(this, APlayerStart::StaticClass(), PlayerStartTag, FoundStarts);
299
300 if (FoundStarts.Num() > 0)
301 {
302 APlayerStart* PlayerStart = Cast<APlayerStart>(FoundStarts[0]);
303 if (PlayerStart)
304 {
305 PC->GetPawn()->SetActorTransform(PlayerStart->GetActorTransform());
306 PC->GetPawn()->SetActorHiddenInGame(false);
307 }
308 }
309 else
310 {
311 PRINTLOG(TEXT("[HouseGameMode] PlayerStart with tag '%s' not found!"), *PlayerStartTag.ToString());
312 }
313 }
314}
