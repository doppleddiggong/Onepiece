155void ALingoGameMode::StartMission(int32 MissionIndex)
156{
157 ALingoGameState* HouseGS = GetGameState<ALingoGameState>();
158 if (!HouseGS) return;
159
160 if (MissionIndex >= TotalMissions) return;
161
162 // 미션 시작
163 HouseGS->CurrentMissionIndex = MissionIndex;
164
165 // 모든 플레이어 미션 상태 리셋
166 for (APlayerState* PS : HouseGS->PlayerArray)
167 {
168 ALingoPlayerState* HousePS = Cast<ALingoPlayerState>(PS);
169 if (HousePS)
170 {
171 HousePS->ResetForNextMission();
172 }
173 }
174
175 ChangeGamePhase(EGamePhase::MissionStart);
176}

194void ALingoGameMode::AdvanceToNextMission()
195{
196 ALingoGameState* HouseGS = GetGameState<ALingoGameState>();
197 if (!HouseGS) return;
198
199 int32 NextMissionIndex = HouseGS->CurrentMissionIndex + 1;
200
201 if (NextMissionIndex >= TotalMissions)
202 {
203 // 승리!
204 ChangeGamePhase(EGamePhase::ChapterComplete);
205
206 GetWorldTimerManager().ClearTimer(MissionTimerHandle);
207 return;
208 }
209
210 // 다음 미션 시작
211 ChangeGamePhase(EGamePhase::MissionComplete);
212
213 FTimerHandle TimerHandle;
214 GetWorldTimerManager().SetTimer(TimerHandle, FTimerDelegate::CreateLambda([this, NextMissionIndex]
215 {
216 StartMission(NextMissionIndex);
217
218 }),1.f, false);
219}

232void ALingoGameMode::ChangeGamePhase(EGamePhase NewState)
233{
234 ALingoGameState* HouseGS = GetGameState<ALingoGameState>();
235 if (!HouseGS) return;
236
237 // 클라이언트쪽 상태 변경
238 HouseGS->CurrentPhase = NewState;
239
240 // 서버쪽 상태 변경 브로드캐스트
241 if (HasAuthority())
242 {
243 HouseGS->OnPhaseChanged.Broadcast(NewState);
244 HouseGS->PreviousPhase = NewState;
245 }
