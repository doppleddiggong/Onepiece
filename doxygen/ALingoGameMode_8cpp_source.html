28void ALingoGameMode::StartReadQuest()
29{
30 if (!HasAuthority())
31 return;
32
33 ALingoGameState* LingoGameState = GetGameState<ALingoGameState>();
34 if (!LingoGameState)
35 {
36 PRINTLOG(TEXT("[GameMode] StartReadQuest - GameState is null"));
37 return;
38 }
39
40 // 플레이어 수 확인 (싱글/멀티 판정)
41 int32 PlayerCount = 0;
42 TArray<ALingoPlayerState*> LingoPlayerStates;
43
44 for (FConstPlayerControllerIterator It = GetWorld()->GetPlayerControllerIterator(); It; ++It)
45 {
46 if (APlayerController* PC = It->Get())
47 {
48 if (ALingoPlayerState* PS = PC->GetPlayerState<ALingoPlayerState>())
49 {
50 LingoPlayerStates.Add(PS);
51 PlayerCount++;
52 }
53 }
54 }
55
56 PRINTLOG(TEXT("[GameMode] StartReadQuest - Player count: %d"), PlayerCount);
57
58 // 역할 할당
59 if (PlayerCount == 1)
60 {
61 // 싱글 플레이: Both
62 if (LingoPlayerStates.Num() > 0)
63 {
64 LingoPlayerStates[0]->QuestRole = EReadQuestRole::Both;
65 PRINTLOG(TEXT("[GameMode] Player 0 assigned role: Both"));
66 }
67 }
68 else if (PlayerCount >= 2)
69 {
70 // 멀티 플레이: OnlyQuestion1, OnlyQuestion2
71 if (LingoPlayerStates.Num() >= 1)
72 {
73 LingoPlayerStates[0]->QuestRole = EReadQuestRole::OnlyQuestion1;
74 PRINTLOG(TEXT("[GameMode] Player 0 assigned role: OnlyQuestion1"));
75 }
76 if (LingoPlayerStates.Num() >= 2)
77 {
78 LingoPlayerStates[1]->QuestRole = EReadQuestRole::OnlyQuestion2;
79 PRINTLOG(TEXT("[GameMode] Player 1 assigned role: OnlyQuestion2"));
80 }
81 }
82
83 // BroadcastManager를 통해 퀘스트 시작 이벤트 브로드캐스트
84 if (UBroadcastManager* BroadcastManager = UBroadcastManager::Get(GetWorld()))
85 {
86 // 추후 BroadcastManager에 퀘스트 시작 이벤트 추가 필요
87 PRINTLOG(TEXT("[GameMode] StartReadQuest - Quest started"));
88 }
89}

91void ALingoGameMode::HandleCarrierSelection(APlayerState* Player, Aluggage* Carrier)
92{
93 if (!HasAuthority())
94 return;
95
96 if (!Player || !Carrier)
97 {
98 PRINTLOG(TEXT("[GameMode] HandleCarrierSelection - Invalid parameters"));
99 return;
100 }
101
102 ALingoPlayerState* LingoPlayerState = Cast<ALingoPlayerState>(Player);
103 if (!LingoPlayerState)
104 {
105 PRINTLOG(TEXT("[GameMode] HandleCarrierSelection - PlayerState is not ALingoPlayerState"));
106 return;
107 }
108
109 // 정답 판정
110 bool bIsCorrect = ValidateAnswer(LingoPlayerState, Carrier);
111
112 if (bIsCorrect)
113 {
114 HandleCorrectAnswer(LingoPlayerState);
115 }
116 else
117 {
118 // 틀린 항목 확인
119 bool bSymbolCorrect = (LingoPlayerState->SelectedWord1 == Carrier->Target1);
120 bool bColorCorrect = (LingoPlayerState->SelectedWord2 == Carrier->Target2);
121
122 HandleWrongAnswer(LingoPlayerState, bSymbolCorrect, bColorCorrect);
123 }
124}

126bool ALingoGameMode::ValidateAnswer(ALingoPlayerState* Player, Aluggage* Carrier)
127{
128 if (!Player || !Carrier)
129 return false;
130
131 bool bSymbolCorrect = (Player->SelectedWord1 == Carrier->Target1);
132 bool bColorCorrect = (Player->SelectedWord2 == Carrier->Target2);
133
134 PRINTLOG(TEXT("[GameMode] ValidateAnswer - Symbol: %s vs %s (%s), Color: %s vs %s (%s)"),
135 *Player->SelectedWord1, *Carrier->Target1, bSymbolCorrect ? TEXT("Correct") : TEXT("Wrong"),
136 *Player->SelectedWord2, *Carrier->Target2, bColorCorrect ? TEXT("Correct") : TEXT("Wrong"));
137
138 return bSymbolCorrect && bColorCorrect;
139}

141void ALingoGameMode::HandleCorrectAnswer(ALingoPlayerState* Player)
142{
143 if (!HasAuthority() || !Player)
144 return;
145
146 ALingoGameState* LingoGameState = GetGameState<ALingoGameState>();
147 if (!LingoGameState)
148 return;
149
150 // QuestResult 업데이트
151 LingoGameState->QuestResult.bSuccess = true;
152 LingoGameState->QuestResult.RemainTime = LingoGameState->GetRemainMissionTime();
153 LingoGameState->QuestResult.AttemptCount = Player->AttemptCount;
154 LingoGameState->QuestResult.SelectedSymbol = Player->SelectedWord1;
155 LingoGameState->QuestResult.SelectedColor = Player->SelectedWord2;
156
157 // 성공 플래그 설정
158 LingoGameState->bQuestSuccess = true;
159
160 // 타이머 중지
161 LingoGameState->StopMissionTimer();
162
163 PRINTLOG(TEXT("[GameMode] HandleCorrectAnswer - Quest completed successfully"));
164
165 // BroadcastManager를 통해 성공 이벤트 브로드캐스트
166 if (UBroadcastManager* BroadcastManager = UBroadcastManager::Get(GetWorld()))
167 {
168 // 추후 BroadcastManager에 퀘스트 성공 이벤트 추가 필요
169 }
170}

172void ALingoGameMode::HandleWrongAnswer(ALingoPlayerState* Player, bool bSymbolCorrect, bool bColorCorrect)
173{
174 if (!HasAuthority() || !Player)
175 return;
176
177 ALingoGameState* LingoGameState = GetGameState<ALingoGameState>();
178 if (!LingoGameState)
179 return;
180
181 // 타이머 패널티 (30초 감소)
182 const float Penalty = 30.f;
183 LingoGameState->RemainMissionTime = FMath::Max(0.f, LingoGameState->RemainMissionTime - Penalty);
184
185 PRINTLOG(TEXT("[GameMode] HandleWrongAnswer - Penalty applied: %.0f seconds, Remaining: %.0f seconds"),
186 Penalty, LingoGameState->RemainMissionTime);
187
188 // 틀린 항목 판정
189 if (!bSymbolCorrect)
190 {
191 Player->SelectedWord1 = TEXT("");
192 Player->bWrongWord1 = true;
193 PRINTLOG(TEXT("[GameMode] Symbol was wrong - Cleared selection"));
194 }
195
196 if (!bColorCorrect)
197 {
198 Player->SelectedWord2 = TEXT("");
199 Player->bWrongWord2 = true;
200 PRINTLOG(TEXT("[GameMode] Color was wrong - Cleared selection"));
201 }
202
203 // 시도 횟수 증가
204 Player->AttemptCount++;
205
206 PRINTLOG(TEXT("[GameMode] HandleWrongAnswer - Attempt count: %d"), Player->AttemptCount);
207}
