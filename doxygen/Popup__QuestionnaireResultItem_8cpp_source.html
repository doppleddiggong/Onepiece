52UTexture2D* UPopup_QuestionnaireResultItem::LoadTextureFromFile(const FString& filePath)
53{
54 TArray<uint8> RawFileData;
55 if (!FFileHelper::LoadFileToArray(RawFileData, *filePath))
56 {
57 PRINTLOG(TEXT("Failed to load file: %s"), *filePath);
58 return nullptr;
59 }
60
61 IImageWrapperModule& ImageWrapperModule = FModuleManager::LoadModuleChecked<IImageWrapperModule>(FName("ImageWrapper"));
62 TSharedPtr<IImageWrapper> ImageWrapper = ImageWrapperModule.CreateImageWrapper(EImageFormat::PNG);
63
64 if (!ImageWrapper.IsValid() || !ImageWrapper->SetCompressed(RawFileData.GetData(), RawFileData.Num()))
65 {
66 PRINTLOG(TEXT("Failed to parse PNG: %s"), *filePath);
67 return nullptr;
68 }
69
70 TArray<uint8> UncompressedBGRA;
71 if (!ImageWrapper->GetRaw(ERGBFormat::BGRA, 8, UncompressedBGRA))
72 {
73 PRINTLOG(TEXT("Failed to decompress PNG: %s"), *filePath);
74 return nullptr;
75 }
76
77 UTexture2D* Texture = UTexture2D::CreateTransient(ImageWrapper->GetWidth(), ImageWrapper->GetHeight(), PF_B8G8R8A8);
78 if (!Texture)
79 {
80 return nullptr;
81 }
82
83 void* TextureData = Texture->GetPlatformData()->Mips[0].BulkData.Lock(LOCK_READ_WRITE);
84 FMemory::Memcpy(TextureData, UncompressedBGRA.GetData(), UncompressedBGRA.Num());
85 Texture->GetPlatformData()->Mips[0].BulkData.Unlock();
86 Texture->UpdateResource();
87
88 return Texture;
89}
