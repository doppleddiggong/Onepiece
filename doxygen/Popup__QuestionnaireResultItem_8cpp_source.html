13void UPopup_QuestionnaireResultItem::InitItem(int32 index, FString questionKor, const FResponseWriteData& data)
14{
15 // 질문 인덱스 설정 (예: "Question.01")
16 if (Text_Index)

21
22 if (Text_Question)
23 {
24 PRINTLOG(TEXT("Question: %s"), *questionKor);
25 Text_Question->SetText(FText::FromString(questionKor));

28 // 이미지 불러와서 적용
29 FString FileName = FString::Printf(TEXT("Answer%d.PNG"), index);
30 UTexture2D* LoadedTexture = LoadTextureFromFile(WriteImagePath / FileName);
31 if (LoadedTexture)
32 {
33 // SizeBox 비율 이미지에 맞게 수정
34 float aspectRatio = LoadedTexture->GetSizeX() / LoadedTexture->GetSizeY();
35 SizeBox_Answer->SetMinAspectRatio(aspectRatio);
36 SizeBox_Answer->SetMaxAspectRatio(aspectRatio);
37 // Image X 사이즈가 1080보다 크면 1080에, 아니면 Image X 사이즈에 1/ratio를 곱한다.
38 SizeBox_Answer->SetHeightOverride(Image_Answer->GetBrush().GetImageSize().X > SizeBox_Answer->GetMaxDesiredWidth() ? SizeBox_Answer->GetMaxDesiredWidth() / aspectRatio : Image_Answer->GetBrush().GetImageSize().X / aspectRatio);
39 Image_Answer->SetBrushFromTexture(LoadedTexture);
40 }
41
42 // 피드백 내용 설정
43 if (Text_Message)
44 {
45 FString text = data.display.message;
46 Text_Message->SetText(FText::FromString(text));
47 }
48 if (Text_Correction)
49 {
50 FString text = data.display.correction;
51 Text_Correction->SetText(FText::FromString(text));
52 }
53}

55UTexture2D* UPopup_QuestionnaireResultItem::LoadTextureFromFile(const FString& filePath)
56{
57 TArray<uint8> RawFileData;
58 if (!FFileHelper::LoadFileToArray(RawFileData, *filePath))
59 {
60 PRINTLOG(TEXT("Failed to load file: %s"), *filePath);
61 return nullptr;
62 }
63
64 IImageWrapperModule& ImageWrapperModule = FModuleManager::LoadModuleChecked<IImageWrapperModule>(FName("ImageWrapper"));
65 TSharedPtr<IImageWrapper> ImageWrapper = ImageWrapperModule.CreateImageWrapper(EImageFormat::PNG);
66
67 if (!ImageWrapper.IsValid() || !ImageWrapper->SetCompressed(RawFileData.GetData(), RawFileData.Num()))
68 {
69 PRINTLOG(TEXT("Failed to parse PNG: %s"), *filePath);
70 return nullptr;
71 }
72
73 TArray<uint8> UncompressedBGRA;
74 if (!ImageWrapper->GetRaw(ERGBFormat::BGRA, 8, UncompressedBGRA))
75 {
76 PRINTLOG(TEXT("Failed to decompress PNG: %s"), *filePath);
77 return nullptr;
78 }
79
80 UTexture2D* Texture = UTexture2D::CreateTransient(ImageWrapper->GetWidth(), ImageWrapper->GetHeight(), PF_B8G8R8A8);
81 if (!Texture)
82 {
83 return nullptr;
84 }
85
86 void* TextureData = Texture->GetPlatformData()->Mips[0].BulkData.Lock(LOCK_READ_WRITE);
87 FMemory::Memcpy(TextureData, UncompressedBGRA.GetData(), UncompressedBGRA.Num());
88 Texture->GetPlatformData()->Mips[0].BulkData.Unlock();
89 Texture->UpdateResource();
90
91 return Texture;
92}
