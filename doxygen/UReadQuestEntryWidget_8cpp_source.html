73void UReadQuestEntryWidget::OnButtonClicked()
74{
75 // 오답 상태이거나 이미 선택된 상태면 무시
76 if (bIsWrong)
77 {
78 PRINTLOG(TEXT("[ReadQuestEntry] Button clicked but entry is marked as wrong: %s"), *ChoiceValue);
79 return;
80 }
81
82 // 델리게이트 브로드캐스트
83 OnEntrySelected.Broadcast(ChoiceValue, this);
84
85 PRINTLOG(TEXT("[ReadQuestEntry] Button clicked: %s"), *ChoiceValue);
86}

88void UReadQuestEntryWidget::UpdateVisualState()
89{
90 if (!StateImage)
91 return;
92
93 // 상태에 따라 이미지 표시
94 if (bIsWrong)
95 {
96 // 오답: 빨간색 X 표시 등
97 StateImage->SetVisibility(ESlateVisibility::Visible);
98 StateImage->SetColorAndOpacity(FLinearColor::Red);
99 }
100 else if (bIsSelected)
101 {
102 // 선택됨: 초록색 체크 표시 등
103 StateImage->SetVisibility(ESlateVisibility::Visible);
104 StateImage->SetColorAndOpacity(FLinearColor::Green);
105 }
106 else
107 {
108 // 기본 상태: 이미지 숨김
109 StateImage->SetVisibility(ESlateVisibility::Hidden);
110 }
111
112 // 버튼 스타일 변경 (선택적)
113 if (SelectButton)
114 {
115 // 오답이면 버튼 비활성화
116 if (bIsWrong)
117 {
118 SelectButton->SetIsEnabled(false);
119 }
120 }
