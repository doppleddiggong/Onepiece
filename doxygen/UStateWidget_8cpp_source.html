65void UStateWidget::UpdateSpectrumVisual(float DeltaTime)
66{
67 if (!SpectrumProgressBar || !SpectrumProgressBar->IsVisible())
68 return;
69
70 const float Delta = FMath::Max(DeltaTime, 0.0f);
71 const float TargetValue = SpectrumDisplayValue;
72 const float CurrentValue = SpectrumProgressBar->GetPercent();
73 const float InterpSpeed = TargetValue > CurrentValue ? SpectrumRiseSpeed : SpectrumDecaySpeed;
74 const float NewPercent = FMath::FInterpConstantTo(CurrentValue, TargetValue, Delta, InterpSpeed);
75 SpectrumProgressBar->SetPercent(FMath::Clamp(NewPercent, 0.0f, 1.0f));
76}
