39void UStateWidget::UpdateSpectrumVisual(float DeltaTime)
40{
41 if (!SpectrumProgressBar || !SpectrumProgressBar->IsVisible())
42 return;
43
44 const float Delta = FMath::Max(DeltaTime, 0.0f);
45 const float TargetValue = SpectrumDisplayValue;
46 const float CurrentValue = SpectrumProgressBar->GetPercent();
47 const float InterpSpeed = TargetValue > CurrentValue ? SpectrumRiseSpeed : SpectrumDecaySpeed;
48 const float NewPercent = FMath::FInterpConstantTo(CurrentValue, TargetValue, Delta, InterpSpeed);
49 SpectrumProgressBar->SetPercent(FMath::Clamp(NewPercent, 0.0f, 1.0f));
50}
