111void UMainWidget::OnUpdateMissionTimerState(bool bIsActive, float TimeLimit)
112{
113 if (!CachedGameState)
114 {
115 if (UWorld* World = GetWorld())
116 CachedGameState = World->GetGameState<ALingoGameState>();
117 }
118
119 // GameState의 RemainMissionTime 업데이트
120 if (CachedGameState && bIsActive && TimeLimit > 0.0f)
121 CachedGameState->RemainMissionTime = TimeLimit;
122
123 if (!QuestInfoWidget)
124 return;
125
126 QuestInfoWidget->SetVisibility(bIsActive ? ESlateVisibility::Visible : ESlateVisibility::Collapsed );
127 HookTargetIndicator->SetVisibility(ESlateVisibility::Visible);
128
129 if (bIsActive)
130 {
131 QuestInfoWidget->InitQuestInfo(ULingoGameHelper::GetLingoPlayerState(GetWorld())->QuestRole);
132 }
133
134 SetMissionTimerState(bIsActive);
135}

146void UMainWidget::UpdateSpeakWidget()
147{
148 USpeakStageSubsystem* Subsystem = GetWorld()->GetSubsystem<USpeakStageSubsystem>();
149 ASpeakStageActor* SpeakStage = Subsystem ? Subsystem->GetSpeakStage() : nullptr;
150 APlayerState* CurrentSpeaker = SpeakStage ? SpeakStage->GetCurrentSpeaker() : nullptr;
151
152 const bool bShouldShow = Subsystem && Subsystem->IsInitialized() && SpeakStage && CurrentSpeaker;
153
154 SpeakWidget->SetWidgetVisibility(bShouldShow);
155
156 if (bShouldShow)
157 {
158 APlayerController* LocalPC = GetWorld()->GetFirstPlayerController();
159 if (!LocalPC)
160 return;
161
162 APlayerState* LocalPlayerState = LocalPC->GetPlayerState<APlayerState>();
163 SpeakWidget->UpdateSpeakStageUI(SpeakStage, LocalPlayerState);
164 }
165}
