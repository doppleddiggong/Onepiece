107void UMainWidget::OnUpdateMissionTimerState(bool bIsActive, float TimeLimit)
108{
109 if (!CachedGameState)
110 {
111 if (UWorld* World = GetWorld())
112 CachedGameState = World->GetGameState<ALingoGameState>();
113 }
114
115 // GameState의 RemainMissionTime 업데이트
116 if (CachedGameState && bIsActive && TimeLimit > 0.0f)
117 CachedGameState->UpdateRemainMissionTime( TimeLimit );
118
119 if (!QuestInfoWidget)
120 return;
121
122 QuestInfoWidget->SetVisibility(bIsActive ? ESlateVisibility::Visible : ESlateVisibility::Collapsed );
123 HookTargetIndicator->SetVisibility(ESlateVisibility::Visible);
124
125 if (bIsActive)
126 {
127 WidgetSwitcher->SetActiveWidgetIndex(0);
128 QuestInfoWidget->InitQuestInfo();
129 }
130
131 SetMissionTimerState(bIsActive);
132}

158bool UMainWidget::CanShowSpeakWidget( ASpeakStageActor*& OutSpeakStage, ALingoPlayerState*& OutLocalPlayerState) const
159{
160 OutSpeakStage = ULingoGameHelper::GetSpeakStageActor(this);
161 if (!OutSpeakStage)
162 return false;
163
164 ALingoPlayerState* CurrentSpeaker = OutSpeakStage->GetCurrentSpeaker();
165 if (!CurrentSpeaker)
166 return false;
167
168 APlayerController* LocalPC = GetWorld()->GetFirstPlayerController();
169 if (!LocalPC)
170 return false;
171
172 OutLocalPlayerState = LocalPC->GetPlayerState<ALingoPlayerState>();
173 if (!OutLocalPlayerState)
174 return false;
175
176 return (OutLocalPlayerState == CurrentSpeaker);
177}
