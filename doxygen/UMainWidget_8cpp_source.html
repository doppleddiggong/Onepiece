81void UMainWidget::OnUpdateMissionTimerState(bool bIsActive, float TimeLimit)
82{
83 if (!CachedGameState)
84 {
85 if (UWorld* World = GetWorld())
86 CachedGameState = World->GetGameState<ALingoGameState>();
87 }
88
89 // GameState의 RemainMissionTime 업데이트
90 if (CachedGameState && bIsActive && TimeLimit > 0.0f)
91 CachedGameState->RemainMissionTime = TimeLimit;
92
93 if (!QuestInfoWidget)
94 return;
95
96 QuestInfoWidget->SetVisibility(bIsActive ? ESlateVisibility::Visible : ESlateVisibility::Collapsed );
97 HookTargetIndicator->SetVisibility(ESlateVisibility::Visible);
98
99 if (bIsActive)
100 {
101 StartMissionTimer();
102 QuestInfoWidget->InitQuestInfo(ULingoGameHelper::GetLingoPlayerState(GetWorld())->QuestRole);
103 }
104 else
105 {
106 StopMissionTimer();
107 }
108}
