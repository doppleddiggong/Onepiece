44void UMainWidget::NativeConstruct()

68 {
69 TutorMessage->SetVisibility(ESlateVisibility::HitTestInvisible);
70
71 if (TutorHideAnim)
72 {
73 FWidgetAnimationDynamicEvent HideDelegate;
74 HideDelegate.BindDynamic(this, &UMainWidget::OnTutorHideComplete);
75 BindToAnimationFinished(TutorHideAnim, HideDelegate);
76 }
77
78 bIsTutorVisible = false;
79 }
80
81 StopMissionTimer();

110void UMainWidget::OnUpdateMissionTimerState(bool bIsActive, float TimeLimit)
111{
112 if (!CachedGameState)
113 {
114 if (UWorld* World = GetWorld())
115 CachedGameState = World->GetGameState<ALingoGameState>();
116 }
117
118 // GameState의 RemainMissionTime 업데이트
119 if (CachedGameState && bIsActive && TimeLimit > 0.0f)
120 CachedGameState->RemainMissionTime = TimeLimit;
121
122 if (!QuestInfoWidget)
123 return;
124
125 QuestInfoWidget->SetVisibility(bIsActive ? ESlateVisibility::Visible : ESlateVisibility::Collapsed );
126 HookTargetIndicator->SetVisibility(ESlateVisibility::Visible);
127
128 if (bIsActive)
129 {
130 StartMissionTimer();
131 QuestInfoWidget->InitQuestInfo(ULingoGameHelper::GetLingoPlayerState(GetWorld())->QuestRole);
132 }
133 else
134 {
135 StopMissionTimer();
136 }

160void UMainWidget::UpdateSpeakWidget()
161{
162 // SpeakWidget이 없으면 리턴 (BindWidgetOptional)
163 if (!SpeakWidget)
164 return;
165
166 // World 가져오기
167 UWorld* World = GetWorld();
168 if (!World)
169 return;
170
171 // SpeakStageSubsystem 가져오기
172 USpeakStageSubsystem* Subsystem = World->GetSubsystem<USpeakStageSubsystem>();
173 if (!Subsystem || !Subsystem->IsInitialized())
174 {
175 // Subsystem이 없으면 SpeakWidget 숨김
176 SpeakWidget->SetWidgetVisibility(false);
177 return;
178 }
179
180 // SpeakStage 가져오기
181 ASpeakStageActor* SpeakStage = Subsystem->GetSpeakStage();
182 if (!SpeakStage)
183 {
184 // SpeakStage가 없으면 SpeakWidget 숨김
185 SpeakWidget->SetWidgetVisibility(false);
186 return;
187 }
188
189 // 현재 발화자 확인
190 APlayerState* CurrentSpeaker = SpeakStage->GetCurrentSpeaker();
191 if (!CurrentSpeaker)
192 {
193 // 발화자가 없으면 (Stage 완료) SpeakWidget 숨김
194 SpeakWidget->SetWidgetVisibility(false);
195 return;
196 }
197
198 // SpeakStage가 활성화되어 있으면 SpeakWidget 표시
199 SpeakWidget->SetWidgetVisibility(true);
200
201 // 로컬 플레이어의 PlayerState 가져오기
202 APlayerController* LocalPC = World->GetFirstPlayerController();
203 if (!LocalPC)
204 return;
205
206 APlayerState* LocalPlayerState = LocalPC->GetPlayerState<APlayerState>();
207
208 // SpeakWidget UI 업데이트
209 SpeakWidget->UpdateSpeakStageUI(SpeakStage, LocalPlayerState);
210}

212void UMainWidget::SetTutorMessage(const FText& NewMessage)
213{
214 if (!TutorMessage)
215 return;
216
217 // 기존 타이머가 있으면 클리어
218 if (UWorld* World = GetWorld())
219 {
220 if (AutoHideTimerHandle.IsValid())
221 {
222 World->GetTimerManager().ClearTimer(AutoHideTimerHandle);
223 }
224 }
225
226 if (bIsTutorVisible)
227 {
228 // 표시 중이면 Hide → SetText → Show
229 PendingMessage = NewMessage;
230 bHasPendingMessage = true;
231
232 if (TutorHideAnim)
233 PlayAnimation(TutorHideAnim);
234 }
235 else
236 {
237 // 숨김 상태면 바로 SetText → Show
238 TutorMessage->SetMessageText(NewMessage);
239
240 if (TutorShowAnim)
241 {
242 PlayAnimation(TutorShowAnim);
243 bIsTutorVisible = true;
244
245 // 자동 Hide 타이머 시작
246 StartAutoHideTimer();
247 }
248 }
249}

271void UMainWidget::StartAutoHideTimer()
272{
273 UWorld* World = GetWorld();
274 if (!World)
275 return;
276
277 // 기존 타이머 클리어 (안전성을 위해)
278 if (AutoHideTimerHandle.IsValid())
279 {
280 World->GetTimerManager().ClearTimer(AutoHideTimerHandle);
281 }
282
283 // 지정된 시간 후 자동으로 Hide 애니메이션 재생
284 World->GetTimerManager().SetTimer(
285 AutoHideTimerHandle,
286 [this]()
287 {
288 if (bIsTutorVisible && TutorHideAnim)
289 {
290 PlayAnimation(TutorHideAnim);
291 }
292 },
293 TutorMessageDisplayDuration,
294 false
295 );
296}

298UAutoDespawnItem* UMainWidget::AddItemToBox()
299{
300 if (!ItemHorizontalBox)
301 {
302 PRINTLOG( TEXT("UMainWidget::AddItemToBox - ItemHorizontalBox is null!"));
303 return nullptr;
304 }
305
306 if (!ItemWidgetClass)
307 {
308 PRINTLOG( TEXT("UMainWidget::AddItemToBox - ItemWidgetClass is not set!"));
309 return nullptr;
310 }
311
312 UWorld* World = GetWorld();
313 if (!World)
314 {
315 PRINTLOG( TEXT("UMainWidget::AddItemToBox - World is null!"));
316 return nullptr;
317 }
318
319 // 새 아이템 생성
320 UAutoDespawnItem* NewItem = CreateWidget<UAutoDespawnItem>(World, ItemWidgetClass);
321 if (!NewItem)
322 {
323 PRINTLOG( TEXT("UMainWidget::AddItemToBox - Failed to create widget!"));
324 return nullptr;
325 }
326
327 // HorizontalBox에 추가 (왼쪽에 추가되지만 Right Alignment로 오른쪽 정렬됨)
328 ItemHorizontalBox->AddChild(NewItem);
329
330 return NewItem;
