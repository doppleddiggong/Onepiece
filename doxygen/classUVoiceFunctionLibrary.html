257{
258 if (InSampleRate == OutSampleRate)
259 return InPCMData;
260
261 // PCM 데이터는 int16 형식
262 const int32 NumSamples = InPCMData.Num() / sizeof(int16);
263 const int16* InSamples = reinterpret_cast<const int16*>(InPCMData.GetData());
264
265 // 리샘플링 비율 계산
266 const double ResampleRatio = static_cast<double>(OutSampleRate) / static_cast<double>(InSampleRate);
267 const int32 OutNumSamples = FMath::CeilToInt(NumSamples * ResampleRatio);
268
269 TArray<uint8> OutPCMData;
270 OutPCMData.Reserve(OutNumSamples * sizeof(int16));
271
272 // Linear interpolation 리샘플링
273 for (int32 i = 0; i < OutNumSamples; ++i)
274 {
275 const double SourceIndex = i / ResampleRatio;
276 const int32 Index0 = FMath::FloorToInt(SourceIndex);
277 const int32 Index1 = FMath::Min(Index0 + 1, NumSamples - 1);
278 const double Fraction = SourceIndex - Index0;
279
280 // Linear interpolation
281 const int16 Sample0 = InSamples[Index0];
282 const int16 Sample1 = InSamples[Index1];
283 const int16 InterpolatedSample = FMath::RoundToInt(Sample0 + (Sample1 - Sample0) * Fraction);
284
285 // 결과 버퍼에 추가
286 const uint8* SampleBytes = reinterpret_cast<const uint8*>(&InterpolatedSample);
287 OutPCMData.Append(SampleBytes, sizeof(int16));
288 }
289
290 return OutPCMData;
291}
