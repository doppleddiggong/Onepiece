17void APlatformSwitch::BeginPlay()
18{
19 Super::BeginPlay();
20
21 this->TriggerDelay = Duration;
22 this->DetectTarget = false;
23 this->ElapsedTime = 0;
24 this->bActivateState = false;
25
26 SwitchButton = FComponentHelper::FindComponentByNameRecursive<UStaticMeshComponent>(this, SWITCH_BUTTON_PATH);
27 SwitchCollision = FComponentHelper::FindComponentByNameRecursive<UPrimitiveComponent>(this, SWITCH_COLLISION_PATH);
28
29 if( SwitchButton != nullptr)
30 {
31 MaterialButton = SwitchButton->CreateDynamicMaterialInstance(0);
32 MaterialButton->SetVectorParameterValue( ColorParam, IdleColor );
33 OriginVector = SwitchButton->GetRelativeLocation();
34
35 SwitchCollision->OnComponentBeginOverlap.AddDynamic(this, &APlatformSwitch::OnBeginOverlap);
36 SwitchCollision->OnComponentEndOverlap.AddDynamic(this, &APlatformSwitch::OnEndOverlap);
37 }
38}

40void APlatformSwitch::Tick(float DeltaTime)
41{
42 Super::Tick(DeltaTime);
43
44 if ( DetectTarget == false)
45 return;
46
47 AddElapsedTime();
48
49 const auto AlphaValue = LerpAlpha();
50 // ULOG(Warning, "LerpAlpha : %f", LerpAlpha() );
51
52 {
53 const auto Color_A = FMaterialHelper::GetVectorParameterValueSafe(MaterialButton, ColorParam);
54 const auto Color_B = WarningColor;
55 const auto Color_Result = UKismetMathLibrary::LinearColorLerpUsingHSV(Color_A, Color_B, AlphaValue);
56
57 MaterialButton->SetVectorParameterValue( ColorParam, Color_Result );
58 }
59
60 {
61 const auto Vector_A = SwitchButton->GetRelativeLocation();
62 const auto Vector_B = EndVector;
63 const auto Vector_Result = UKismetMathLibrary::VLerp( Vector_A, Vector_B, AlphaValue);
64
65 SwitchButton->SetRelativeLocation(Vector_Result);
66 }
67
68 if( ActivateTrigger() )
69 {
70 // 물건이 TriggerDelay 타임 이상 올라가면 발동
71 ChangeActivateState(true);
72 }
73}

75void APlatformSwitch::AddElapsedTime()

80float APlatformSwitch::LerpAlpha() const
81{
82 return UKismetMathLibrary::FClamp( ElapsedTime / Duration, 0.0f, 1.0f );

85bool APlatformSwitch::ActivateTrigger()
86{
87 TriggerDelay -= GetWorld()->GetDeltaSeconds();
88
89 if ( TriggerDelay < 0 )
90 {
91 TriggerDelay = Duration;
92 return true;
93 }
94
95 return false;
