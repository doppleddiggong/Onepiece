53void UPopup_DailyResult::InitQuestionList()
54{
55 if (!VerticalBox)
56 return;
57
58 // 기존 아이템 제거
59 VerticalBox->ClearChildren();
60
61 // QuestionList와 AnswerList를 페어로 사용하여 아이템 생성
62 for (int32 i = 0; i < StudyResult.QuestionList.Num(); ++i)
63 {
64 // AnswerList 범위 체크
65 if (!StudyResult.AnswerList.IsValidIndex(i))
66 continue;
67
68 // 아이템 위젯 생성 및 초기화
69 if (auto ItemWidget = CreateWidget<UPopup_DailyResultItem>(GetWorld(), AnswerItemClass))
70 {
71 // 간단하게 3개 파라미터만 전달 (FWordData에서 한글 단어 추출)
72 ItemWidget->InitData(
73 i+1,
74 StudyResult.QuestionList[i].Kor, // Question (한글)
75 StudyResult.AnswerList[i].final_feedback, // FeedBack
76 StudyResult.AnswerList[i].final_overall_score // Score
77 );
78
79 VerticalBox->AddChildToVerticalBox(ItemWidget);
80
81 // 마지막 아이템이 아니면 Spacer 추가 (아이템 간 간격)
82 if (i < StudyResult.QuestionList.Num() - 1)
83 {
84 USpacer* Spacer = NewObject<USpacer>(this);
85 if (Spacer)
86 {
87 Spacer->SetSize(FVector2D(1.0f, 15.0f));
88 VerticalBox->AddChildToVerticalBox(Spacer);
89 }
90 }
91 }
92 }
93}
