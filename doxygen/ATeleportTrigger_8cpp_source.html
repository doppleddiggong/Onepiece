14ATeleportTrigger::ATeleportTrigger()
15{
16 PrimaryActorTick.bCanEverTick = true;
17 bReplicates = true;
18
19 // 루트 컴포넌트로 BoxComponent 생성
20 TriggerBox = CreateDefaultSubobject<UBoxComponent>(TEXT("TriggerBox"));
21 RootComponent = TriggerBox;
22
23 // 박스 크기 기본값 설정
24 TriggerBox->SetBoxExtent(FVector(100.0f, 100.0f, 100.0f));
25
26 // Overlap 이벤트 활성화
27 TriggerBox->SetGenerateOverlapEvents(true);
28 TriggerBox->SetCollisionEnabled(ECollisionEnabled::QueryOnly);
29 TriggerBox->SetCollisionResponseToAllChannels(ECR_Ignore);
30 TriggerBox->SetCollisionResponseToChannel(ECC_Pawn, ECR_Overlap);
31
32 // 초기값 설정
33 bIsOneShot = true;
34 bIsTriggered = false;
35 bShowDebugBox = true;
36 DebugBoxColor = FColor::Green;
37}

47void ATeleportTrigger::Tick(float DeltaTime)
48{
49 Super::Tick(DeltaTime);
50
51 if (bShowDebugBox && TriggerBox)
52 {
53 FVector BoxCenter = TriggerBox->GetComponentLocation();
54 FVector BoxExtent = TriggerBox->GetScaledBoxExtent();
55 FRotator BoxRotation = TriggerBox->GetComponentRotation();
56
57 // 디버그 박스 표시 (트리거 활성화 상태일 때만)
58 if (!bIsTriggered)
59 {
60 DrawDebugBox(
61 GetWorld(),
62 BoxCenter,
63 BoxExtent,
64 BoxRotation.Quaternion(),
65 DebugBoxColor,
66 false,
67 -1.0f,
68 0,
69 2.0f
70 );
71 }
72
73 // 트리거 정보를 텍스트로 표시
74 FString StatusText = bIsTriggered ? TEXT("[TRIGGERED]") : TEXT("[ACTIVE]");
75 FColor TextColor = bIsTriggered ? FColor::Red : FColor::Green;
76
77 FVector TextLocation = BoxCenter + FVector(0.0f, 0.0f, BoxExtent.Z + 50.0f);
78

80 DrawDebugString(
81 GetWorld(),
82 TextLocation,
83 StatusText,
84 nullptr,
85 TextColor,
86 0.0f,

89 );
90 }

93void ATeleportTrigger::OnTriggerBeginOverlap(
94 UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
95 UPrimitiveComponent* OtherComp, int32 OtherBodyIndex,
96 bool bFromSweep, const FHitResult& SweepResult)
97{
98 // 서버에서만 처리
99 if (!HasAuthority())
100 return;
101
102 // 원샷 모드이고 이미 트리거되었으면 무시
103 if (bIsOneShot && bIsTriggered)
104 return;
105
106 // PlayerActor인지 확인
107 APlayerActor* PlayerActor = Cast<APlayerActor>(OtherActor);
108 if (PlayerActor)
109 {
110 // TeleportOut이 설정되어 있는지 확인
111 if (!TeleportOut)
112 {
113 PRINTLOG(TEXT("ATeleportTrigger: TeleportOut이 설정되지 않았습니다!"));
114 return;
115 }
116
117 // TeleportOut의 위치 가져오기
118 FVector TargetLocation = TeleportOut->GetActorLocation();
119
120 // NetworkBroadcastActor를 통해 모든 플레이어 텔레포트
121 if (ANetworkBroadcastActor* BroadcastActor = ANetworkBroadcastActor::Get(this))
122 {
123 BroadcastActor->SendTeleportAllPlayers(TargetLocation, this);
124 PRINTLOG( TEXT("텔레포트 요청: %s"), *TargetLocation.ToString());
125 }
126 else
127 {
128 PRINTLOG(TEXT("ATeleportTrigger: NetworkBroadcastActor를 찾을 수 없습니다!"));
129 }
130
131 OnActivate();
132
133 // 원샷 모드일 때만 트리거 상태 변경
134 if (bIsOneShot)
135 bIsTriggered = true;
136 }
137}
