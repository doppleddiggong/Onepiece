36AActor* ADropper::SpawnActor(TSubclassOf<class AActor> SpawnActorClass)
37{
38 if (!SpawnActorClass || !SpawnPos)
39 {
40 PRINTLOG( TEXT("SpawnActorClass or SpawnPos is not set"));
41 return nullptr;
42 }
43
44 FActorSpawnParameters SpawnParams;
45 SpawnParams.Owner = this;
46 SpawnParams.Instigator = GetInstigator();
47
48 auto SpawnedActor = GetWorld()->SpawnActor<AActor>(SpawnActorClass,
49 SpawnPos->GetComponentLocation(),
50 SpawnPos->GetComponentRotation(),
51 SpawnParams);
52
53 if (SpawnedActor)
54 {
55 // Start timer for delayed animation and collision disable
56 GetWorldTimerManager().SetTimer(
57 DelayTimerHandle,
58 this,
59 &ADropper::PlayAnimationAndDisableCollision,
60 Delay,
61 false
62 );
63 }
64
65 return SpawnedActor;
66}

68void ADropper::PlayAnimationAndDisableCollision()
69{
70 if (SkeletalMesh && AnimToPlay)
71 SkeletalMesh->PlayAnimation(AnimToPlay, false);
72
73 if (BoxCollision)
74 {
75 BoxCollision->SetCollisionEnabled(ECollisionEnabled::NoCollision);
76
77 // NoCollision 후 다시 BlockAll로 복구하는 타이머 설정
78 GetWorldTimerManager().SetTimer(
79 RestoreTimerHandle,
80 this,
81 &ADropper::RestoreCollision,
82 Delay,
83 false
84 );
85 }
86}

88void ADropper::RestoreCollision()

90 if (BoxCollision)
91 {
92 BoxCollision->SetCollisionEnabled(ECollisionEnabled::QueryAndPhysics);
93 BoxCollision->SetCollisionResponseToAllChannels(ECR_Block);
94 }
