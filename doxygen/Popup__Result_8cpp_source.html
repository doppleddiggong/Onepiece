66void UPopup_Result::SetWrongList()
67{
68 if (!Scrl_WrongList)
69 return;
70
71 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
72 if (!GS) return;
73 // 틀린 인덱스 리스트
74 TArray<int32> WrongList = GS->WrongReadAnswerList;
75 // 전체 캐리어 정보
76 const TArray<FScenarioTargetData>& ScenarioData = GS->GetReadScenarioData().target_data;
77
78 // 기존 항목 제거
79 Scrl_WrongList->ClearChildren();
80
81 // WrongList의 각 항목을 텍스트로 추가
82 for (int32 i=0; i<WrongList.Num(); i++)
83 {
84 int32 WrongIndex = WrongList[i];
85 auto SD = ScenarioData[WrongIndex];
86
87 if (UTextBlock* TextBlock = NewObject<UTextBlock>(this))
88 {
89 // 맨 마지막 인덱스는 정답
90 if (i == WrongList.Num()-1)
91 {
92 TextBlock->SetText(FText::FromString(FString::Printf(TEXT("[정답] %s, %s"),
93 *SD.word1.name, *SD.word2.name)));
94 TextBlock->SetColorAndOpacity(FSlateColor(FLinearColor::Red));
95
96 Scrl_WrongList->AddChild(TextBlock);
97
98 break;
99 }
100
101 TextBlock->SetText(FText::FromString(FString::Printf(TEXT("Try %d - [오답] %s, %s"),
102 i+1, *SD.word1.name, *SD.word2.name)));
103 TextBlock->SetColorAndOpacity(FSlateColor(FLinearColor::Black));
104
105 Scrl_WrongList->AddChild(TextBlock);
106 }
107 }
108}

110void UPopup_Result::SetTimeTaken()
111{
112 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
113 if (GS)
114 {
115 auto TimeTaken = GS->GetTimeTaken();
116 const int32 Minutes = FMath::FloorToInt(TimeTaken / 60.f);
117 const int32 Seconds = FMath::FloorToInt(TimeTaken) % 60;
118 Txt_TimeTaken->SetText(FText::FromString(FString::Printf(TEXT("%02d:%02d"), Minutes, Seconds)));
119
120 const auto TimeRank = ULingoGameHelper::GetTimeRank(TimeTaken);
121 Txt_TimeRank->SetText(FText::FromString(TimeRank));
122
123 const auto Accuracy = ULingoGameHelper::GetAccuracyPercentage(GS->GetWrongReadAnswerNum());
124 Txt_Accuracy->SetText(FText::FromString(Accuracy));
125 }
126}

128void UPopup_Result::RequestReadResult()
129{
130 if (auto GS = Cast<ALingoGameState>(GetWorld()->GetGameState()))
131 {
132 // 네트워크 송신
133 if (UKLingoNetworkSystem* Network = UKLingoNetworkSystem::Get(GetWorld()))
134 {
135 FRequestReadResult ReadRequest;
136 ReadRequest.room_id = GS->GetRoomId();
137 ReadRequest.user_id = Cast<APlayerControl>(GetOwningPlayer())->GetUserId();
138 ReadRequest.scenario_id = 1;
139 ReadRequest.stage_type = ULingoGameHelper::GetStageTypeIndex(EQuestType::Read);
140 ReadRequest.state_type = 0;
141 ReadRequest.result_time = GS->GetTimeTaken();
142 ReadRequest.wrong_idx = GS->WrongReadAnswerList;
143
144 Network->RequestReadResult(ReadRequest,
145 FResponseReadResultDelegate::CreateUObject(this, &UPopup_Result::OnResponseReadResult));
146 }
147 }
