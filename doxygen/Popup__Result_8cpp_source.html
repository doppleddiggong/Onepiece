29void UPopup_Result::InitPopup(const EQuestType InQuestType)
30{
31 if (Btn_Confirm)
32 {
33 Btn_Confirm->OnButtonClickedEvent.RemoveDynamic(this, &UPopup_Result::OnClickClose);
34 Btn_Confirm->OnButtonClickedEvent.AddDynamic(this, &UPopup_Result::OnClickClose);
35 }
36
37 this->QuestType =InQuestType;
38
39 InitWordWidget();
40 InitWrongList();
41
42 // GameState에서 결과 확인
43 if (auto GS = Cast<ALingoGameState>(GetWorld()->GetGameState()))
44 {
45 if (QuestType == EQuestType::Read)
46 {
47 GS->OnReadResultUpdated.RemoveDynamic(this, &UPopup_Result::InitReadResult);
48 GS->OnReadResultUpdated.AddDynamic(this, &UPopup_Result::InitReadResult);
49 }
50 else if (QuestType == EQuestType::Listen)
51 {
52 GS->OnListenResultUpdated.RemoveDynamic(this, &UPopup_Result::InitListenResult);
53 GS->OnListenResultUpdated.AddDynamic(this, &UPopup_Result::InitListenResult);
54 }
55
56 bool bHasResult = false;
57
58 if (QuestType == EQuestType::Read && !GS->ReadResult.grade.IsEmpty())
59 {
60 // 이미 결과가 있으면 바로 표시
61 PRINTLOG(TEXT("[Popup_Result] Read result already exists in GameState"));
62 InitReadResult(GS->ReadResult);
63 bHasResult = true;
64 }
65 else if (QuestType == EQuestType::Listen && !GS->ListenResult.grade.IsEmpty())
66 {
67 // 이미 결과가 있으면 바로 표시
68 PRINTLOG(TEXT("[Popup_Result] Listen result already exists in GameState"));
69 InitListenResult(GS->ListenResult);
70 bHasResult = true;
71 }
72
73 // 결과가 없으면 Host만 요청
74 if (!bHasResult && GetOwningPlayer()->HasAuthority())
75 {
76 RequestResult();
77 }
78 }
79}

109void UPopup_Result::InitWrongList()
110{
111 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
112 if (!GS)
113 return;
114
115 // 틀린 인덱스 리스트
116 TArray<int32> WrongList;
117 // 전체 캐리어 정보
118 TArray<FScenarioTargetData> ScenarioData;
119 FScenarioTargetData CorrectData;
120
121 if ( QuestType == EQuestType::Read)
122 {
123 WrongList = GS->WrongReadAnswerList;
124 ScenarioData = GS->GetReadScenarioData().target_data;
125 CorrectData = GS->GetReadScenarioData().GetCorrectAnswerData();
126 }
127 else if ( QuestType == EQuestType::Listen )
128 {
129 WrongList = GS->WrongListenAnswerList;
130 ScenarioData = GS->TryListenAnswerData.target_data;
131 CorrectData = GS->GetListenScenarioData().GetCorrectAnswerData();
132 }
133
134 // 기존 항목 제거
135 VerticalBox->ClearChildren();
136 // WrongList의 각 항목을 UAnswerItem으로 추가
137 for (int32 i = 0; i < WrongList.Num(); i++)
138 {
139 const FScenarioTargetData& SD = ScenarioData[WrongList[i]];
140
141 // UAnswerItem 위젯 생성
142 if (AnswerItemClass)
143 {
144 UAnswerItem* AnswerItem = CreateWidget<UAnswerItem>(this, AnswerItemClass);
145 if (AnswerItem)
146 {
147 // AnswerItem 초기화
148 AnswerItem->InitInfo(QuestType, i + 1, SD, CorrectData);
149
150 // HorizontalBox에 추가
151 VerticalBox->AddChild(AnswerItem);
152
153 {
154 USpacer* Spacer = NewObject<USpacer>(VerticalBox);
155 Spacer->SetSize(FVector2D(1.f, 10.f));
156 VerticalBox->AddChild(Spacer);
157 }
158 }
159 }
160 }

245void UPopup_Result::RequestResult()
246{
247 APlayerController* PC = GetOwningPlayer();
248 if (!PC)
249 return;
250
251 // Host 플레이어(첫 번째 플레이어)인지 확인
252 bool bIsHost = false;
253 if (APlayerState* PS = PC->GetPlayerState<APlayerState>())
254 {
255 if (AGameStateBase* GameStateBase = GetWorld()->GetGameState())
256 {
257 int32 PlayerIndex = GameStateBase->PlayerArray.IndexOfByKey(PS);
258 bIsHost = (PlayerIndex == 0);
259 }
260 }
261
262 // Host가 아니면 요청하지 않음
263 if (!bIsHost)
264 {
265 PRINTLOG(TEXT("[Popup_Result] Guest player - skipping request, waiting for Host"));
266 return;
267 }
268
269 PRINTLOG(TEXT("[Popup_Result] Host player - sending result request"));
270
271 if (auto GS = Cast<ALingoGameState>(GetWorld()->GetGameState()))
272 {
273 // 네트워크 송신 (Host만)
274 if (UKLingoNetworkSystem* Network = UKLingoNetworkSystem::Get(GetWorld()))
275 {
276 if ( QuestType == EQuestType::Read )
277 {
278 FRequestReadResult ReadRequest;
279 ReadRequest.room_id = GS->GetRoomId();
280 ReadRequest.user_id = Cast<APlayerControl>(GetOwningPlayer())->GetUserId();
281 ReadRequest.scenario_id = 1;
282 ReadRequest.stage_type = ULingoGameHelper::GetStageTypeIndex(QuestType);
283 ReadRequest.state_type = 0;
284 ReadRequest.result_time = GS->GetTimeTaken();
285 ReadRequest.wrong_idx = GS->WrongReadAnswerList;
286 Network->RequestReadResult(ReadRequest,
287 FResponseReadResultDelegate::CreateUObject(this, &UPopup_Result::OnResponseReadResult));
288 }
289 else if ( QuestType == EQuestType::Listen )
290 {
291 FRequestListenResult ListenRequest;
292 ListenRequest.room_id = GS->GetRoomId();
293 ListenRequest.user_id = Cast<APlayerControl>(GetOwningPlayer())->GetUserId();
294 ListenRequest.scenario_id = 1;
295 ListenRequest.stage_type = ULingoGameHelper::GetStageTypeIndex(QuestType);
296 ListenRequest.state_type = 0;
297 ListenRequest.result_time = GS->GetTimeTaken();
298 ListenRequest.wrong_idx = GS->WrongListenAnswerList;
299 Network->RequestListenResult(ListenRequest,
300 FResponseListenResultDelegate::CreateUObject(this, &UPopup_Result::OnResponseListenResult));
301 }
302 }
303 }
304}

306void UPopup_Result::OnResponseReadResult(FResponseReadResult& ResponseData, bool bWasSuccessful)
307{
308 if (bWasSuccessful)
309 {
310 PRINTLOG(TEXT("[Result] Grade: %s, average_score: %d, Top Percent: %.2f%%"),
311 *ResponseData.grade, ResponseData.average_score, ResponseData.top_percent);
312
313 // GameState에 결과 저장 (자동으로 복제됨)
314 if (auto GS = Cast<ALingoGameState>(GetWorld()->GetGameState()))
315 {
316 GS->ReadResult = ResponseData;
317 GS->OnReadResultUpdated.Broadcast(ResponseData);
318 }
319
320 this->InitReadResult(ResponseData);
321 }
322 else
323 {
324 PRINTLOG(TEXT("[Result] Quest result Failed"));
325 }
