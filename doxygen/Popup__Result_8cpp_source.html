71void UPopup_Result::InitWrongList()
72{
73 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
74 if (!GS)
75 return;
76
77 // 틀린 인덱스 리스트
78 TArray<int32> WrongList;
79 // 전체 캐리어 정보
80 TArray<FScenarioTargetData> ScenarioData;
81
82 if ( QuestType == EQuestType::Read)
83 {
84 WrongList = GS->WrongReadAnswerList;
85 ScenarioData = GS->GetReadScenarioData().target_data;
86 }
87 else if ( QuestType == EQuestType::Listen )
88 {
89 WrongList = GS->WrongListenAnswerList;
90 ScenarioData = GS->GetListenScenarioData().target_data;
91 }
92
93 // 기존 항목 제거
94 VerticalBox->ClearChildren();
95 // WrongList의 각 항목을 UAnswerItem으로 추가
96 for (int32 i = 0; i < WrongList.Num(); i++)
97 {
98 const FScenarioTargetData& SD = ScenarioData[WrongList[i]];
99
100 // UAnswerItem 위젯 생성
101 if (AnswerItemClass)
102 {
103 UAnswerItem* AnswerItem = CreateWidget<UAnswerItem>(this, AnswerItemClass);
104 if (AnswerItem)
105 {
106 // 마지막 인덱스는 정답
107 const bool bCorrect = (i == WrongList.Num() - 1);
108 const int32 Order = i + 1;
109 const int32 Word1Code = FCString::Atoi(*SD.word1.code);
110 const int32 Word2Code = FCString::Atoi(*SD.word2.code);
111
112 // AnswerItem 초기화
113 AnswerItem->InitInfo(QuestType, bCorrect, Order, Word1Code, Word2Code);
114
115 // HorizontalBox에 추가
116 VerticalBox->AddChild(AnswerItem);
117 }
118 }
119 }
120}

122void UPopup_Result::InitReadResult(const FResponseReadResult& ResponseData)
123{
124 if (auto GS = Cast<ALingoGameState>(GetWorld()->GetGameState()))
125 {
126 // Symbol로 타임 처리
127 auto TimeTaken = GS->GetTimeTaken();
128 const int32 Minutes = FMath::FloorToInt(TimeTaken / 60.f);
129 const int32 Seconds = FMath::FloorToInt(TimeTaken) % 60;
130
131 FResultStatData TimeResultData;
132 TimeResultData.WidgetType = EResultItemWidgetType::Symbol;
133 TimeResultData.ColorType = EColorStyleType::Blue;
134 TimeResultData.TitleText = FText::FromString(TEXT("TIME"));
135 TimeResultData.SymbolTextureType = EResourceTextureType::Time;
136 TimeResultData.SymbolValue = FString::Printf(TEXT("%02d:%02d"), Minutes, Seconds);
137 Result_Time->InitData(TimeResultData);
138 }
139
140 FResultStatData GradeResultData;
141 GradeResultData.WidgetType = EResultItemWidgetType::Grade;
142 GradeResultData.ColorType = EColorStyleType::Green;
143 GradeResultData.TitleText = FText::FromString(TEXT("GRADE"));
144 GradeResultData.GradeTextureType = ULingoGameHelper::ConvertGradeString(ResponseData.grade);
145 Result_Grade->InitData(GradeResultData);
146
147 FResultStatData TopRateResultData;
148 TopRateResultData.WidgetType = EResultItemWidgetType::Rate;
149 TopRateResultData.ColorType = EColorStyleType::Red;
150 TopRateResultData.TitleText = FText::FromString(TEXT("TOP"));
151 TopRateResultData.RatePercent = ResponseData.top_percent;
152 Result_TopRate->InitData(TopRateResultData);
153
154 FResultStatData AverageScoreResultData;
155 AverageScoreResultData.WidgetType = EResultItemWidgetType::Symbol;
156 AverageScoreResultData.ColorType = EColorStyleType::Purple;
157 AverageScoreResultData.TitleText = FText::FromString(TEXT("SCORE"));
158 AverageScoreResultData.SymbolTextureType = EResourceTextureType::Score;
159 AverageScoreResultData.SymbolValue = FString::Printf(TEXT("%d"), ResponseData.average_score);
160 Result_AverageScore->InitData(AverageScoreResultData);
161}

204void UPopup_Result::RequestResult()
205{
206 if (auto GS = Cast<ALingoGameState>(GetWorld()->GetGameState()))
207 {
208 // 네트워크 송신
209 if (UKLingoNetworkSystem* Network = UKLingoNetworkSystem::Get(GetWorld()))
210 {
211 if ( QuestType == EQuestType::Read )
212 {
213 FRequestReadResult ReadRequest;
214 ReadRequest.room_id = GS->GetRoomId();
215 ReadRequest.user_id = Cast<APlayerControl>(GetOwningPlayer())->GetUserId();
216 ReadRequest.scenario_id = 1;
217 ReadRequest.stage_type = ULingoGameHelper::GetStageTypeIndex(QuestType);
218 ReadRequest.state_type = 0;
219 ReadRequest.result_time = GS->GetTimeTaken();
220 ReadRequest.wrong_idx = GS->WrongReadAnswerList;
221 Network->RequestReadResult(ReadRequest,
222 FResponseReadResultDelegate::CreateUObject(this, &UPopup_Result::OnResponseReadResult));
223 }
224 else if ( QuestType == EQuestType::Listen )
225 {
226 FRequestListenResult ListenRequest;
227 ListenRequest.room_id = GS->GetRoomId();
228 ListenRequest.user_id = Cast<APlayerControl>(GetOwningPlayer())->GetUserId();
229 ListenRequest.scenario_id = 1;
230 ListenRequest.stage_type = ULingoGameHelper::GetStageTypeIndex(QuestType);
231 ListenRequest.state_type = 0;
232 ListenRequest.result_time = GS->GetTimeTaken();
233 ListenRequest.wrong_idx = GS->WrongListenAnswerList;
234 Network->RequestListenResult(ListenRequest,
235 FResponseListenResultDelegate::CreateUObject(this, &UPopup_Result::OnResponseListenResult));
236 }
237 }
238 }
