53void UPopup_Result::SetTimeTaken()
54{
55 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
56 if (GS)
57 {
58 float TimeRemain = GS->GetRemainMissionTime();
59 TimeTaken = 300 - TimeRemain;
60
61 int32 Minutes = FMath::FloorToInt(TimeRemain / 60.f);
62 int32 Seconds = FMath::FloorToInt(TimeRemain) % 60;
63
64 FString Format = FString::Printf(TEXT("%02d:%02d"), Minutes, Seconds);
65
66 Txt_TimeTaken->SetText(FText::FromString(Format));
67 }
