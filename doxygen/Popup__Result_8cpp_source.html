61void UPopup_Result::SetWrongList()
62{
63 if (!Scrl_WrongList) return;
64
65 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
66 if (!GS) return;
67 // 틀린 인덱스 리스트
68 TArray<int32> WrongList = GS->WrongLuggageList;
69 // 전체 캐리어 정보
70 const TArray<FScenarioTargetData>& ScenarioData = GS->GetScenarioData().target_data;
71
72 // 기존 항목 제거
73 Scrl_WrongList->ClearChildren();
74
75 // WrongList의 각 항목을 텍스트로 추가
76 for (int32 i=0; i<WrongList.Num(); i++)
77 {
78 int32 WrongIndex = WrongList[i];
79 auto SD = ScenarioData[WrongIndex];
80
81 if (UTextBlock* TextBlock = NewObject<UTextBlock>(this))
82 {
83 // 맨 마지막 인덱스는 정답
84 if (i == WrongList.Num()-1)
85 {
86 TextBlock->SetText(FText::FromString(FString::Printf(TEXT("[정답] %s, %s"),
87 *SD.word1.name, *SD.word2.name)));
88 TextBlock->SetColorAndOpacity(FSlateColor(FLinearColor::Red));
89
90 Scrl_WrongList->AddChild(TextBlock);
91
92 break;
93 }
94
95 TextBlock->SetText(FText::FromString(FString::Printf(TEXT("Try %d - [오답] %s, %s"),
96 i+1, *SD.word1.name, *SD.word2.name)));
97 TextBlock->SetColorAndOpacity(FSlateColor(FLinearColor::Black));
98
99 Scrl_WrongList->AddChild(TextBlock);
100 }
101 }
102}

117void UPopup_Result::SetTimeTaken()
118{
119 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
120 if (GS)
121 {
122 float TimeRemain = GS->GetRemainMissionTime();
123 TimeTaken = 300 - TimeRemain;
124
125 int32 Minutes = FMath::FloorToInt(TimeTaken / 60.f);
126 int32 Seconds = FMath::FloorToInt(TimeTaken) % 60;
127
128 FString Format = FString::Printf(TEXT("%02d:%02d"), Minutes, Seconds);
129
130 Txt_TimeTaken->SetText(FText::FromString(Format));
131
132 SetTimeRank();
133 }
134}
