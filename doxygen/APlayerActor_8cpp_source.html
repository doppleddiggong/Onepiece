241void APlayerActor::ApplyAnotherValue()
242{
243 USkeletalMeshComponent* MeshComp = GetMesh();
244 if (!MeshComp)
245 return;
246
247 // Dynamic Material Instance 생성 및 파라미터 설정
248 for (int32 i = 0; i < MeshComp->GetNumMaterials(); ++i)
249 {
250 UMaterialInterface* Material = MeshComp->GetMaterial(i);
251 if (!Material)
252 continue;
253
254 UMaterialInstanceDynamic* DynamicMaterial = Cast<UMaterialInstanceDynamic>(Material);
255 if (!DynamicMaterial)
256 DynamicMaterial = MeshComp->CreateAndSetMaterialInstanceDynamic(i);
257
258 if (DynamicMaterial)
259 DynamicMaterial->SetScalarParameterValue(FName("Another"), AnotherValue);
260 }

291void APlayerActor::Cmd_Move_Implementation(const FVector2D& Axis)
292{
293 if ( !Controller)
294 {
295 return;
296 }
297
298 // Use the controller's rotation to determine movement direction.
299 const FRotator ControlRotation = Controller->GetControlRotation();
300 const EMovementMode CurrentMovementMode = GetCharacterMovement()->MovementMode;
301
302 if (CurrentMovementMode == MOVE_Walking || CurrentMovementMode == MOVE_Falling)
303 {
304 // For ground movement, only use the Yaw rotation to prevent pitching into the ground.
305 const FRotator YawRotation(0.0f, ControlRotation.Yaw, 0.0f);
306
307 // Calculate forward and right vectors based on the Yaw rotation.
308 const FVector ForwardDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::X);
309 const FVector RightDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::Y);
310
311 AddMovementInput(ForwardDirection, Axis.Y);
312 AddMovementInput(RightDirection, Axis.X);
313 }
314}

426void APlayerActor::OnTeleportAllPlayers(FVector TargetLocation)
427{

430
431 // 로컬 플레이어만 페이드 처리

435 PRINTLOG(TEXT("APlayerActor::OnTeleportAllPlayers - Start teleport to %s"), *TargetLocation.ToString());

448 if (!FadeWidget)

451 return;
452 }
453
454 // FadeOut 완료 시 텔레포트 실행
455 FadeWidget->OnFadeOutComplete.AddDynamic(this, &APlayerActor::OnFadeOutCompleteForTeleport);
