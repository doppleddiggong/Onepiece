189void APlayerActor::Cmd_Move_Implementation(const FVector2D& Axis)
190{
191 if ( !Controller)
192 {
193 return;
194 }
195
196 // Use the controller's rotation to determine movement direction.
197 const FRotator ControlRotation = Controller->GetControlRotation();
198 const EMovementMode CurrentMovementMode = GetCharacterMovement()->MovementMode;
199
200 if (CurrentMovementMode == MOVE_Walking || CurrentMovementMode == MOVE_Falling)
201 {
202 // For ground movement, only use the Yaw rotation to prevent pitching into the ground.
203 const FRotator YawRotation(0.0f, ControlRotation.Yaw, 0.0f);
204
205 // Calculate forward and right vectors based on the Yaw rotation.
206 const FVector ForwardDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::X);
207 const FVector RightDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::Y);
208
209 AddMovementInput(ForwardDirection, Axis.Y);
210 AddMovementInput(RightDirection, Axis.X);
211 }
212}
