308void APlayerActor::ApplyAnotherValue()
309{
310 USkeletalMeshComponent* MeshComp = GetMesh();
311 if (!MeshComp)
312 return;
313
314 // Dynamic Material Instance 생성 및 파라미터 설정
315 for (int32 i = 0; i < MeshComp->GetNumMaterials(); ++i)
316 {
317 UMaterialInterface* Material = MeshComp->GetMaterial(i);
318 if (!Material)
319 continue;
320
321 UMaterialInstanceDynamic* DynamicMaterial = Cast<UMaterialInstanceDynamic>(Material);
322 if (!DynamicMaterial)
323 DynamicMaterial = MeshComp->CreateAndSetMaterialInstanceDynamic(i);
324
325 if (DynamicMaterial)
326 DynamicMaterial->SetScalarParameterValue(FName("Another"), AnotherValue);
327 }

358void APlayerActor::Cmd_Move_Implementation(const FVector2D& Axis)
359{
360 if ( !Controller)
361 {
362 return;
363 }
364
365 // Use the controller's rotation to determine movement direction.
366 const FRotator ControlRotation = Controller->GetControlRotation();
367 const EMovementMode CurrentMovementMode = GetCharacterMovement()->MovementMode;
368
369 if (CurrentMovementMode == MOVE_Walking || CurrentMovementMode == MOVE_Falling)
370 {
371 // For ground movement, only use the Yaw rotation to prevent pitching into the ground.
372 const FRotator YawRotation(0.0f, ControlRotation.Yaw, 0.0f);
373
374 // Calculate forward and right vectors based on the Yaw rotation.
375 const FVector ForwardDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::X);
376 const FVector RightDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::Y);
377
378 AddMovementInput(ForwardDirection, Axis.Y);
379 AddMovementInput(RightDirection, Axis.X);
380 }
381}

404void APlayerActor::Cmd_Info_Implementation()
405{
406 auto GS = ULingoGameHelper::GetLingoGameState(GetWorld());
407 auto PS = GetPlayerState<ALingoPlayerState>();
408
409 if ( !GS->IsQuestIng() )
410 {
411 if (auto SpeakStageActor = ULingoGameHelper::GetSpeakStageActor(GetWorld()))
412 {
413 if ( SpeakStageActor->IsMyTurn(PS) )
414 {
415 PlaySpeakInfo( SpeakStageActor->GetCurrentStepIndex() );
416 }
417 }
418 return;
419 }
420
421 if ( GS->GetCurrentQuestType() == EQuestType::Read)
422 {
423 if (auto Popup = UPopupManager::ShowPopupAs<UPopup_ReadQuest>(GetWorld(), EPopupType::ReadQuest))
424 Popup->InitRead(GS->ReadScenarioData);
425 }
426 else
427 {
428 auto QuestRole = GetQuestRole();
429
430 if ( QuestRole == EQuestRole::Both)
431 RequestListenAudio( GS->ListenScenarioData.full_data.Kor);
432 else if ( QuestRole == EQuestRole::OnlyQuestion1)
433 RequestListenAudio( GS->ListenScenarioData.word_data1.Kor);
434 else if ( QuestRole == EQuestRole::OnlyQuestion2)
435 RequestListenAudio( GS->ListenScenarioData.word_data2.Kor);
436 }
437}
