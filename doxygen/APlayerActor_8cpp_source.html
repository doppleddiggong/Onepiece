174UInteractableComponent* APlayerActor::DetectInteractable()
175{
176 FVector CameraLocation = FollowCamera->GetComponentLocation();
177 FVector CameraForward = FollowCamera->GetForwardVector();
178
179 FVector TraceStart = CameraLocation;
180 FVector TraceEnd = TraceStart + (CameraForward * InteractDistance);
181
182 // Hit 결과
183 FHitResult HitResult;
184
185 // Ray trace 실행
186 bool bHit = GetWorld()->LineTraceSingleByChannel(
187 HitResult, TraceStart, TraceEnd, ECC_Visibility);
188
189 DrawDebugLine(GetWorld(), TraceStart, TraceEnd, bHit?FColor::Green : FColor::Red,
190 false, 1.0f, 0, 2.0f);
191
192 // Hit한 경우
193 if (bHit && HitResult.GetActor())
194 {
195 // Actor에서 InteractableComponent 찾기
196 UInteractableComponent* InteractComp =
197 HitResult.GetActor()->FindComponentByClass<UInteractableComponent>();
198
199 if (InteractComp) return InteractComp;
200 }
201
202 return nullptr;
203}

256void APlayerActor::Cmd_Move_Implementation(const FVector2D& Axis)
257{
258 if ( !Controller)
259 {
260 return;
261 }
262
263 // Use the controller's rotation to determine movement direction.
264 const FRotator ControlRotation = Controller->GetControlRotation();
265 const EMovementMode CurrentMovementMode = GetCharacterMovement()->MovementMode;
266
267 if (CurrentMovementMode == MOVE_Walking || CurrentMovementMode == MOVE_Falling)
268 {
269 // For ground movement, only use the Yaw rotation to prevent pitching into the ground.
270 const FRotator YawRotation(0.0f, ControlRotation.Yaw, 0.0f);
271
272 // Calculate forward and right vectors based on the Yaw rotation.
273 const FVector ForwardDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::X);
274 const FVector RightDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::Y);
275
276 AddMovementInput(ForwardDirection, Axis.Y);
277 AddMovementInput(RightDirection, Axis.X);
278 }
279 else if (CurrentMovementMode == MOVE_Flying)
280 {
281 // For flying, use the full 3D rotation of the controller.
282 const FRotator FullRotation(ControlRotation.Pitch, ControlRotation.Yaw, 0.0f);
283
284 const FVector ForwardDirection = FRotationMatrix(FullRotation).GetUnitAxis(EAxis::X);
285 const FVector RightDirection = FRotationMatrix(FullRotation).GetUnitAxis(EAxis::Y);
286
287 AddMovementInput(ForwardDirection, Axis.Y);
288 AddMovementInput(RightDirection, Axis.X);
289 }
290}
