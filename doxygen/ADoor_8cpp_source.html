73void ADoor::OnDoorMessage(int32 InDoorIndex, bool bInOpen, AActor* EventInstigator)
74{
75 if (InDoorIndex != DoorIndex)
76 return;
77
78 CurCount += (bInOpen ? 1 : -1);
79 CurCount = FMath::Clamp(CurCount, 0, ReqCount); // 안전
80
81 if (CurCount >= ReqCount)
82 OpenDoor();
83 else
84 CloseDoor();
85}
