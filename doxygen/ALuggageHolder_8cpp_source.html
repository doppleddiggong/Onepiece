118void ALuggageHolder::OnBoxOverlapBegin(
119 UPrimitiveComponent* OverlappedComponent,
120 AActor* OtherActor,
121 UPrimitiveComponent* OtherComp,
122 int32 OtherBodyIndex,
123 bool bFromSweep,
124 const FHitResult& SweepResult)
125{
126 if (!OtherActor)
127 return;
128
129 if (!HasAuthority())
130 return;
131
132 if ( bIsActivated )
133 return;
134
135 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
136 if (!GS)
137 return;
138
139 // Luggage인지 확인
140 if (auto Luggage = Cast<Aluggage>(OtherActor))
141 {
142 // Hook 중인 Luggage는 무시 (PlayerActor가 들고 있는 상태)
143 if (Luggage->bIsBeingHooked)
144 return;
145
146 const bool bSuccess = CheckLuggage(Luggage, GS->GetReadScenarioData().correct_answer_index );
147
148 // 블루프린트 이벤트 호출
149 OnActivate(bSuccess);
150
151 if (bSuccess)
152 {
153 // 정답인 경우
154 int32 CorrectIdx = Luggage->GetSpawnIdx();
155
156 // 마커 표시
157 GS->SetAllCompassVisibility(false);
158 GS->SetCompassVisibilityByTag("ReadQuestEnd", true);
159
160 FTimerHandle TimerHandle;
161 GetWorldTimerManager().SetTimer(TimerHandle, [this, CorrectIdx]
162 {
163 // 모든 클라이언트에 정답 인덱스와 함께 결과 팝업 표시
164 Multicast_ShowResultPopup(CorrectIdx);
165
166 }, 0.5f, false);
167 }
168 else
169 {
170 // 오답인 경우
171 FTimerHandle TimerHandle;
172 GetWorldTimerManager().SetTimer(TimerHandle, [this, Luggage, GS]
173 {
174 GS->AddWrongReadAnswer(Luggage->GetSpawnIdx());
175
176 // 모든 클라이언트에 오답 메시지 표시
177 Multicast_ShowWrongPopup(Luggage->GetColor(), Luggage->GetPattern());
178
179 // Dissolve 재생
180 GetWorldTimerManager().SetTimer(DestroyTimerHandle, [this, Luggage]
181 {
182 if (Luggage->UpdateDissolve())
183 {
184 GetWorldTimerManager().ClearTimer(DestroyTimerHandle);
185 // 큐브 소거 (서버에서만, 자동 복제됨)
186 Luggage->Destroy();
187 }
188 },GetWorld()->DeltaTimeSeconds, true);
189
190 }, 0.5f, false);
191 }
192 }
193}

195bool ALuggageHolder::CheckLuggage(Aluggage* TargetLuggage, int CorrectIndex)
196{
197 if (!TargetLuggage)
198 return false;
199
200 // Luggage의 모든 충돌 비활성화 (pickup, hook 등 모든 상호작용 차단)
201 TargetLuggage->SetAllCollision(false);
202
203 // ReadQuest 정답 인덱스 가져오기
204 if (CorrectIndex == TargetLuggage->GetSpawnIdx())
205 {
206 // Success: Luggage를 HoldPos 위치보다 살짝 위에 배치
207 FVector ActivatedLocation = HoldPos->GetComponentLocation();
208 ActivatedLocation.Z += ActivatedHeightOffset;
209 TargetLuggage->SetActorLocation(ActivatedLocation);
210 TargetLuggage->SetActorRotation(HoldPos->GetComponentRotation());
211
212 // Activate 상태로 전환
213 bIsActivated = true;
214 CurTarget = TargetLuggage;
215
216 // 서버에서도 머티리얼 업데이트 (클라이언트는 OnRep_IsActivated에서 호출됨)
217 UpdateActivateState(true);
218
219 return true;
220 }
221 else
222 {
223 // Fail: 서버에서 머티리얼 업데이트 (오답)
224 bIsActivated = false;
225 UpdateActivateState(false);
226
227 return false;
228 }
229}

231void ALuggageHolder::UpdateActivateState(bool State)
232{
233 // 머티리얼 파라미터 설정 (비활성화)
234 if (MeshComponent && MeshComponent->GetNumMaterials() > 0)
235 {
236 UMaterialInstanceDynamic* DynamicMaterial = Cast<UMaterialInstanceDynamic>(MeshComponent->GetMaterial(0));
237 if (!DynamicMaterial)
238 DynamicMaterial = MeshComponent->CreateDynamicMaterialInstance(0);
239
240 if (DynamicMaterial)
241 DynamicMaterial->SetScalarParameterValue(FName("Activate"), State ? 1.0f : 0.0f);
242 }
243}

245void ALuggageHolder::BillboardInteractWidget()
246{
247 // 위젯이 없으면 빌보드화 안 함
248 if (!WidgetGuideComp)
249 return;
250
251 // Visibility 체크 - 보이지 않으면 빌보드화 안 함
252 if (!WidgetGuideComp->IsVisible())
253 return;
254
255 // 카메라 가져오기
256 AActor* Camera = UGameplayStatics::GetPlayerCameraManager(GetWorld(), 0);
257 if (!Camera)
258 return;
259
260 // 카메라를 향하도록 회전 계산
261 FRotator Rotation = UKismetMathLibrary::MakeRotFromXZ( -Camera->GetActorForwardVector(), Camera->GetActorUpVector() );
262 Rotation.Pitch = 0;
263
264 // 위젯 회전 설정
265 WidgetGuideComp->SetWorldRotation(Rotation);
266}
