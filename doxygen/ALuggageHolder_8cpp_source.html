96void ALuggageHolder::OnBoxOverlapBegin(
97 UPrimitiveComponent* OverlappedComponent,
98 AActor* OtherActor,
99 UPrimitiveComponent* OtherComp,
100 int32 OtherBodyIndex,
101 bool bFromSweep,
102 const FHitResult& SweepResult)
103{
104 if (!OtherActor)
105 return;
106
107 if (!HasAuthority())
108 return;
109
110 if ( bIsActivated )
111 return;
112
113 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
114 if (!GS)
115 return;
116
117 // Luggage인지 확인
118 if (auto Luggage = Cast<Aluggage>(OtherActor))
119 {
120 // Hook 중인 Luggage는 무시 (PlayerActor가 들고 있는 상태)
121 if (Luggage->bIsBeingHooked)
122 return;
123
124 const bool bSuccess = CheckLuggage(Luggage, GS->GetReadScenarioData().correct_answer_index );
125
126 // 블루프린트 이벤트 호출
127 OnActivate(bSuccess);
128
129 if (bSuccess)
130 {
131 // 정답인 경우
132 int32 CorrectIdx = Luggage->GetSpawnIdx();
133 FTimerHandle TimerHandle;
134 GetWorldTimerManager().SetTimer(TimerHandle, [this, CorrectIdx]
135 {
136 // 모든 클라이언트에 정답 인덱스와 함께 결과 팝업 표시
137 Multicast_ShowResultPopup(CorrectIdx);
138 }, 0.5f, false);
139 }
140 else
141 {
142 // 오답인 경우
143 FTimerHandle TimerHandle;
144 GetWorldTimerManager().SetTimer(TimerHandle, [this, Luggage, GS]
145 {
146 GS->AddWrongReadAnswer(Luggage->GetSpawnIdx());
147
148 // 모든 클라이언트에 오답 메시지 표시
149 Multicast_ShowWrongPopup(Luggage->GetColor(), Luggage->GetPattern());
150
151 // 큐브 소거 (서버에서만, 자동 복제됨)
152 Luggage->Destroy();
153 }, 0.5f, false);
154 }
155 }
156}

158bool ALuggageHolder::CheckLuggage(Aluggage* TargetLuggage, int CorrectIndex)
159{
160 if (!TargetLuggage)
161 return false;
162
163 // Luggage의 모든 충돌 비활성화 (pickup, hook 등 모든 상호작용 차단)
164 TargetLuggage->SetAllCollision(false);
165
166 // ReadQuest 정답 인덱스 가져오기
167 if (CorrectIndex == TargetLuggage->GetSpawnIdx())
168 {
169 // Success: Luggage를 HoldPos 위치보다 살짝 위에 배치
170 FVector ActivatedLocation = HoldPos->GetComponentLocation();
171 ActivatedLocation.Z += ActivatedHeightOffset;
172 TargetLuggage->SetActorLocation(ActivatedLocation);
173 TargetLuggage->SetActorRotation(HoldPos->GetComponentRotation());
174
175 // Activate 상태로 전환
176 bIsActivated = true;
177 CurTarget = TargetLuggage;
178
179 // 서버에서도 머티리얼 업데이트 (클라이언트는 OnRep_IsActivated에서 호출됨)
180 UpdateActivateState(true);
181
182 return true;
183 }
184 else
185 {
186 // Fail: 서버에서 머티리얼 업데이트 (오답)
187 bIsActivated = false;
188 UpdateActivateState(false);
189
190 return false;
191 }
192}

194void ALuggageHolder::UpdateActivateState(bool State)
195{
196 // 머티리얼 파라미터 설정 (비활성화)
197 if (MeshComponent && MeshComponent->GetNumMaterials() > 0)
198 {
199 UMaterialInstanceDynamic* DynamicMaterial = Cast<UMaterialInstanceDynamic>(MeshComponent->GetMaterial(0));
200 if (!DynamicMaterial)
201 DynamicMaterial = MeshComponent->CreateDynamicMaterialInstance(0);
202
203 if (DynamicMaterial)
204 DynamicMaterial->SetScalarParameterValue(FName("Activate"), State ? 1.0f : 0.0f);
205 }
206}
