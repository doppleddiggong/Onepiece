48void UPopup_SpeakResult::InitQuestionList()
49{
50 auto PS= ULingoGameHelper::GetLingoPlayerState(GetWorld());
51
52 VerticalBox->ClearChildren();
53
54 for (int32 i = 0; i < SpeakResult.scores.Num(); ++i)
55 {
56 FSpeakResultItem InItemData;
57 InItemData.Index = i+1;
58 InItemData.Question = PS->SpeakScenarioData.speak_quest_data[i].eng;
59 InItemData.Feedback = SpeakResult.scores[i].desc;
60 InItemData.Score = SpeakResult.scores[i].score;
61
62 if( auto ItemWidget = CreateWidget<UPopup_SpeakResultItem>(GetWorld(), AnswerItemClass) )
63 {
64 ItemWidget->InitData(InItemData);
65 VerticalBox->AddChildToVerticalBox(ItemWidget);
66
67 if (i < SpeakResult.scores.Num() - 1)
68 {
69 USpacer* Spacer = NewObject<USpacer>(this);
70 if (Spacer)
71 {
72 Spacer->SetSize(FVector2D(1.0f, 15.0f));
