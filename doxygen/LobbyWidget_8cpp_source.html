122void ULobbyWidget::OnFindComplete(int32 idx, FString sessionName)
123{
124 if (idx == -1)
125 {
126 // 조회 버튼 문구 변경
127 Txt_Update->SetText(FText::FromString(TEXT("Update Session")));
128 // 조회 버튼 활성화
129 Btn_Update->SetIsEnabled(true);
130 }
131 else
132 {
133 // sessionInfoWidget 만들자.
134 USessionInfoWidget* item = CreateWidget<USessionInfoWidget>(GetWorld(), sessionInfoWidget);
135 // 만들어진 item 을 scrollSessionList 에 추가
136 Scrl_SessionList->AddChild(item);
137 // 만들어지 item 정보 설정
138 item->SetSessionInfo(idx, sessionName);
139 }
140}
