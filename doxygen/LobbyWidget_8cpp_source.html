14void ULobbyWidget::NativeConstruct()
15{
16 Super::NativeConstruct();
17
18 GI = Cast<ULingoGameInstance>(GetGameInstance());
19 GI->onFindComplete.BindUObject(this, &ULobbyWidget::OnFindComplete);
20
21 // Lobby Canvas
22 if (Btn_Single) Btn_Single->OnClicked.AddDynamic(this, &ULobbyWidget::OnSingleClicked);
23 if (Btn_Host) Btn_Host->OnClicked.AddDynamic(this, &ULobbyWidget::OnHostClicked);
24 if (Btn_Join) Btn_Join->OnClicked.AddDynamic(this, &ULobbyWidget::OnJoinClicked);
25 if (Btn_LogOut) Btn_LogOut->OnClicked.AddDynamic(this, &ULobbyWidget::OnLogOutClicked);
26
27 // Create Canvas
28 if (Btn_Create) Btn_Create->OnClicked.AddDynamic(this, &ULobbyWidget::OnCreateClicked);
29 EdtTxt_SessionName->OnTextChanged.AddDynamic(this, &ULobbyWidget::OnValueChangedSessionName);
30
31 // Join Canvas
32 if (Btn_Update) Btn_Update->OnClicked.AddDynamic(this, &ULobbyWidget::OnClickFind);
33}

82void ULobbyWidget::OnFindComplete(int32 idx, FString sessionName)
83{
84 if (idx == -1)
85 {
86 // 조회 버튼 문구 변경
87 Txt_Update->SetText(FText::FromString(TEXT("Update Session")));
88 // 조회 버튼 활성화
89 Btn_Update->SetIsEnabled(true);
90 }
91 else
92 {
93 // sessionInfoWidget 만들자.
94 USessionInfoWidget* item = CreateWidget<USessionInfoWidget>(GetWorld(), sessionInfoWidget);
95 // 만들어진 item 을 scrollSessionList 에 추가
96 Scrl_SessionList->AddChild(item);
97 // 만들어지 item 정보 설정
98 item->SetSessionInfo(idx, sessionName);
99 }
100}
