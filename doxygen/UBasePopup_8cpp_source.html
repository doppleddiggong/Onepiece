43void UBasePopup::UpdateAnimation(float InDeltaTime)
44{
45 if (!bIsOpenAnim)
46 {
47 return;
48 }
49
50 OpenElapsedTime += InDeltaTime;
51
52 const float Alpha = FMath::Clamp(OpenElapsedTime / OpenDuration, 0.0f, 1.0f);
53 const float EasedAlpha = FEaseHelper::Ease(Alpha, OpenEaseType);
54 const float NewScale = FMath::Lerp(OpenStartScale, OpenTargetScale, EasedAlpha);
55
56 SetRenderScale(FVector2D(NewScale, NewScale));
57
58 if (Alpha >= 1.0f)
59 {
60 bIsOpenAnim = false;
61 }
62}
