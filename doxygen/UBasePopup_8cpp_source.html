35void UBasePopup::UpdateAnimation(float InDeltaTime)
36{
37 if (!bIsOpenAnim)
38 {
39 return;
40 }
41
42 OpenElapsedTime += InDeltaTime;
43
44 const float Alpha = FMath::Clamp(OpenElapsedTime / OpenDuration, 0.0f, 1.0f);
45 const float EasedAlpha = FMath::InterpEaseOut(0.0f, 1.0f, Alpha, OpenEaseExponent);
46 const float NewScale = FMath::Lerp(OpenStartScale, OpenTargetScale, EasedAlpha);
47
48 SetRenderScale(FVector2D(NewScale, NewScale));
49
50 if (Alpha >= 1.0f)
51 {
52 bIsOpenAnim = false;
53 }
54}
