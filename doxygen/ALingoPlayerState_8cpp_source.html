103void ALingoPlayerState::Server_NotifySpeakDataReady_Implementation()
104{
105 // 월드에서 SpeakStageActor를 찾습니다.
106 if (UWorld* World = GetWorld())
107 {
108 for (TActorIterator<ASpeakStageActor> It(World); It; ++It)
109 {
110 ASpeakStageActor* SpeakStage = *It;
111 if (SpeakStage)
112 {
113 // SpeakStage를 통해 퀘스트를 시작합니다.
114 SpeakStage->StartStageForPlayer(this);
115 PRINTLOG(TEXT("[ALingoPlayerState] Client is ready. Starting SpeakQuest for: %s"), *GetPlayerName());
116 return; // 첫 번째로 찾은 SpeakStage를 사용하고 종료
117 }
118 }
119 }
120
121 PRINTLOG(TEXT("[ALingoPlayerState] Server_NotifySpeakDataReady - ASpeakStageActor not found in world!"));
122}
