31void UChatWidget::SendMessage(FResponseUserMe sendUser, FText inMessage)
32{
33 auto* PC = Cast<APlayerControl>(GetWorld()->GetFirstPlayerController<APlayerController>());
34 if (!PC)
35 {
36 PRINTLOG(TEXT("Cannot get playercontroller"));
37 }
38
39 // 플레이어 정보 가져오기
40 FResponseUserMe info = PC->GetUserInfo();
41
42 UChatBoxWidget* newChat = nullptr;
43
44 // 플레이어에 따라 분류
45 if (info.id == sendUser.id)
46 {
47 // 왼쪽 정렬
48 PRINTLOG(TEXT("left - sendUserId : %d"), sendUser.id);
49
50 newChat = CreateWidget<UChatBoxWidget>(GetWorld(), LeftChatBoxWidgetClass);
51 }
52 else
53 {
54 // 오른쪽 정렬
55 PRINTLOG(TEXT("right - sendUserId : %d"), sendUser.id);
56
57 newChat = CreateWidget<UChatBoxWidget>(GetWorld(), RightChatBoxWidgetClass);
58 }
59
60 newChat->SetContent(inMessage);
61
62 // TODO: 플레이어 색상 넣기
63 newChat->SetPlayerBGColor(FColor::FromHex(TEXT("E94C4CFF")));
64 // 플레이어 이름 넣기
65 newChat->SetPlayerName(FText::FromString(info.username));
66
67 // 현재 스크롤 값 & 마지막 스크롤 값
68 float scrollOffset = ScrollBox_ChatBox->GetScrollOffset();
69 float scrollOffsetOfEnd = ScrollBox_ChatBox->GetScrollOffsetOfEnd();
70 PRINTLOG(TEXT("before scrollend - ScrollOffset: %f, ScrollOffsetOfEnd: %f"), scrollOffset, scrollOffsetOfEnd);
71
72 ScrollBox_ChatBox->SetScrollOffset(scrollOffsetOfEnd);
73
74 // ChatBox에 추가
75 ScrollBox_ChatBox->AddChild(newChat);
76
77 // 스크롤이 맨 마지막일 때
78 if (info.id == sendUser.id || scrollOffset == scrollOffsetOfEnd)
79 {
80 FTimerHandle handle;
81 GetWorld()->GetTimerManager().SetTimer(handle, [this, &scrollOffset, &scrollOffsetOfEnd]()
82 {
83 // 스크롤 위치를 맨 끝으로 해라!
84 ScrollBox_ChatBox->ScrollToEnd();
85 }, 0.1f, false);
86 }
87}
