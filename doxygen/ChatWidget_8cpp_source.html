37void UChatWidget::SendMessage(FResponseUserMe sendUser, FText inMessage)
38{
39 auto* PC = Cast<APlayerControl>(GetWorld()->GetFirstPlayerController<APlayerController>());
40 if (!PC)
41 {
42 PRINTLOG(TEXT("Cannot get playercontroller"));
43 }
44
45 // 메시지 받은 플레이어 정보 가져오기
46 FResponseUserMe info = PC->GetUserInfo();
47 bool bIsSender = (info.id == sendUser.id);
48
49 // ChatBox 생성
50 UChatBoxWidget* newChat = CreateChatBox(bIsSender);
51 newChat->SetMessage(inMessage);
52
53 // 플레이어 색상 & 넣기
54 int32 PlayerIndex = -1;
55 if (ALingoPlayerState* PS = PC->GetPlayerState<ALingoPlayerState>())
56 {
57 if (AGameStateBase* GS = GetWorld()->GetGameState())
58 {
59 PlayerIndex = GS->PlayerArray.IndexOfByKey(PS);
60 }
61 }
62
63 newChat->SetPlayerProfile(
64 sendUser.GetChatProfileBg(PlayerIndex),
65 sendUser.GetChatProfileTextureType(PlayerIndex));
66
67 newChat->SetPlayerName(FText::FromString(sendUser.username));
68 newChat->SetMessage(inMessage);
69 newChat->SetChatBubbleColor(bIsSender);
70
71
72 // 현재 스크롤 값 & 마지막 스크롤 값
73 float scrollOffset = ScrollBox_ChatBox->GetScrollOffset();
74 float scrollOffsetOfEnd = ScrollBox_ChatBox->GetScrollOffsetOfEnd();
75 PRINTLOG(TEXT("before scrollend - ScrollOffset: %f, ScrollOffsetOfEnd: %f"), scrollOffset, scrollOffsetOfEnd);
76
77 ScrollBox_ChatBox->SetScrollOffset(scrollOffsetOfEnd);
78
79 // ChatBox에 추가 & 정렬
80 VerticalBox_Content->AddChild(newChat);
81
82 UScrollBoxSlot* parentSlot = Cast<UScrollBoxSlot>(newChat->Slot);
83 if (parentSlot)
84 {
85 parentSlot->SetHorizontalAlignment((bIsSender ? HAlign_Left : HAlign_Right));
86 }
87
88 // 메시지 받은 자가 Sender일 때 or 스크롤이 맨 마지막일 때
89 if (bIsSender || scrollOffset == scrollOffsetOfEnd)
90 {
91 FTimerHandle handle;
92 GetWorld()->GetTimerManager().SetTimer(handle, [this]()
93 {
94 // 스크롤 위치를 맨 끝으로 해라!
95 ScrollBox_ChatBox->ScrollToEnd();
96 }, 0.1f, false);
97 }
98
99 UWidgetBlueprintLibrary::SetFocusToGameViewport();
100 PC->SetInputMode(FInputModeGameOnly());
101}
