52void ALuggageManager::SpawnLuggage()
53{
54 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
55 if (!GS) return;
56
57 const TArray<FScenarioTargetData>& ScenarioData = GS->GetScenarioData().target_data;
58
59 // 모두 스폰했으면 타이머 중지
60 if (CurrentSpawnIndex >= ScenarioData.Num())
61 {
62 GetWorldTimerManager().ClearTimer(SpawnTimerHandle);
63 return;
64 }
65
66 // 현재 인덱스의 luggage 스폰
67 auto SD = ScenarioData[CurrentSpawnIndex];
68 Aluggage* NewLuggage = GetWorld()->SpawnActor<Aluggage>(
69 LuggageClass,
70 GetActorLocation(),
71 FRotator::ZeroRotator
72 );
73
74 if (NewLuggage)
75 {
76 NewLuggage->SpawnIdx = CurrentSpawnIndex;
77 int32 ColorIdx = FCString::Atoi(*SD.word2.code);
78 NewLuggage->ApplyColorToMesh(ColorIdx);
79 NewLuggage->ApplyPatternToMesh(SD.word1.name);
80 }
81
82 CurrentSpawnIndex++;
83}
