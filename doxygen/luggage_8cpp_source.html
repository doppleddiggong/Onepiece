82void Aluggage::ApplyColorToMesh(int32 InColorIdx)
83{
84 ColorIndex = InColorIdx;
85
86 FColorData ColorData;
87 if (UGameDataManager::Get(GetWorld())->GetColorData(InColorIdx, ColorData))
88 {
89 FLinearColor Color = ColorData.GetLinearColor();
90
91 UMaterialInterface* OriginalMaterial = Mesh3Comp->GetMaterial(0);
92 if (OriginalMaterial)
93 {
94 UMaterialInstanceDynamic* NewMaterial = UMaterialInstanceDynamic::Create(OriginalMaterial, this);
95 if (NewMaterial && Mesh3Comp)
96 {
97 // BaseColorFactor로 변경!
98 NewMaterial->SetVectorParameterValue(FName("BaseColorFactor"), Color);
99 Mesh3Comp->SetMaterial(0, NewMaterial);
100 }
101 }
102 }
103}

125void Aluggage::OnInteract(AActor* Interactor)
126{
127 if (!Interactor)
128 {
129 PRINTLOG(TEXT("[Luggage] OnInteract - Interactor is null"));
130 return;
131 }
132
133 // 플레이어 컨트롤러와 PlayerState 가져오기
134 APlayerController* PC = Cast<APlayerController>(Interactor->GetInstigatorController());
135 if (!PC)
136 {
137 PRINTLOG(TEXT("[Luggage] OnInteract - PlayerController is null"));
138 return;
139 }
140
141 APlayerState* PS = PC->GetPlayerState<APlayerState>();
142 if (!PS)
143 {
144 PRINTLOG(TEXT("[Luggage] OnInteract - PlayerState is null"));
145 return;
146 }
147
148 PRINTLOG(TEXT("[Luggage] OnInteract - Player selected luggage with Target1: %s, Target2: %s"), *Target1, *Target2);
149
150 // 서버에 선택 알림
151 ServerNotifySelection(PS);
152}

154void Aluggage::ServerNotifySelection_Implementation(APlayerState* Player)
155{
156 if (!Player)
157 {
158 PRINTLOG(TEXT("[Luggage] ServerNotifySelection - Player is null"));
159 return;
160 }
161
162 PRINTLOG(TEXT("[Luggage] ServerNotifySelection - Processing selection for player"));
163
164 // GameMode에 캐리어 선택 알림
165 ALingoGameMode* GameMode = GetWorld()->GetAuthGameMode<ALingoGameMode>();
166 if (GameMode)
167 {
168 GameMode->HandleCarrierSelection(Player, this);
169 }
170 else
171 {
172 PRINTLOG(TEXT("[Luggage] ServerNotifySelection - GameMode is null or not ALingoGameMode"));
173 }
174}
