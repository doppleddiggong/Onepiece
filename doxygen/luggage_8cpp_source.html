165void Aluggage::ApplyColorToMesh(int32 InColorIdx)
166{
167 ColorIdx = InColorIdx;
168
169 FColorData ColorData;
170 if (UGameDataManager::Get(GetWorld())->GetColorData(InColorIdx, ColorData))
171 {
172 // Color 변수 설정 (모든 클라이언트에서 데이터 테이블에서 가져옴)
173 Color = ColorData.Desc;
174
175 // if ( auto InfoWidget = Cast<ULuggageInfoWidget>(WidgetComp->GetWidget()))
176 // InfoWidget->UpdateType1Data(ColorData.Desc);
177
178 UMaterialInterface* OriginalMaterial = Mesh3Comp->GetMaterial(0);
179 if (OriginalMaterial)
180 {
181 UMaterialInstanceDynamic* NewMaterial = UMaterialInstanceDynamic::Create(OriginalMaterial, this);
182 if (NewMaterial && Mesh3Comp)
183 {
184 // BaseColorFactor로 변경
185 NewMaterial->SetVectorParameterValue(FName("BaseColorFactor"), ColorData.GetLinearColor());
186 Mesh3Comp->SetMaterial(0, NewMaterial);
187 }
188 }
189 }
190}

192void Aluggage::ApplyPatternToMesh(int32 InPatternIdx)
193{
194 PatternIdx = InPatternIdx;
195
196 // 데칼 바꾸기
197
198 FReadData ReadData;
199 UGameDataManager::Get(GetWorld())->GetReadData(InPatternIdx, ReadData);
200
201 // Pattern 변수 설정 (모든 클라이언트에서 데이터 테이블에서 가져옴)
202 Pattern = ReadData.Word;
203
204 // if ( auto InfoWidget = Cast<ULuggageInfoWidget>(WidgetComp->GetWidget()))
205 // InfoWidget->UpdateType2Data(ReadData.Word);
206
207 UTexture2D* LoadedTexture = nullptr;
208 if (ReadData.Texture.IsValid())
209 LoadedTexture = ReadData.Texture.Get();
210 else
211 LoadedTexture = ReadData.Texture.LoadSynchronous();
212
213 if ( UMaterialInterface* OriginalMaterial = Mesh2Comp->GetMaterial(0) )
214 {
215 UMaterialInstanceDynamic* NewMaterial = UMaterialInstanceDynamic::Create(OriginalMaterial, this);
216 if (NewMaterial)
217 {
218 // BaseColorFactor로 변경
219 NewMaterial->SetTextureParameterValue(FName("TextureParam"), LoadedTexture);
220 Mesh2Comp->SetMaterial(0, NewMaterial);
221 }
222 }
223}

276void Aluggage::OnInteract(AActor* Interactor)
277{
278 if (!Interactor)
279 {
280 PRINTLOG(TEXT("[Luggage] OnInteract - Interactor is null"));
281 return;
282 }
283
284 // 플레이어 컨트롤러와 PlayerState 가져오기
285 APlayerController* PC = Cast<APlayerController>(Interactor->GetInstigatorController());
286 if (!PC)
287 {
288 PRINTLOG(TEXT("[Luggage] OnInteract - PlayerController is null"));
289 return;
290 }
291
292 APlayerState* PS = PC->GetPlayerState<APlayerState>();
293 if (!PS)
294 {
295 PRINTLOG(TEXT("[Luggage] OnInteract - PlayerState is null"));
296 return;
297 }
298
299 //PRINTLOG(TEXT("[Luggage] OnInteract - Player selected luggage with Target1: %s, Target2: %s"), *Target1, *Target2);
300
301 // 서버에 선택 알림
302 ServerNotifySelection(PS);
303}

305void Aluggage::ServerNotifySelection_Implementation(APlayerState* Player)
306{
307 if (!Player)
308 {
309 PRINTLOG(TEXT("[Luggage] ServerNotifySelection - Player is null"));
310 return;
311 }
312
313 PRINTLOG(TEXT("[Luggage] ServerNotifySelection - Processing selection for player"));
314
315 // GameMode에 캐리어 선택 알림
316 ALingoGameMode* GameMode = GetWorld()->GetAuthGameMode<ALingoGameMode>();
317 if (GameMode)
318 {
319 GameMode->HandleCarrierSelection(Player, this);
320 }
321 else
322 {
323 PRINTLOG(TEXT("[Luggage] ServerNotifySelection - GameMode is null or not ALingoGameMode"));
324 }

340void Aluggage::RequestListenAudio(const FString& AudioText)
341{
342 if (bIsRequest)
343 return;
344
345 if (auto KLingoNetwork = UKLingoNetworkSystem::Get(GetWorld()))
346 {
347 bIsRequest = true;
348
349 KLingoNetwork->RequestListenAudio(
350 AudioText,
351 FResponseListenAudioDelegate::CreateUObject(this, &Aluggage::OnResponseListenAudio)
352 );
353 }
354}

356void Aluggage::OnResponseListenAudio(FResponseListenAudio& ResponseData, bool bWasSuccessful)
357{
358 bIsRequest = false;
359
360 if (bWasSuccessful)
361 {
362 if (auto PlayerActor = ULingoGameHelper::GetPlayerActor(this))
363 PlayerActor->PlayTTSAudio(ResponseData.audio_base64);
364 }
