111void Aluggage::ApplyColorToMesh(int32 InColorIdx)
112{
113 ColorIdx = InColorIdx;
114
115 FColorData ColorData;
116 if (UGameDataManager::Get(GetWorld())->GetColorData(InColorIdx, ColorData))
117 {
118 FLinearColor LinearColor = ColorData.GetLinearColor();
119
120 UMaterialInterface* OriginalMaterial = Mesh3Comp->GetMaterial(0);
121 if (OriginalMaterial)
122 {
123 UMaterialInstanceDynamic* NewMaterial = UMaterialInstanceDynamic::Create(OriginalMaterial, this);
124 if (NewMaterial && Mesh3Comp)
125 {
126 // BaseColorFactor로 변경
127 NewMaterial->SetVectorParameterValue(FName("BaseColorFactor"), LinearColor);
128 Mesh3Comp->SetMaterial(0, NewMaterial);
129 }
130 }
131 }

162void Aluggage::BillboardInfoWidget()
163{

178void Aluggage::OnInteract(AActor* Interactor)
179{
180 if (!Interactor)
181 {
182 PRINTLOG(TEXT("[Luggage] OnInteract - Interactor is null"));
183 return;
184 }
185
186 // 플레이어 컨트롤러와 PlayerState 가져오기
187 APlayerController* PC = Cast<APlayerController>(Interactor->GetInstigatorController());
188 if (!PC)
189 {
190 PRINTLOG(TEXT("[Luggage] OnInteract - PlayerController is null"));
191 return;
192 }
193
194 APlayerState* PS = PC->GetPlayerState<APlayerState>();
195 if (!PS)
196 {
197 PRINTLOG(TEXT("[Luggage] OnInteract - PlayerState is null"));
198 return;
199 }
200
201 //PRINTLOG(TEXT("[Luggage] OnInteract - Player selected luggage with Target1: %s, Target2: %s"), *Target1, *Target2);
202
203 // 서버에 선택 알림
204 ServerNotifySelection(PS);
205}

207void Aluggage::ServerNotifySelection_Implementation(APlayerState* Player)
208{
209 if (!Player)
210 {
211 PRINTLOG(TEXT("[Luggage] ServerNotifySelection - Player is null"));
212 return;
213 }
214
215 PRINTLOG(TEXT("[Luggage] ServerNotifySelection - Processing selection for player"));
216
217 // GameMode에 캐리어 선택 알림
218 ALingoGameMode* GameMode = GetWorld()->GetAuthGameMode<ALingoGameMode>();
219 if (GameMode)
220 {
221 GameMode->HandleCarrierSelection(Player, this);
222 }
223 else
224 {
225 PRINTLOG(TEXT("[Luggage] ServerNotifySelection - GameMode is null or not ALingoGameMode"));
226 }
227}
