60void UCompassWidget::SetMarkerPosition(UImage* InMarker, float TargetRotation, bool bSideLock)
61{
62 float CompassSize = Img_Compass->GetDesiredSize().X;
63 float MarkerRotation = TargetRotation * (CompassSize/360.f);
64
65 UCanvasPanelSlot* MarkerSlot = Cast<UCanvasPanelSlot>(InMarker->Slot);
66 if (MarkerSlot)
67 {
68 if (!bSideLock)
69 {
70 MarkerSlot->SetPosition(FVector2D(MarkerRotation, 0.f));
71 }
72 else
73 {
74 UCanvasPanelSlot* CompassSlot = Cast<UCanvasPanelSlot>(Pnl_Compass->Slot);
75 if (CompassSlot)
76 {
77 float PositionX = (CompassSlot->GetSize().X / 2);
78
79 if (FMath::Abs(MarkerRotation) < PositionX)
80 {
81 MarkerSlot->SetPosition(FVector2D(MarkerRotation, 0.f));
82 }
83 else if (MarkerRotation > PositionX)
84 {
85 MarkerSlot->SetPosition(FVector2D(PositionX, 0.f));
86 }
87 else
88 {
89 MarkerSlot->SetPosition(FVector2D(-PositionX, 0.f));
90 }
91
92 }
93 }
94 }

98void UCompassWidget::UpdateCompassMarkers(TArray<float>& CompassMarkers)
99{
100 for (int32 i=0; i<CompassMarkers.Num(); i++)
101 {
102 if (Markers[i])
103 {
104 SetMarkerPosition(Markers[i], CompassMarkers[i], false);
105 Markers[i]->SetVisibility(ESlateVisibility::Visible);
106 }
107 else
108 {
109 UImage* NewImage = AddCompassMarker();
110 SetMarkerPosition(NewImage, CompassMarkers[i], false);
111 }
112 }
113
114 if (CompassMarkers.Num() < Markers.Num())
115 {
116 for (int32 i=CompassMarkers.Num(); i<Markers.Num(); i++)
117 {
118 Markers[i]->SetVisibility(ESlateVisibility::Hidden);
119 }
120 }
121}
