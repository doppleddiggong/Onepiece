39UImage* UCompassWidget::AddCompassMarker(ECompassMarkerType MarkerType)
40{
41 UImage* NewMarker = NewObject<UImage>(this);
42 if (NewMarker)
43 {
44 UTexture2D* CompassTex = GetTextureForMarkerType(MarkerType);
45 if (CompassTex)
46 {
47 NewMarker->SetBrushFromTexture(CompassTex);
48 }
49
50 Pnl_Compass->AddChild(NewMarker);
51
52 UCanvasPanelSlot* MarkerSlot = Cast<UCanvasPanelSlot>(NewMarker->Slot);
53 if (MarkerSlot)
54 {
55 MarkerSlot->SetAnchors(FAnchors(0.5, 0.5, 0.5, 0.5));
56 MarkerSlot->SetSize(FVector2D(60.f, 60.f));
57 MarkerSlot->SetAlignment(FVector2D(0.5f, 0.5f));
58 MarkerSlot->SetPosition(FVector2D(0.f, 0.f));
59 }
60
61 Markers.Add(NewMarker);
62
63 return NewMarker;
64 }
65
66 return nullptr;

80void UCompassWidget::SetMarkerPosition(UImage* InMarker, float TargetRotation, bool bSideLock)
81{
82 float CompassSize = Img_Compass->GetDesiredSize().X;
83 float MarkerRotation = TargetRotation * (CompassSize/360.f);
84
85 UCanvasPanelSlot* MarkerSlot = Cast<UCanvasPanelSlot>(InMarker->Slot);
86 if (MarkerSlot)
87 {
88 if (!bSideLock)
89 {
90 MarkerSlot->SetPosition(FVector2D(MarkerRotation, 0.f));
91 }
92 else
93 {
94 UCanvasPanelSlot* CompassSlot = Cast<UCanvasPanelSlot>(Pnl_Compass->Slot);
95 if (CompassSlot)
96 {
97 float PositionX = (CompassSlot->GetSize().X / 2);
98
99 if (FMath::Abs(MarkerRotation) < PositionX)
100 {
101 MarkerSlot->SetPosition(FVector2D(MarkerRotation, 0.f));
102 }
103 else if (MarkerRotation > PositionX)
104 {
105 MarkerSlot->SetPosition(FVector2D(PositionX, 0.f));
106 }
107 else
108 {
109 MarkerSlot->SetPosition(FVector2D(-PositionX, 0.f));
110 }
111
112 }
113 }
114 }
