117void UHookSystem::TryHook()
118{
119 if (!OwnerPlayer || !OwnerPlayer->InteractionSystem)
120 {
121 PRINTLOG(TEXT("UHookSystem: OwnerPlayer or InteractionSystem is null!"));
122 return;
123 }
124
125 // 이미 Hook 중이면 무시
126 if (HookState != EHookState::Idle)
127 {
128 PRINTLOG(TEXT("UHookSystem: Already in hook state!"));
129 return;
130 }
131
132 // InteractionSystem의 LineTrace 사용
133 FHitResult HitResult;
134 if (!PerformCenterLineTrace(HitResult))
135 {
136 PRINTLOG(TEXT("UHookSystem: LineTrace failed!"));
137 return;
138 }
139
140 // Hit된 Actor가 UHookComponent를 가지고 있는지 확인
141 AActor* HitActor = HitResult.GetActor();
142 if (!HitActor)
143 {
144 PRINTLOG(TEXT("UHookSystem: No actor hit!"));
145 return;
146 }
147
148 UHookComponent* HookComp = HitActor->FindComponentByClass<UHookComponent>();
149 if (!HookComp || !HookComp->bIsHookable)
150 {
151 PRINTLOG(TEXT("UHookSystem: Target is not hookable!"));
152 return;
153 }
154
155 // Server에 Hook 요청
156 ServerTryHook(HitResult);
157}

171void UHookSystem::ServerTryHook_Implementation(const FHitResult& ClientHint)
172{
173 // 서버 권한 체크
174 if (!OwnerPlayer || !OwnerPlayer->HasAuthority())
175 {
176 PRINTLOG(TEXT("ServerTryHook: No authority"));
177 return;
178 }
179
180 // [개선 1] 서버에서 라인트레이스 재실행
181 FHitResult ServerHit;
182 if (!PerformServerLineTrace(ServerHit))
183 {
184 PRINTLOG(TEXT("ServerTryHook: Server-side line trace failed"));
185 return;
186 }
187
188 // [개선 2] 유효성 검증
189 AActor* HitActor = ServerHit.GetActor();
190 if (!HitActor)
191 {
192 PRINTLOG(TEXT("ServerTryHook: No actor hit"));
193 return;
194 }
195
196 UHookComponent* HookComp = HitActor->FindComponentByClass<UHookComponent>();
197 if (!HookComp || !HookComp->bIsHookable)
198 {
199 PRINTLOG(TEXT("ServerTryHook: Target is not hookable"));
200 return;
201 }
202
203 // [개선 3] 거리 검증
204 float Distance = FVector::Dist(OwnerPlayer->GetActorLocation(), HitActor->GetActorLocation());
205 if (Distance > MaxHookDistance)
206 {
207 PRINTLOG(TEXT("ServerTryHook: Target too far! Distance: %.2f, Max: %.2f"), Distance, MaxHookDistance);
208 return;
209 }
210
211 // [개선 4] Luggage인 경우 추가 검증
212 Aluggage* Luggage = Cast<Aluggage>(HitActor);
213 if (Luggage)
214 {
215 // 픽업 중인 Luggage는 훅 불가
216 if (Luggage->InteractableComp && Luggage->InteractableComp->IsPickedUp())
217 {
218 PRINTLOG(TEXT("ServerTryHook: Luggage is already picked up"));
219 return;
220 }
221
222 // [개선 5] 이미 다른 플레이어가 훅 중인지 확인
223 if (Luggage->bIsBeingHooked && Luggage->HookedBy != OwnerPlayer)
224 {
225 PRINTLOG(TEXT("ServerTryHook: Luggage is already hooked by %s"),
226 Luggage->HookedBy ? *Luggage->HookedBy->GetName() : TEXT("Unknown"));
227 return;
228 }
229
230 // 훅 플래그 설정 (복제됨)
231 Luggage->bIsBeingHooked = true;
232 Luggage->HookedBy = OwnerPlayer;
233
234 PRINTLOG(TEXT("ServerTryHook: %s set as hooked by %s"), *Luggage->GetName(), *OwnerPlayer->GetName());
235 }
236
237 // Hook 시작 (서버 HitResult 사용)
238 StartHook(ServerHit);
239}

241void UHookSystem::StartHook(const FHitResult& Hit)
242{
243 AActor* TargetActor = Hit.GetActor();
244 if (!TargetActor)
245 {
246 PRINTLOG(TEXT("UHookSystem: StartHook - Target actor is null!"));
247 return;
248 }
249
250 // 발사 상태로 전환
251 HookState = EHookState::Launching;
252 HookedTarget = TargetActor;
253
254 // 물리 객체인 경우 물리 및 충돌 끄기 (Hook 중에는 방해 없이 부드럽게 이동)
255 UPrimitiveComponent* RootPrimitive = Cast<UPrimitiveComponent>(HookedTarget->GetRootComponent());
256 if (RootPrimitive)
257 {
258 // 원래 충돌 상태 저장
259 OriginalCollisionEnabled = RootPrimitive->GetCollisionEnabled();
260
261 // 물리와 충돌 모두 끄기
262 if (RootPrimitive->IsSimulatingPhysics())
263 {
264 RootPrimitive->SetSimulatePhysics(false);
265 }
266 RootPrimitive->SetCollisionEnabled(ECollisionEnabled::NoCollision);
267
268 PRINTLOG(TEXT("UHookSystem: Physics and collision disabled for %s"), *HookedTarget->GetName());
269 }
270
271 // 발사 위치 및 방향 설정
272 HookProjectileLocation = OwnerPlayer->GetActorLocation();
273 HookLaunchDirection = (Hit.Location - HookProjectileLocation).GetSafeNormal();
274
275 // Cable 및 Projectile Mesh 표시
276 if (CableMesh)
277 {
278 CableMesh->SetVisibility(true);
279 }
280
281 if (ProjectileMesh)
282 {
283 ProjectileMesh->SetVisibility(true);
284 ProjectileMesh->SetWorldLocation(HookProjectileLocation);
285 }
286
287 PRINTLOG(TEXT("UHookSystem: Hook launched toward %s"), *HookedTarget->GetName());
288}

294void UHookSystem::ReleaseHook()
295{
296 // 클라이언트에서 호출 시 Server에 요청
297 if (OwnerPlayer && !OwnerPlayer->HasAuthority())
298 {
299 ServerReleaseHook();
300 return;
301 }
302
303 // 서버에서 실행
304 if (HookState != EHookState::Idle)
305 {
306 // [개선] Luggage 훅 플래그 해제
307 if (HookedTarget && IsValid(HookedTarget))
308 {
309 Aluggage* Luggage = Cast<Aluggage>(HookedTarget);
310 if (Luggage)
311 {
312 // 훅 플래그 해제 (복제됨)
313 Luggage->bIsBeingHooked = false;
314 Luggage->HookedBy = nullptr;
315
316 Luggage->PlayTTSAudio();
317 }
318
319 // 물리와 충돌 복원
320 UPrimitiveComponent* RootPrimitive = Cast<UPrimitiveComponent>(HookedTarget->GetRootComponent());
321 if (RootPrimitive)
322 {
323 // 속도를 초기화한 후 물리 켜기 (튕겨나가는 것 방지)
324 RootPrimitive->SetPhysicsLinearVelocity(FVector::ZeroVector, false);
325 RootPrimitive->SetPhysicsAngularVelocityInDegrees(FVector::ZeroVector, false);
326
327 // 충돌 복원
328 RootPrimitive->SetCollisionEnabled(OriginalCollisionEnabled);
329
330 // 물리 복원
331 RootPrimitive->SetSimulatePhysics(true);
332 }
333 }
334
335 HookState = EHookState::Idle;
336 HookedTarget = nullptr;
337
338 // Cable 및 Projectile Mesh 숨김
339 if (CableMesh)
340 CableMesh->SetVisibility(false);
341
342 if (ProjectileMesh)
343 ProjectileMesh->SetVisibility(false);
344 }

356void UHookSystem::DetectHookTarget()
357{
358 if (!OwnerPlayer || !OwnerPlayer->InteractionSystem)
359 return;
360
361 // LineTrace로 타겟 감지
362 FHitResult HitResult;
363 if (OwnerPlayer->InteractionSystem->PerformCenterLineTrace(HitResult))
364 {
365 AActor* HitActor = HitResult.GetActor();
366 if (HitActor)
367 {
368 UHookComponent* HookComp = HitActor->FindComponentByClass<UHookComponent>();
369 if (HookComp && HookComp->bIsHookable)
370 {
371 // [개선] 타겟이 바뀌면 이전 타겟의 Outline 끄기
372 if (CurHookTarget_Luggage && CurHookTarget != HitActor)
373 {
374 CurHookTarget_Luggage->OutlineOff();
375 PRINTLOG(TEXT("DetectHookTarget: Previous target %s OutlineOff"), *CurHookTarget_Luggage->GetName());
376 }
377
378 CurHookTarget = HitActor;
379 CurHookTarget_Luggage = Cast<Aluggage>(HitActor);
380 return;
381 }
382 }
383 }
384
385 // [개선] 타겟을 잃었을 때 이전 타겟의 Outline 끄기
386 if (CurHookTarget_Luggage)
387 {
388 CurHookTarget_Luggage->OutlineOff();
389 PRINTLOG(TEXT("DetectHookTarget: Lost target %s, OutlineOff"), *CurHookTarget_Luggage->GetName());
390 }
391
392 CurHookTarget = nullptr;
393 CurHookTarget_Luggage = nullptr;
394}

397bool UHookSystem::PerformCenterLineTrace(FHitResult& OutHit)
398{
399 if (!OwnerPlayer)
400 return false;
401
402 // 플레이어 컨트롤러 가져오기
403 APlayerController* PC = Cast<APlayerController>(OwnerPlayer->GetController());
404 if (!PC)
405 return false;
406
407 APlayerCameraManager* CameraManager = PC->PlayerCameraManager;
408 if (!CameraManager)
409 return false;
410
411 FVector CameraLocation = CameraManager->GetCameraLocation();
412 FVector CameraForward = CameraManager->GetCameraRotation().Vector();
413
414 // 레이 트레이스 시작/끝 지점
415 FVector TraceStart = CameraLocation;
416 FVector TraceEnd = TraceStart + (CameraForward * InteractionDistance);
417
418 // Ray trace 실행
419 bool bHit = GetWorld()->LineTraceSingleByChannel(
420 OutHit, TraceStart, TraceEnd, ECC_Visibility
421 );
422
423 // 디버그 라인
424 if (bShowDebugInfo)
425 {
426 DrawDebugLine(
427 GetWorld(), TraceStart, TraceEnd,
428 bHit ? FColor::Green : FColor::Red,
429 false, 0.0f, 0, 0.5f
430 );
431 }
432
433 return bHit;
434}

462
463 FVector TraceStart = CameraLocation;
464 FVector TraceEnd = TraceStart + (CameraRotation.Vector() * InteractionDistance);
465
466 // LineTrace 실행
467 bool bHit = GetWorld()->LineTraceSingleByChannel(
468 OutHit, TraceStart, TraceEnd, ECC_Visibility
469 );
470
471 // 디버그 라인 (서버)
472 if (bShowDebugInfo)
473 {
474 DrawDebugLine(
475 GetWorld(), TraceStart, TraceEnd,

478 );

481 {
482 DrawDebugSphere(
483 GetWorld(), OutHit.Location,

485 FColor::Yellow,

487 );
488 }
489 }
490
491 if (bHit)
492 {
493 PRINTLOG(TEXT("PerformServerLineTrace: Hit %s at distance %.2f"),
494 *OutHit.GetActor()->GetName(),
495 FVector::Dist(CameraLocation, OutHit.Location));
496 }
497
498 return bHit;

546void UHookSystem::UpdateHookLaunching(float DeltaTime)
547{
548 if (!HookedTarget || !IsValid(HookedTarget))
549 {
550 PRINTLOG(TEXT("UHookSystem: Target lost during launch"));
551 ReleaseHook();
552 return;
553 }
554
555 // 서버 권한 체크 - 상태 변경은 서버에서만
556 if (!OwnerPlayer || !OwnerPlayer->HasAuthority())
557 {
558 return;
559 }
560
561 // 훅 발사체 이동
562 HookProjectileLocation += HookLaunchDirection * LaunchSpeed * DeltaTime;
563
564 // 타겟 위치
565 FVector TargetLocation = HookedTarget->GetActorLocation();
566
567 // 타겟에 도달했는지 확인
568 float DistanceToTarget = FVector::Dist(HookProjectileLocation, TargetLocation);
569
570 if (DistanceToTarget < 50.0f)
571 {
572 // 타겟에 도달 - Pulling 상태로 전환
573 HookState = EHookState::Pulling;
574 PRINTLOG(TEXT("UHookSystem: Hook reached target, start pulling"));
575 }
576 else
577 {
578 // 최대 거리 체크
579 float DistanceFromPlayer = FVector::Dist(HookProjectileLocation, OwnerPlayer->GetActorLocation());
580 if (DistanceFromPlayer > MaxHookDistance)
581 {
582 PRINTLOG(TEXT("UHookSystem: Hook exceeded max distance"));
583 ReleaseHook();
584 }
585 }
586
587 // 디버그 표시
588 if (bShowDebugInfo)
589 {
590 DrawDebugSphere(
591 GetWorld(),
592 HookProjectileLocation,
593 20.0f,
594 8,
595 FColor::Orange,
596 false,
597 0.0f,
598 0,
599 2.0f
600 );
601 }

604void UHookSystem::UpdateHookPulling(float DeltaTime)
605{
606 // 안전 체크: 대상이 유효하지 않으면 자동 해제
607 if (!HookedTarget || !IsValid(HookedTarget))
608 {
609 PRINTLOG(TEXT("UHookSystem: Hooked target became invalid, releasing hook"));
610 ReleaseHook();
611 return;
612 }
613
614 if (!OwnerPlayer)
615 {
616 ReleaseHook();
617 return;
618 }
619
620 // 플레이어 앞 목표 위치 계산
621 FVector PlayerLocation = OwnerPlayer->GetActorLocation();
622 FVector PlayerForward = OwnerPlayer->GetActorForwardVector();
623 FVector TargetLocation = PlayerLocation + (PlayerForward * DesiredDistance);
624
625 // 현재 대상 위치
626 FVector CurrentLocation = HookedTarget->GetActorLocation();
627
628 // 목표 위치의 높이를 현재 Luggage 높이로 유지 (바닥에 떨어지도록)
629 TargetLocation.Z = CurrentLocation.Z;
630
631 // 서버에서만 실제 위치 업데이트 및 완료 체크
632 if (OwnerPlayer->HasAuthority())
633 {
634 // 목표 위치까지의 거리 확인
635 float DistanceToTarget = FVector::Dist(CurrentLocation, TargetLocation);
636
637 // 너무 가까우면 Hook 완료
638 if (DistanceToTarget < CompleteThreshold)
639 {
640 // 완료 전 정확한 목표 위치에 배치 (튕겨나가는 것 방지)
641 HookedTarget->SetActorLocation(TargetLocation);
642 PRINTLOG(TEXT("UHookSystem: Hook completed, placed at exact target location"));
643 ReleaseHook();
644 return;
645 }
646
647 // 부드럽게 위치 보간 (물리는 이미 꺼진 상태)
648 FVector NewLocation = FMath::VInterpTo(
649 CurrentLocation,
650 TargetLocation,
651 DeltaTime,
652 HookSpeed
653 );
654
655 // 위치 업데이트 (서버에서만 실행됨)
656 HookedTarget->SetActorLocation(NewLocation);
657 }
658
659 // 디버그 표시
660 if (bShowDebugInfo)
661 {
662 DrawDebugLine(
663 GetWorld(),
664 CurrentLocation,
665 TargetLocation,
666 FColor::Green,
667 false,
668 0.0f,
669 0,
670 2.0f
671 );
672
673 DrawDebugSphere(
674 GetWorld(),
675 TargetLocation,
676 50.0f,
677 12,
678 FColor::Green,
679 false,
680 0.0f,
681 0,
682 1.0f
683 );
684 }

755void UHookSystem::UpdateCableMeshTransform(const FVector& CableStart, const FVector& CableEnd)
756{
757 if (!CableMesh)
758 {
759 return;
760 }
761
762 FVector CableDelta = CableEnd - CableStart;
763 float CableLength = CableDelta.Size();
764 if (CableLength <= KINDA_SMALL_NUMBER)
765 {
766 CableMesh->SetVisibility(false);
767 return;
768 }
769
770 FVector CableMidpoint = CableStart + (CableDelta * 0.5f);
771 CableMesh->SetWorldLocation(CableMidpoint);
772
773 FVector Direction = CableDelta / CableLength;
774 FRotator CableRotation = FRotationMatrix::MakeFromZ(Direction).Rotator();
775 CableMesh->SetWorldRotation(CableRotation);
776
777 const float LengthScale = CableLength / FMath::Max(CableMeshBaseLength, KINDA_SMALL_NUMBER);
778 const float TargetThickness = FMath::Max(CableThickness, 1.0f);
779 const float MeshDiameter = FMath::Max(CableMeshBaseRadius * 2.0f, KINDA_SMALL_NUMBER);
780 const float RadiusScale = TargetThickness / MeshDiameter;
781
782 CableMesh->SetWorldScale3D(FVector(RadiusScale, RadiusScale, LengthScale));
783 CableMesh->SetVisibility(true);
784}
