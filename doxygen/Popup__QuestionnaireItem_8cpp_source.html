48void UPopup_QuestionnaireItem::OnClickButton()
49{
50 if (auto Popup = UPopupManager::ShowPopupAs<UPopup_WriteBoard>(GetWorld(), EPopupType::WriteBoard))
51 {
52 // TODO: 단어 수 및 글자 수 구하기
53 TArray<FString> Tokens;
54 QuestionData.answer_kor.ParseIntoArrayWS(Tokens);
55 for (const auto& token : Tokens)
56 {
57 PRINT_STRING(TEXT("Question 글 : %s"), *token);
58 }
59 PRINT_STRING(TEXT("Question 단어 수 : %d"), Tokens.Num());
60 PRINT_STRING(TEXT("Question 글자 수 : %d"), QuestionData.answer_kor.Len());
61 PRINT_STRING(TEXT("Question 첫 단어 글자 수 : %d"), Tokens[0].Len());
62
63 Popup->InitPopup(QuestionData.Id, QuestionData);
64 }
65}
