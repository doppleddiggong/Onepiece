58void USpeakWidget::UpdateSpeakStageUI(ASpeakStageActor* SpeakStage, APlayerState* LocalPlayerState)
59{
60 if (!SpeakStage)
61 {
62 PRINTLOG(TEXT("[SpeakWidget] UpdateSpeakStageUI - SpeakStage is null"));
63 return;
64 }
65
66 // 캐싱
67 CachedSpeakStage = SpeakStage;
68 CachedLocalPlayerState = LocalPlayerState;
69
70 // 1. 현재 발화자 표시
71 if (CurrentSpeakerText)
72 {
73 APlayerState* CurrentSpeaker = SpeakStage->GetCurrentSpeaker();
74 if (CurrentSpeaker)
75 {
76 FString SpeakerName = CurrentSpeaker->GetPlayerName();
77
78 // 내 턴인지 확인
79 if (LocalPlayerState && CurrentSpeaker == LocalPlayerState)
80 {
81 CurrentSpeakerText->SetText(FText::FromString(
82 FString::Printf(TEXT("Your Turn: %s"), *SpeakerName)
83 ));
84 }
85 else
86 {
87 CurrentSpeakerText->SetText(FText::FromString(
88 FString::Printf(TEXT("Current Speaker: %s"), *SpeakerName)
89 ));
90 }
91 }
92 else
93 {
94 CurrentSpeakerText->SetText(FText::FromString(TEXT("Waiting for players...")));
95 }
96 }
97
98 // 2. 현재 질문 표시
99 if (CurrentQuestionText)
100 {
101 FString Question = SpeakStage->GetCurrentQuestion();
102 if (!Question.IsEmpty())
103 {
104 CurrentQuestionText->SetText(FText::FromString(Question));
105 }
106 else
107 {
108 CurrentQuestionText->SetText(FText::FromString(TEXT("No question available")));
109 }
110 }
111
112 // 3. 진행률 표시 (예: 3/5)
113 if (ProgressText)
114 {
115 int32 CurrentStep = SpeakStage->GetCurrentStepIndex();
116 int32 TotalSteps = SpeakStage->GetTotalQuestions();
117
118 // 질문이 있으면 1부터 시작 (사용자 친화적)
119 if (TotalSteps > 0)
120 {
121 ProgressText->SetText(FText::FromString(
122 FString::Printf(TEXT("Progress: %d/%d"), CurrentStep + 1, TotalSteps)
123 ));
124 }
125 else
126 {
127 ProgressText->SetText(FText::FromString(TEXT("Progress: 0/0")));
128 }
129 }
130
131 // 4. 대기 순서 표시 (선택사항)
132 if (QueuePositionText && LocalPlayerState)
133 {
134 APlayerState* CurrentSpeaker = SpeakStage->GetCurrentSpeaker();
135
136 // 내 턴이 아니면 대기 메시지 표시
137 if (CurrentSpeaker && CurrentSpeaker != LocalPlayerState)
138 {
139 QueuePositionText->SetText(FText::FromString(TEXT("Waiting for your turn...")));
140 }
141 else if (CurrentSpeaker && CurrentSpeaker == LocalPlayerState)
142 {
143 QueuePositionText->SetText(FText::FromString(TEXT("It's your turn! Speak now.")));
144 }
145 else
146 {
147 QueuePositionText->SetText(FText::FromString(TEXT("")));
148 }
149 }
150}
