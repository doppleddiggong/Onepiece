55FReply UPopup_WriteBoard::NativeOnMouseButtonDown(const FGeometry& InGeometry, const FPointerEvent& InMouseEvent)
56{
57 // Check Mouse is in Canvas
58 const FGeometry CanvasGeometry = Image_Canvas->GetCachedGeometry();
59 if (!CanvasGeometry.IsUnderLocation(InMouseEvent.GetScreenSpacePosition()))
60 {
61 return FReply::Unhandled();
62 }
63
64 // Get Mouse Position in Local Image Coordinate System
65 // Save Current MousePos to prevMousePos
66 prevMousePos = GetLocalMousePos(InMouseEvent.GetScreenSpacePosition());
67 bIsDrawing = true;
68
69 // Draw Point Once
70 // PRINT_STRING(TEXT("%s"), *InMouseEvent.GetEffectingButton().GetFName().ToString());
71 if (InMouseEvent.IsMouseButtonDown(EKeys::LeftMouseButton)) // Draw
72 {
73 DrawPoint(GetLocalMousePos(InMouseEvent.GetScreenSpacePosition()), FLinearColor::Black);
74 }
75 else if (InMouseEvent.IsMouseButtonDown(EKeys::RightMouseButton)) // Erase
76 {
77 DrawPoint(GetLocalMousePos(InMouseEvent.GetScreenSpacePosition()), FLinearColor::White);
78 }
79
80 return FReply::Handled();
81}

178void UPopup_WriteBoard::DrawPoint(FVector2D mousePos, FLinearColor drawColor)
179{
180 // Begin Draw Canvas To Render Target
181 UCanvas* canvas = nullptr;
182 FVector2D size;
183 FDrawToRenderTargetContext context;
184 UKismetRenderingLibrary::BeginDrawCanvasToRenderTarget(this, RT_Canvas, canvas, size, context);
185
186 // Set thickness Whether now in Draw or Erase
187 float thickness = (drawColor == FLinearColor::Black) ? 10 : 30;
188 // Draw Box
189 canvas->K2_DrawBox(mousePos, FVector2D(1, 1), thickness, drawColor);
190
191 UKismetRenderingLibrary::EndDrawCanvasToRenderTarget(this, context);

194void UPopup_WriteBoard::DrawLines(FVector2D mousePos, FLinearColor drawColor)
195{
196 // Begin Draw Canvas To Render Target
197 UCanvas* canvas = nullptr;
198 FVector2D size;
199 FDrawToRenderTargetContext context;
200 UKismetRenderingLibrary::BeginDrawCanvasToRenderTarget(this, RT_Canvas, canvas, size, context);
201
202 // Calculate Draw Positions
203 FVector2D currPos = prevMousePos;
204 int32 div = 64;
205 FVector2D drawOffset = (mousePos - currPos) / div;
206
207 // Draw
208 for (int32 i = 1; i < div; ++i)
209 {
210 currPos = prevMousePos + drawOffset * i;
211
212 // Set thickness Whether now in Draw or Erase
213 float thickness = (drawColor == FLinearColor::Black) ? 12 : 30;
214 // Draw Line
215 canvas->K2_DrawLine(prevMousePos, currPos, thickness, drawColor);
216 }
217 prevMousePos = mousePos;
218
219 UKismetRenderingLibrary::EndDrawCanvasToRenderTarget(this, context);
220}

222FVector2D UPopup_WriteBoard::GetLocalMousePos(FVector2D mousePos)
223{
224 // Get Absolute Local Pos
225 const FGeometry& geometry = Image_Canvas->GetCachedGeometry();
226 FVector2D localPos = geometry.AbsoluteToLocal(mousePos);
227
228 // Get Canvas Size
229 const FVector2D canvasSize = geometry.GetLocalSize();
230 // Transform localPos(in Image_Canvas Coord) to RT_Canvas Coord && Clamp upto RT_Canvas' Border
231 localPos.X = FMath::Clamp((localPos.X / canvasSize.X * RT_Canvas->SizeX), 0.f, RT_Canvas->SizeX);
232 localPos.Y = FMath::Clamp((localPos.Y / canvasSize.Y * RT_Canvas->SizeY), 0.f, RT_Canvas->SizeY);
233 return localPos;
234}

259bool UPopup_WriteBoard::SaveRenderTargetToPNG(UTextureRenderTarget2D* RenderTarget, const FString& FullFilePath)
260{
261 FTextureRenderTargetResource* RTResource = RenderTarget->GameThread_GetRenderTargetResource();
262 if (!RTResource)
263 {
264 return false;
265 }
266
267 // Get Width & Height
268 const int32 Width = RenderTarget->SizeX;
269 const int32 Height = RenderTarget->SizeY;
270
271 // Set Bitmap array
272 TArray<FColor> Bitmap;
273 Bitmap.AddUninitialized(Width * Height);
274
275 // Read BGRA8 pixels in RenderTarget
276 RTResource->ReadPixels(Bitmap);
277
278 // PNG Encoder
279 IImageWrapperModule& ImageWrapperModule =
280 FModuleManager::LoadModuleChecked<IImageWrapperModule>("ImageWrapper");
281
282 TSharedPtr<IImageWrapper> ImageWrapper =
283 ImageWrapperModule.CreateImageWrapper(EImageFormat::PNG);
284
285 if (!ImageWrapper.IsValid())
286 {
287 return false;
288 }
289
290 ImageWrapper->SetRaw(
291 Bitmap.GetData(),
292 Bitmap.GetAllocatedSize(),
293 Width,
294 Height,
295 ERGBFormat::BGRA,
296 8
297 );
298
299 // Compress to PNG
300 const TArray64<uint8>& PNGData = ImageWrapper->GetCompressed(100);
301
302 // Make Directory
303 const FString Directory = FPaths::GetPath(FullFilePath);
304 IFileManager::Get().MakeDirectory(*Directory, true);
305
306 return FFileHelper::SaveArrayToFile(PNGData, *FullFilePath);
307}
