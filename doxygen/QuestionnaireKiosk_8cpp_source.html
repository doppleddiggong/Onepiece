120void AQuestionnaireKiosk::OnResponseData(FQuestWriteInfo& InResponseData, bool bWasSuccessful)
121{
122 if (bWasSuccessful)
123 {
124 PRINTLOG(TEXT("--- Write Question Response SUCCESS ---"));
125
126 // 쓰기 퀘스트 json 데이터 받기 요청
127 QuestionnaireData = InResponseData;
128
129 for (int32 i = 1; i <= QuestionnaireData.question.Num(); ++i)
130 {
131 FWriteQuestionData& data = QuestionnaireData.question[i - 1];
132 data.Id = i;
133 PRINTLOG(TEXT("%d kor: %s"), i, *data.word_data.kor);
134 PRINTLOG(TEXT("%d answer_kor: %s"), i, *data.answer_kor);
135 }
136
137 // 테스트용 더미 데이터 생성
138 // dCreateTestData(QuestionnaireData);
139
140 // 모든 플레이어의 WriteQuest 진행 상태 설정
141 if (ALingoGameState* GS = GetWorld()->GetGameState<ALingoGameState>())
142 {
143 for (APlayerState* PS : GS->PlayerArray)
144 {
145 if (ALingoPlayerState* LingoPS = Cast<ALingoPlayerState>(PS))
146 {
147 LingoPS->SetWriteQuestIng(true);
148 }
149 }
150 }
151
152 ShowPopup();
153 }
154 else
155 {
156 PRINTLOG(TEXT("--- Write Question Response FAILED ---"));
157 }
158}

217void AQuestionnaireKiosk::BillboardInteractWidget()
218{
219 // 위젯이 없으면 빌보드화 안 함
220 if (!WidgetGuideComp)
221 return;
222
223 // Visibility 체크 - 보이지 않으면 빌보드화 안 함
224 if (!WidgetGuideComp->IsVisible())
225 return;
226
227 // 카메라 가져오기
228 AActor* Camera = UGameplayStatics::GetPlayerCameraManager(GetWorld(), 0);
229 if (!Camera)
230 return;
231
232 // 카메라를 향하도록 회전 계산
233 FRotator Rotation = UKismetMathLibrary::MakeRotFromXZ( -Camera->GetActorForwardVector(), Camera->GetActorUpVector() );
234 Rotation.Pitch = 0;
235
236 // 위젯 회전 설정
237 WidgetGuideComp->SetWorldRotation(Rotation);
238}
