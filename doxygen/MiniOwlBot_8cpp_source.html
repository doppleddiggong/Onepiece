55void AMiniOwlBot::UpdateLocation()// float DeltaTime

66 FVector rotatedDir = radius * currTarget->GetActorForwardVector().RotateAngleAxis(angle, FVector::UpVector);
67 valX += rotatedDir.X;
68 valY += rotatedDir.Y;
69 }
70
71 SetActorLocation(FVector(valX, valY, valZ));

75bool AMiniOwlBot::CheckAngleOutofCamera()
76{
77 // target의 카메라 방향 벡터
78 FVector2D targetCameraDir(currTarget->GetCameraForwardVector().X, currTarget->GetCameraForwardVector().Y);
79 targetCameraDir.Normalize();
80
81 // 자신의 방향벡터
82 FVector2D botDir(GetActorForwardVector().X, GetActorForwardVector().Y);
83 botDir.Normalize();
84
85 // 각도 구하기
86 float dotP = FVector2D::DotProduct(targetCameraDir, botDir);
87 PRINTLOG(TEXT("dotP angle : %f, %f"), FMath::RadiansToDegrees(FMath::Acos(dotP)), angle);
88 return FMath::RadiansToDegrees(FMath::Acos(dotP)) <= angle;
89}
