322void APlayerControl::RequestDrop(APlayerControl* Requester)
323{
324 if (auto World = GetWorld())
325 {
326 for (TActorIterator<ADropper> It(World); It; ++It)
327 {
328 ADropper* Dropper = *It;
329 if (Dropper)
330 {
331 if (Dropper->IsBusy() )
332 {
333 Requester->Client_ToastMessage( TEXT("Dropper is Busy") );
334 return;
335 }
336
337 FLuggageData tmpData;
338 tmpData.word1 = tmpData.word1.GetRandomAnimal();
339 tmpData.word2 = tmpData.word2.GetRandomColor();
340
341 Dropper->SetSpawnData(tmpData);
342 Dropper->SetSpawnClass( LoadClass<AActor>(nullptr, TEXT("/Game/CustomContents/Blueprints/Interactables/BP_Luggage.BP_Luggage_C")));
343 Dropper->RequestSpawn();
344 return;
345 }
346 }
347 }

403void APlayerControl::Client_RequestSpeakScenario_Implementation(AWheatly* Wheatly)
404{
405 if (!Wheatly)
406 {
407 PRINTLOG(TEXT("[APlayerControl] Client_RequestSpeakScenario: Wheatly is null"));
408 return;
409 }
410
411 // 현재 조종중인 PlayerActor 획득
412 APlayerActor* PlayerActor = Cast<APlayerActor>(GetPawn());
413 if (!PlayerActor)
414 {
415 PRINTLOG(TEXT("[APlayerControl] Client_RequestSpeakScenario: PlayerActor is null"));
416 return;
417 }
418
419 // Client에서 네트워크 요청 수행
420 if (auto KLingoNetwork = UKLingoNetworkSystem::Get(GetWorld()))
421 {
422 // Lambda를 사용하여 응답 처리
423 // this 캡처: APlayerControl의 Server RPC 호출을 위해
424 KLingoNetwork->RequestSpeakScenario( FResponseSpeakScenarioDelegate::CreateLambda(
425 [this, Wheatly](FResponseSpeakScenario& ResponseData, bool bWasSuccessful)
426 {
427 if (bWasSuccessful && Wheatly)
428 {
429 // 성공 시 자신의 Server RPC 호출 (PlayerControl은 Client 소유!)
430 Server_SyncSpeakScenarioData(Wheatly, ResponseData);
431 PRINTLOG(TEXT("[APlayerControl] Client successfully received scenario data, syncing to server"));
432 }
433 else
434 {
435 // 실패 시 에러는 이미 ShowNetworkErrorPopup으로 표시됨
436 PRINTLOG(TEXT("[APlayerControl] Client failed to receive scenario data"));
437 }
438 }
439 )
440 );
441 }
442 else
443 {
444 PRINTLOG(TEXT("[APlayerControl] Client_RequestSpeakScenario: Failed to get KLingoNetworkSystem"));
445 }
446}

640void APlayerControl::OnDoorMessage(int32 InDoorIndex, bool bInOpen)
641{
642 // 서버에서만 실행
643 if (!HasAuthority())
644 return;
645
646 // 문이 열릴 때만 처리
647 if (!bInOpen)
648 return;
649
650 // PlayerState 가져오기
651 ALingoPlayerState* PS = GetPlayerState<ALingoPlayerState>();
652 if (!PS)
653 return;
654
655 // End DoorIndex에 따라 퀘스트 완료 처리
656 if (InDoorIndex == DoorGroup::Step1_End)
657 {
658 PS->SetReadQuestCompleted();
659 }
660 else if (InDoorIndex == DoorGroup::Step2_End)
661 {
662 PS->SetListenQuestCompleted();
663 }
664 else if (InDoorIndex == DoorGroup::Step3_End)
665 {
666 PS->SetSpeakQuestCompleted();
667 }
668 else if (InDoorIndex == DoorGroup::Step4_End)
669 {
670 PS->SetWriteQuestCompleted();
671 }
