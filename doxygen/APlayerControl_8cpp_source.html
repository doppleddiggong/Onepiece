415void APlayerControl::RequestDrop(APlayerControl* Requester)
416{
417 if (auto World = GetWorld())
418 {
419 for (TActorIterator<ADropper> It(World); It; ++It)
420 {
421 ADropper* Dropper = *It;
422 if (Dropper)
423 {
424 if (Dropper->IsBusy() )
425 {
426 Requester->Client_ToastMessage( TEXT("Dropper is Busy") );
427 return;
428 }
429
430 FLuggageData tmpData;
431 tmpData.word1 = tmpData.word1.GetRandomAnimal();
432 tmpData.word2 = tmpData.word2.GetRandomColor();
433
434 Dropper->SetSpawnData(tmpData);
435 Dropper->SetSpawnClass( LoadClass<AActor>(nullptr, TEXT("/Game/CustomContents/Blueprints/Interactables/BP_Luggage.BP_Luggage_C")));
436 Dropper->RequestSpawn();
437 return;
438 }
439 }
440 }

516void APlayerControl::Client_RequestSpeakScenario_Implementation(AWheatly* Wheatly)
517{
518 if (!Wheatly)
519 {
520 PRINTLOG(TEXT("[APlayerControl] Client_RequestSpeakScenario: Wheatly is null"));
521 return;
522 }
523
524 // 현재 조종중인 PlayerActor 획득
525 APlayerActor* PlayerActor = Cast<APlayerActor>(GetPawn());
526 if (!PlayerActor)
527 {
528 PRINTLOG(TEXT("[APlayerControl] Client_RequestSpeakScenario: PlayerActor is null"));
529 return;
530 }
531
532 // Client에서 네트워크 요청 수행
533 if (auto KLingoNetwork = UKLingoNetworkSystem::Get(GetWorld()))
534 {
535 // Lambda를 사용하여 응답 처리
536 // this 캡처: APlayerControl의 Server RPC 호출을 위해
537 KLingoNetwork->RequestSpeakScenario( FResponseSpeakScenarioDelegate::CreateLambda(
538 [this, Wheatly](FResponseSpeakScenario& ResponseData, bool bWasSuccessful)
539 {
540 if (bWasSuccessful && Wheatly)
541 {
542 // 성공 시 자신의 Server RPC 호출 (PlayerControl은 Client 소유!)
543 Server_SyncSpeakScenarioData(Wheatly, ResponseData);
544 PRINTLOG(TEXT("[APlayerControl] Client successfully received scenario data, syncing to server"));
545 }
546 else
547 {
548 // 실패 시 에러는 이미 ShowNetworkErrorPopup으로 표시됨
549 PRINTLOG(TEXT("[APlayerControl] Client failed to receive scenario data"));
550 }
551 }
552 )
553 );
554 }
555 else
556 {
557 PRINTLOG(TEXT("[APlayerControl] Client_RequestSpeakScenario: Failed to get KLingoNetworkSystem"));
558 }
559}

774void APlayerControl::OnDoorMessage(int32 InDoorIndex, bool bInOpen, AActor* EventInstigator)
775{
776 // 서버에서만 실행
777 if (!HasAuthority())
778 return;
779
780 // 문이 열릴 때만 처리
781 if (!bInOpen)
782 return;
783
784 // PlayerState 가져오기
785 ALingoPlayerState* PS = GetPlayerState<ALingoPlayerState>();
786 if (!PS)
787 return;
788
789 // End DoorIndex에 따라 퀘스트 완료 처리
790 if (InDoorIndex == DoorGroup::Step1_End)
791 {
792 // ReadQuest: 협력 퀘스트, 모든 플레이어 완료
793 PS->SetReadQuestCompleted();
794 }
795 else if (InDoorIndex == DoorGroup::Step2_End)
796 {
797 // ListenQuest: 협력 퀘스트, 모든 플레이어 완료
798 PS->SetListenQuestCompleted();
799 }
800 else if (InDoorIndex == DoorGroup::Step3_End)
801 {
802 // SpeakQuest: 개인 퀘스트, EventInstigator 확인
803 if (EventInstigator != nullptr && EventInstigator != GetPawn())
804 return;
805
806 PS->SetSpeakQuestCompleted();
807 }
808 else if (InDoorIndex == DoorGroup::Step4_End)
809 {
810 // WriteQuest: 개인 퀘스트, EventInstigator 확인
811 if (EventInstigator != nullptr && EventInstigator != GetPawn())
812 return;
813
814 PS->SetWriteQuestCompleted();
815 }
816
817 if (InDoorIndex == DoorGroup::Step1_Tutorial)
818 {
819 Multicast_OpenHowToPlay(EQuestType::Read);
820 }
821 else if (InDoorIndex == DoorGroup::Step2_Tutorial)
822 {
823 Multicast_OpenHowToPlay(EQuestType::Listen);
824 }
825 else if (InDoorIndex == DoorGroup::Step3_Tutorial)
826 {
827 // 개인 퀘스트: Instigator가 본인인지 확인 후 해당 클라이언트에게만 알림
828 if (EventInstigator == GetPawn())
829 {
830 Client_OpenHowToPlay(EQuestType::Speak);
831 }
832 }
833 else if (InDoorIndex == DoorGroup::Step3_Tutorial)
834 {
835 // 개인 퀘스트: Instigator가 본인인지 확인 후 해당 클라이언트에게만 알림
836 if (EventInstigator == GetPawn())
837 {
838 Client_OpenHowToPlay(EQuestType::Write);
839 }
840 }
841}
