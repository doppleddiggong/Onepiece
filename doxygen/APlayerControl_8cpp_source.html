54void APlayerControl::BeginPlay()
55{
56 Super::BeginPlay();
57
58 if (auto LP = GetLocalPlayer())
59 {
60 if (auto SubSystem = LP->GetSubsystem<UEnhancedInputLocalPlayerSubsystem>())
61 {
62 if (IMC_Default)
63 {
64 SubSystem->ClearAllMappings();
65 SubSystem->AddMappingContext(IMC_Default, 0);
66 }
67 }
68 }
69 if (IsLocalController())
70 {
71 UserInfo = ULingoGameInstanceSubsystem::Get(GetWorld())->GetUserInfo();
72
73 Server_SetUserInfo(UserInfo);
74 }
75}

109IControllable* APlayerControl::GetControllable() const
110{
111 APawn* P = GetPawn();
112 if (!P)
113 return nullptr;
114
115 // UObject 기반 UInterface 라면 Cast 가능
116 if (IControllable* C = Cast<IControllable>(P))
117 return C;
118
119 return nullptr;
120}
