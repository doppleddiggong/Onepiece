51TArray<ALingoPlayerState*> ULingoGameHelper::GetLingoPlayerStateList(const UObject* WorldContextObject)
52{
53 TArray<ALingoPlayerState*> PlayerStateList;
54 for (FConstPlayerControllerIterator It = WorldContextObject->GetWorld()->GetPlayerControllerIterator(); It; ++It)
55 {
56 if (APlayerController* PC = It->Get())
57 {
58 if (ALingoPlayerState* PS = PC->GetPlayerState<ALingoPlayerState>())
59 {
60 PlayerStateList.Add(PS);
61 }
62 }
63 }
64
65 return PlayerStateList;
