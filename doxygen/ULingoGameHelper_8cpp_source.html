80TArray<ALingoPlayerState*> ULingoGameHelper::GetLingoPlayerStateList(const UObject* WorldContextObject)
81{
82 TArray<ALingoPlayerState*> PlayerStateList;
83 for (FConstPlayerControllerIterator It = WorldContextObject->GetWorld()->GetPlayerControllerIterator(); It; ++It)
84 {
85 if (APlayerController* PC = It->Get())
86 {
87 if (ALingoPlayerState* PS = PC->GetPlayerState<ALingoPlayerState>())
88 {
89 PlayerStateList.Add(PS);
90 }
91 }
92 }
93
94 return PlayerStateList;
95}

158EResourceTextureType ULingoGameHelper::ConvertGradeScore(const float Score)
159{
160 if (Score >= 90.0f)
161 return EResourceTextureType::Rarity_S;
162 else if (Score >= 80.0f)
163 return EResourceTextureType::Rarity_A;
164 else if (Score >= 70.0f)
165 return EResourceTextureType::Rarity_B;
166 else if (Score >= 60.0f)
167 return EResourceTextureType::Rarity_C;
168 else
169 return EResourceTextureType::Rarity_D;
170}

172EResourceTextureType ULingoGameHelper::ConvertGradeString(const FString& Grade)
173{
174 if (Grade == "D") return EResourceTextureType::Rarity_D;
175 if (Grade == "C") return EResourceTextureType::Rarity_C;
176 if (Grade == "B") return EResourceTextureType::Rarity_B;
177 if (Grade == "A") return EResourceTextureType::Rarity_A;
178 if (Grade == "S") return EResourceTextureType::Rarity_S;
179
180 return EResourceTextureType::Rarity_D;
181}

183FString ULingoGameHelper::GetFormatTimer(const float InRemainTime )
184{
185 const int32 Minutes = FMath::FloorToInt(InRemainTime / 60.f);
186 const int32 Seconds = FMath::FloorToInt(InRemainTime) % 60;

266FString ULingoGameHelper::GetAccuracyPercentage(int WrongCnt)
267{
268 // 정답률 계산
269 const float Percentage = ((10.f - WrongCnt) / 10.f) * 100.f;
270 const int32 RoundedPercentage = FMath::RoundToInt(Percentage);
271
272 // FString으로 변환
273 return FString::Printf(TEXT("%d%%"), RoundedPercentage);
