17AContactTrigger::AContactTrigger()
18{
19 PrimaryActorTick.bCanEverTick = true;
20 bReplicates = true;
21
22 // 루트 컴포넌트로 BoxComponent 생성
23 TriggerBox = CreateDefaultSubobject<UBoxComponent>(TEXT("TriggerBox"));
24 RootComponent = TriggerBox;
25
26 // 박스 크기 기본값 설정
27 TriggerBox->SetBoxExtent(FVector(100.0f, 100.0f, 100.0f));
28
29 // Overlap 이벤트 활성화
30 TriggerBox->SetGenerateOverlapEvents(true);
31 TriggerBox->SetCollisionEnabled(ECollisionEnabled::QueryOnly);
32 TriggerBox->SetCollisionResponseToAllChannels(ECR_Ignore);
33 TriggerBox->SetCollisionResponseToChannel(ECC_Pawn, ECR_Overlap);
34
35 // 초기값 설정
36 bIsTriggered = false;
37 bShowDebugBox = true;
38 DebugBoxColor = FColor::Green;
39
40 VoiceConversationSystem = CreateDefaultSubobject<UVoiceConversationSystem>(TEXT("VoiceConversationSystem"));
41
42 QuestType = EQuestType::Read;
43}

115void AContactTrigger::OnTriggerBeginOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
116 UPrimitiveComponent* OtherComp, int32 OtherBodyIndex, bool bFromSweep, const FHitResult& SweepResult)
117{

120 return;

124 if (PlayerActor)

126 // 서버 RPC 호출

131void AContactTrigger::ServerRPC_OnTrigger_Implementation(AActor* TriggeringActor)
132{

144void AContactTrigger::OnTriggerScenario(EQuestType InQuestType)
145{
146 if (auto KLingoNetwork = UKLingoNetworkSystem::Get(GetWorld()))
147 {
148 switch (InQuestType)
149 {
150 case EQuestType::Read:
151 KLingoNetwork->RequestReadScenario( FResponseReadScenarioDelegate::CreateUObject(this, &AContactTrigger::OnReadResponseScenario));
152 break;
153
154 case EQuestType::Listen: // 듣기
155 KLingoNetwork->RequestListenScenario( FResponseListenScenarioDelegate::CreateUObject(this, &AContactTrigger::OnListenResponseScenario));
156 break;
157 }
158 }
159}

161void AContactTrigger::OnReadResponseScenario(FResponseReadScenario& ResponseData, bool bWasSuccessful)
162{
163 if (!bWasSuccessful)
164 return;
165
166 // ALingoGameState에 시나리오 데이터 전체 저장
167 if (UWorld* World = GetWorld())
168 {
169 if (auto GM = ULingoGameHelper::GetLingoGameMode(World))
170 GM->BeginReadQuest( ResponseData);
171
172 ALuggageManager* LuggageManager = Cast<ALuggageManager>( UGameplayStatics::GetActorOfClass(World, ALuggageManager::StaticClass()));
173
174 if (LuggageManager)
175 {
176 LuggageManager->StartSpawning();
177 // LuggageManager->InitHolder(ResponseData);
178 }
179 }
180}

182void AContactTrigger::OnListenResponseScenario(FResponseListenScenario& ResponseData, bool bWasSuccessful)
183{
184 if (!bWasSuccessful)
185 return;
186
187 if (UWorld* World = GetWorld())
188 {
189 if (auto GM = ULingoGameHelper::GetLingoGameMode(World))
190 GM->BeginListenQuest(ResponseData);
191
192 VoiceConversationSystem->PlayVoiceAudio(ResponseData.voice_data);
193
194 AFoodCourtManager* FCourtManager = Cast<AFoodCourtManager>(UGameplayStatics::GetActorOfClass(
195 World, AFoodCourtManager::StaticClass()));
196 if (FCourtManager)
197 {
198 FCourtManager->SetFoodCourtInfo();
199 }
200 }
