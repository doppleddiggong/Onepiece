42void UPopup_InputMsg::OnClickOk()
43{
44 const FString InputString = Edit_Name->GetText().ToString();
45
46 if ( PopupType == EPopupType::Login )
47 RequestUserToken(InputString);
48 else
49 RequestUserRegister(InputString);
50
51 // PopupManager를 통해 팝업 닫기 (마우스 커서 처리 포함)
52 if (UPopupManager* PopupMgr = UPopupManager::Get(GetWorld()))

83void UPopup_InputMsg::RequestUserRegister(const FString& UserInput)
84{
85 if (UserInput.IsEmpty())
86 return;
87
88 if (auto KLingoNetwork = UKLingoNetworkSystem::Get(GetWorld()))
89 {
90 PRINTLOG(TEXT("[TEST] RequestUserRegister - UserName: %s"), *UserInput);
91 KLingoNetwork->RequestUserRegister(
92 UserInput,
93 FResponseUserRegisterDelegate::CreateUObject(this,
94 &UPopup_InputMsg::OnResponseUserRegister)
95 );
96 }
97 else
98 {
99 PRINTLOG(TEXT("UKLingoNetworkSystem not found!"));
100 }
101}
