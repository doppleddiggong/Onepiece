77void ALingoGameState::Tick(float DeltaSeconds)
78{
79 Super::Tick(DeltaSeconds);
80
81 // 서버에서만 타이머 처리
82 if (!HasAuthority() || !bIsTimerActive)
83 return;
84
85 RemainMissionTime -= DeltaSeconds;
86
87 if (RemainMissionTime <= 0.f)
88 {
89 RemainMissionTime = 0.f;
90 OnMissionTimerEnd();
91 }
92}

110void ALingoGameState::SetReadScenarioData( const FResponseReadScenario& InResponseData)
111{
112 // 반드시 서버 전용
113 if (!HasAuthority())
114 return;
115
116 this->QuestType = EQuestType::Read;
117 this->ReadScenarioData = InResponseData;
118
119 // 미션 타이머 시작
120 this->StartMissionTimer( ULingoGameHelper::GetMissionPlayTime() );
121
122 // 모든 플레이어의 ReadQuest 진행 상태 설정
123 for (APlayerState* PS : PlayerArray)
124 {
125 if (ALingoPlayerState* LingoPS = Cast<ALingoPlayerState>(PS))
126 {
127 LingoPS->SetReadQuestIng(true);
128 }
129 }
130
131 if (HasAuthority())
132 {
133 Multicast_UpdateQuestType(QuestType);
134 Multicast_ShowReadQuestPopup(ReadScenarioData);
135 }
136}

222void ALingoGameState::OnMissionTimerEnd()
223{
224 if (!HasAuthority())
225 return;
226
227 bIsTimerActive = false;
228
229 // NetworkBroadcastActor를 통해 모든 클라이언트에 타이머 종료 알림
230 if (ANetworkBroadcastActor* BroadcastActor = ANetworkBroadcastActor::Get(this))
231 BroadcastActor->SendUpdateMissionTimerState(false, 0.0f, this);
232
233 auto EndMessage = ULingoGameHelper::GetStageEndMessage(QuestType);
234
235 PRINTLOG( TEXT("[GameState] Mission Timer Ended - Sending: %s"), *EndMessage);
236
237 // Read, Listen Quest일 때 자동으로 결과 팝업 표시
238 if (QuestType == EQuestType::Read || QuestType == EQuestType::Listen)
239 {
240 if (auto Popup = UPopupManager::ShowPopupAs<UPopup_Result>(GetWorld(), EPopupType::Result))
241 {
242 Popup->InitPopup(QuestType);
243 }
244 }
245
246 // 모든 플레이어에게 메시지 전송
247 if (UWorld* World = GetWorld())
248 {
249 for (FConstPlayerControllerIterator It = World->GetPlayerControllerIterator(); It; ++It)
250 {
251 if (APlayerController* PC = It->Get())
252 {
253 if (APlayerActor* PlayerActor = Cast<APlayerActor>(PC->GetPawn()))
254 {
255 PlayerActor->OnGameMessage(EndMessage);
256 }
257 }
258 }
259 }
260}
