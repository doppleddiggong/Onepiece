51void ALingoGameState::Tick(float DeltaSeconds)
52{
53 Super::Tick(DeltaSeconds);
54
55 // 서버에서만 타이머 처리
56 if (!HasAuthority() || !bIsTimerActive)
57 return;
58
59 RemainMissionTime -= DeltaSeconds;
60
61 if (RemainMissionTime <= 0.f)
62 {
63 RemainMissionTime = 0.f;
64 OnMissionTimerEnd();
65 }
66}

68void ALingoGameState::SetStageData(int InStageIndex, const FResponseScenario& InResponseData)
69{
70 PRINTLOG(TEXT("[GameState] SetStageData - StageIndex: %d, HasAuthority: %s"),
71 InStageIndex, HasAuthority() ? TEXT("true") : TEXT("false"));
72
73 if ( InStageIndex == 1)
74 GameState = EGameState::Stage1;
75 else if ( InStageIndex == 2)
76 GameState = EGameState::Stage2;
77 else if ( InStageIndex == 3)
78 GameState = EGameState::Stage3;
79 else if ( InStageIndex == 4)
80 GameState = EGameState::Stage4;
81
82 PRINTLOG(TEXT("[GameState] GameState changed to: %d"), static_cast<int32>(GameState));
83
84 this->StageIndex = InStageIndex;
85 this->CurScenarioData = InResponseData;
86
87 // 미션 타이머 시작
88 this->StartMissionTimer( ULingoGameHelper::GetMissionPlayTime(ScenarioLevel) );
89
90 // 모든 클라이언트(서버 포함)에 팝업 표시 요청
91 if (HasAuthority())
92 {
93 Multicast_ShowReadQuestPopup(InStageIndex, CurScenarioData);
94 }
95}

97void ALingoGameState::StartMissionTimer(float TimeLimit)
98{
99 // 서버에서만 실행
100 if (!HasAuthority())
101 return;
102
103 RemainMissionTime = TimeLimit;
104 bIsTimerActive = true;
105
106 // NetworkBroadcastActor를 통해 모든 클라이언트에 타이머 시작 알림
107 if (ANetworkBroadcastActor* BroadcastActor = ANetworkBroadcastActor::Get(this))
108 {
109 BroadcastActor->SendUpdateMissionTimerState(true, TimeLimit, this);
110 }
111
112 PRINTLOG( TEXT("[GameState] Mission Timer Started - %.0f seconds"), TimeLimit);
113}

115void ALingoGameState::StopMissionTimer()
116{
117 if (!HasAuthority())
118 return;
119
120 bIsTimerActive = false;
121
122 // NetworkBroadcastActor를 통해 모든 클라이언트에 타이머 중지 알림
123 if (ANetworkBroadcastActor* BroadcastActor = ANetworkBroadcastActor::Get(this))
124 {
125 BroadcastActor->SendUpdateMissionTimerState(false, 0.0f, this);
126 }
127
128 PRINTLOG( TEXT("[GameState] Mission Timer Stopped"));
129}

131void ALingoGameState::OnMissionTimerEnd()
132{
133 if (!HasAuthority())
134 return;
135
136 bIsTimerActive = false;
137
138 // NetworkBroadcastActor를 통해 모든 클라이언트에 타이머 종료 알림
139 if (ANetworkBroadcastActor* BroadcastActor = ANetworkBroadcastActor::Get(this))
140 {
141 BroadcastActor->SendUpdateMissionTimerState(false, 0.0f, this);
142 }
143
144 auto EndMessage = ULingoGameHelper::GetStageEndMessage(StageIndex);
145
146 PRINTLOG( TEXT("[GameState] Mission Timer Ended - Sending: %s"), *EndMessage);
147
148 // 모든 플레이어에게 메시지 전송
149 if (UWorld* World = GetWorld())
150 {
151 for (FConstPlayerControllerIterator It = World->GetPlayerControllerIterator(); It; ++It)
152 {
153 if (APlayerController* PC = It->Get())
154 {
155 if (APlayerActor* PlayerActor = Cast<APlayerActor>(PC->GetPawn()))
156 {
157 PlayerActor->OnGameMessage(EndMessage);
158 }
159 }
160 }
161 }
162}

179void ALingoGameState::Multicast_ShowReadQuestPopup_Implementation(int InStageIndex, const FResponseScenario& InScenarioData)
180{
181 PRINTLOG(TEXT("[GameState] Multicast_ShowReadQuestPopup - StageIndex: %d, Role: %s"),
182 InStageIndex, GetLocalRole() == ROLE_Authority ? TEXT("Server") : TEXT("Client"));
183
184 // Stage1일 때만 Read Quest 팝업 표시
185 if (InStageIndex == 1)
186 {
187 if (UPopupManager* PopupMgr = UPopupManager::Get(GetWorld()))
188 {
189 if (UPopup_ReadQuest* Popup = Cast<UPopup_ReadQuest>(PopupMgr->ShowPopup(EPopupType::ReadQuest)))
190 {
191 Popup->InitPopup(InScenarioData);
192 PRINTLOG(TEXT("[GameState] Read Quest Popup displayed for Stage1"));
193 }
194 }
195 }
