44void ALingoGameState::Tick(float DeltaSeconds)
45{
46 Super::Tick(DeltaSeconds);
47
48 // 서버에서만 타이머 처리
49 if (!HasAuthority() || !bIsTimerActive)
50 return;
51
52 RemainMissionTime -= DeltaSeconds;
53
54 if (RemainMissionTime <= 0.f)
55 {
56 RemainMissionTime = 0.f;
57 OnMissionTimerEnd();
58 }
59}

85void ALingoGameState::StartMissionTimer(float InTimeLimit)
86{
87 // 서버에서만 실행
88 if (!HasAuthority())
89 return;
90
91 this->TimeLimit = InTimeLimit;
92 this->RemainMissionTime = InTimeLimit;
93 this->bIsTimerActive = true;
94
95 if (ANetworkBroadcastActor* BroadcastActor = ANetworkBroadcastActor::Get(this))
96 BroadcastActor->SendUpdateMissionTimerState(true, InTimeLimit, this);
97}

124void ALingoGameState::OnMissionTimerEnd()
125{
126 if (!HasAuthority())
127 return;
128
129 bIsTimerActive = false;
130
131 // NetworkBroadcastActor를 통해 모든 클라이언트에 타이머 종료 알림
132 if (ANetworkBroadcastActor* BroadcastActor = ANetworkBroadcastActor::Get(this))
133 BroadcastActor->SendUpdateMissionTimerState(false, 0.0f, this);
134
135 auto EndMessage = ULingoGameHelper::GetStageEndMessage(CurrentQuestData.QuestType);
136
137 PRINTLOG( TEXT("[GameState] Mission Timer Ended - Sending: %s"), *EndMessage);
138
139 // 모든 플레이어에게 메시지 전송
140 if (UWorld* World = GetWorld())
141 {
142 for (FConstPlayerControllerIterator It = World->GetPlayerControllerIterator(); It; ++It)
143 {
144 if (APlayerController* PC = It->Get())
145 {
146 if (APlayerActor* PlayerActor = Cast<APlayerActor>(PC->GetPawn()))
147 {
148 PlayerActor->OnGameMessage(EndMessage);
149 }
150 }
151 }
152 }
153}
