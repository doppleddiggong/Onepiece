48UUserWidget* UPopupManager::ShowPopup(EPopupType Type)
49{
50 UUserWidget* PopupWidget = EnsurePopupWidget(Type);
51 if (!PopupWidget)
52 {
53 PRINTLOG(TEXT("[UPopupManager] Failed to create popup: %s"), *ENUM_TO_NAME(EPopupType, Type));
54 return nullptr;
55 }
56
57 // 첫 팝업이 열릴 때 마우스 커서 표시
58 bool bIsFirstPopup = (PopupStack.Num() == 0);
59
60 // 이미 뷰포트에 있으면 스택 맨 위로 이동
61 if (PopupWidget->IsInViewport())
62 {
63 RemovePopupFromStack(Type);
64 }
65 else
66 {
67 // UBasePopup으로 캐스팅하여 애니메이션 재생
68 if (UBasePopup* BasePopup = Cast<UBasePopup>(PopupWidget))
69 {
70 BasePopup->OpenAnimation();
71 }
72 else
73 {
74 // UBasePopup이 아닌 경우 기본 방식으로 추가
75 PopupWidget->AddToViewport(GameLayer::Popup);
76 }
77 }
78
79 // 스택에 추가
80 PushPopupToStack(Type);
81
82 // 첫 팝업이 열렸다면 마우스 커서 표시
83 if (bIsFirstPopup)
84 {
85 ULingoGameHelper::ShowMouseCursor(GetWorld());
86 }
87
88 return PopupWidget;
89}

91void UPopupManager::HidePopup(EPopupType Type, bool bDestroyWidget)
92{
93 UUserWidget** PopupWidgetPtr = PopupWidgetMap.Find(Type);
94 if (!PopupWidgetPtr || !(*PopupWidgetPtr))
95 return;
96
97 UUserWidget* PopupWidget = *PopupWidgetPtr;
98
99 // 뷰포트에서 제거
100 if (PopupWidget->IsInViewport())
101 {
102 PopupWidget->RemoveFromParent();
103 }
104
105 // 스택에서 제거
106 RemovePopupFromStack(Type);
107
108 // 모든 팝업이 닫혔다면 마우스 커서 숨기기
109 if (PopupStack.Num() == 0)
110 {
111 FString MapName = GetWorld()->GetMapName();
112 MapName.RemoveFromStart(GetWorld()->StreamingLevelsPrefix);
113 if (MapName.Contains(TEXT("Map1")) ||
114 MapName.Contains(TEXT("Person")) )
115 {
116 ULingoGameHelper::HideMouseCursor(GetWorld());
117 }
118 }
119
120 // 위젯 파괴
121 if (bDestroyWidget)
122 {
123 PopupWidgetMap.Remove(Type);
124 }
125}

171 return nullptr;

262UUserWidget* UPopupManager::EnsurePopupWidget(EPopupType Type)
263{
264 UWorld* World = GetWorld();
265 if (!World || !World->IsGameWorld())
266 return nullptr;
267
268 // 이미 생성된 위젯이 있는지 확인
269 UUserWidget** ExistingWidgetPtr = PopupWidgetMap.Find(Type);
270 if (ExistingWidgetPtr && IsValid(*ExistingWidgetPtr))
271 {
272 UUserWidget* ExistingWidget = *ExistingWidgetPtr;
273 // 같은 월드인지 확인
274 if (ExistingWidget->GetWorld() == World)
275 {
276 return ExistingWidget;
277 }
278 else
279 {
280 // 다른 월드면 제거
281 if (ExistingWidget->IsInViewport())
282 ExistingWidget->RemoveFromParent();
283 PopupWidgetMap.Remove(Type);
284 }
285 }
286
287 // 팝업 클래스 가져오기
288 TSubclassOf<UUserWidget>* PopupClassPtr = PopupClassMap.Find(Type);
289 if (!PopupClassPtr || !(*PopupClassPtr))
290 {
291 PRINTLOG(TEXT("[UPopupManager] No popup class registered for type: %s"),
292 *ENUM_TO_NAME(EPopupType, Type));
293 return nullptr;
294 }
295
296 // 플레이어 컨트롤러 가져오기
297 ULocalPlayer* LocalPlayer = GetLocalPlayer();
298 if (!LocalPlayer)
299 return nullptr;
300
301 APlayerController* PC = LocalPlayer->GetPlayerController(World);
302 if (!PC)
303 return nullptr;
304
305 // 위젯 생성
306 UUserWidget* NewWidget = CreateWidget<UUserWidget>(PC, *PopupClassPtr);
307 if (!NewWidget)
308 {
309 PRINTLOG(TEXT("[UPopupManager] Failed to create widget for type: %s"),
310 *ENUM_TO_NAME(EPopupType, Type));
311 return nullptr;
312 }
313
314 // 맵에 추가
315 PopupWidgetMap.Add(Type, NewWidget);
316
317 return NewWidget;
318}
