40void UPopup_Questionnaire::InitPopup(const FQuestWriteInfo& QuestionData)
41{
42 // 질문 데이터 저장
43 if (ALingoPlayerState* lps = ULingoGameHelper::GetLingoPlayerState(GetWorld()))
44 {
45 lps->WriteQuestionData = QuestionData;
46 }
47 SavedQuestions = QuestionData.question;
48
49 // 기존 항목 제거
50 VerticalBox->ClearChildren();
51
52 // 각 질문에 대해 항목 위젯 생성
53 for (int32 i = 0; i < SavedQuestions.Num(); ++i)
54 {
55 const FWriteQuestionData& Data = SavedQuestions[i];
56
57 // 인터뷰 항목 위젯 생성
58 UPopup_QuestionnaireItem* ItemWidget = CreateWidget<UPopup_QuestionnaireItem>(
59 GetWorld(), QuestionnaireItemClass);
60 ItemWidget->InitItem(Data);
61 VerticalBox->AddChildToVerticalBox(ItemWidget);
62
63 // 마지막 항목이 아니면 Spacer 추가
64 if (i < SavedQuestions.Num() - 1)
65 {
66 USpacer* Spacer = NewObject<USpacer>(this);
67 if (Spacer)
68 {
69 Spacer->SetSize(FVector2D(1.0f, ItemSpacing));
70 VerticalBox->AddChildToVerticalBox(Spacer);
71 }
72 }
73 }
74}

85void UPopup_Questionnaire::OnClickSubmit()
86{
87 // 사진 파일 모으기
88 // 네트워크 전송
89 if (auto KLingoNetwork = UKLingoNetworkSystem::Get(GetWorld()))
90 {
91 // Question마다 png파일 1개로 보내기
92 TArray<FString> pngFiles;
93 TArray<FString> targetTexts;
94 for (const auto& question : SavedQuestions)
95 {
96 FString imagePath = WriteImagePath + FString::Printf(TEXT("Answer%d.PNG"), question.Id);
97 PRINTLOG(TEXT("[TEST] RequestOcrExtract - ImagePath: %s"), *imagePath);
98
99 pngFiles.Add(imagePath);
100 targetTexts.Add(question.answer_kor);
101 }
102
103 KLingoNetwork->RequestWriteSubmit(
104 pngFiles,
105 targetTexts,
106 FResponseWriteSubmitDelegate::CreateUObject(this, &UPopup_Questionnaire::OnResponseOcrExtract)
107 );
108
109 OnClickClose();
110 }
111 else
112 {
113 PRINTLOG(TEXT("UKLingoNetworkSystem not found!"));
114 }
115}

117void UPopup_Questionnaire::OnResponseOcrExtract(FResponseWriteSubmit& ResponseData, bool bWasSuccessful)
118{
119 if (bWasSuccessful)
120 {
121 PRINTLOG(TEXT("--- OCR Extract SUCCESS ---"));
122
123 if (ALingoPlayerState* lps = ULingoGameHelper::GetLingoPlayerState(GetWorld()))
124 {
125 lps->WriteSubmitResultData = ResponseData;
126 }
127
128 if (auto GS = Cast<ALingoGameState>(GetWorld()->GetGameState()))
129 {
130 // 네트워크 송신 (Host만)
131 if (UKLingoNetworkSystem* KLingoNetwork = UKLingoNetworkSystem::Get(GetWorld()))
132 {
133 FRequestWriteResult WriteRequest;
134 WriteRequest.room_id = GS->GetRoomId();
135 WriteRequest.user_id = Cast<APlayerControl>(GetOwningPlayer())->GetUserId();
136 WriteRequest.scenario_id = 1;
137 WriteRequest.stage_type = ULingoGameHelper::GetStageTypeIndex(EQuestType::Write);
138 WriteRequest.state_type = 0;
139 WriteRequest.result_time = GS->GetTimeTaken();
140 WriteRequest.wrong_idx = GS->WrongReadAnswerList;
141
142 KLingoNetwork->RequestWriteResult(
143 WriteRequest,
144 FResponseWriteResultDelegate::CreateUObject(this, &UPopup_Questionnaire::OnResponseWriteResult)
145 );
146 }
147 }
148 }
149 else
150 {
151 PRINTLOG(TEXT("--- OCR Extract FAILED ---"));
152 }
153}

155void UPopup_Questionnaire::OnResponseWriteResult(FResponseWriteResult& ResponseData, bool bWasSuccessful)
156{
157 if (bWasSuccessful)
158 {
159 PRINTLOG(TEXT("--- OCR Extract SUCCESS ---"));
160
161 ALingoPlayerState* lps = ULingoGameHelper::GetLingoPlayerState(GetWorld());
162 if (lps)
163 {
164 lps->WriteWholeResultData = ResponseData;
165 }
166
167 // 피드백 팝업 창 생성
168 if (auto Popup = UPopupManager::ShowPopupAs<UPopup_QuestionnaireResult>(GetWorld(), EPopupType::QuestionnaireResult))
169 {
170 // 팝업 초기화
171 Popup->InitPopup(lps->WriteSubmitResultData, ResponseData);
172
173 PRINTLOG(TEXT("[PopupTester] Result popup opened"));
174 }
175 }
176 else
177 {
178 PRINTLOG(TEXT("--- OCR Extract FAILED ---"));
179 }
