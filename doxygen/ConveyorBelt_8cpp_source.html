67void AConveyorBelt::MoveOverlappedSkeletals(float deltaDistance)
68{
69 TArray<AActor*> overlappedActors;
70 BeltBoxComp->GetOverlappingActors(overlappedActors);
71 for (const auto& actor : overlappedActors)
72 {
73 if (Cast<APlayerActor>(actor) || Cast<ANPCBase>(actor))
74 {
75 FVector deltaLoc = MoveDirArrowComp->GetForwardVector() * deltaDistance;
76 deltaLoc = bIsForward ? deltaLoc : -deltaLoc;
77 actor->AddActorWorldOffset(deltaLoc);
78 }
79 }
80}

82void AConveyorBelt::MoveOverlappedStatics(float deltaDistance)
83{
84 TArray<UPrimitiveComponent*> overlappedComponents;
85 BeltBoxComp->GetOverlappingComponents(overlappedComponents);
86 for (const auto& comp : overlappedComponents)
87 {
88 AActor* owner = comp->GetOwner();
89 if (Cast<Aluggage>(owner) && Cast<UStaticMeshComponent>(comp))
90 {
91 FVector deltaLoc = MoveDirArrowComp->GetForwardVector() * deltaDistance;
92 deltaLoc = bIsForward ? deltaLoc : -deltaLoc;
93 owner->AddActorWorldOffset(deltaLoc);
94 }
95 }
96}
