55void USpeakStageSubsystem::CreateSpeakStageSystem()
56{
57 UWorld* World = GetWorld();
58 if (!World)
59 return;
60
61 // 이미 초기화되었으면 중복 생성 방지
62 if (bIsInitialized)
63 return;
64
65 // 특정 맵에서만 생성: Map1, Lvl_ThirdPerson
66 FString MapName = World->GetMapName();
67 MapName.RemoveFromStart(World->StreamingLevelsPrefix); // PIE prefix 제거
68
69 if (!MapName.Equals(TEXT("Map1")) && !MapName.Equals(TEXT("Lvl_ThirdPerson")))
70 {
71 // 다른 맵에서는 초기화 완료로 표시
72 bIsInitialized = true;
73 return;
74 }
75
76 // 클라이언트 월드에서는 생성하지 않음
77 if (World->IsNetMode(NM_Client))
78 {
79 return;
80 }
81
82 // 1. SpeakStageActor 생성
83 CreateSpeakStageActor();
84
85 // 2. NPC Examiner 생성
86 CreateNPCExaminer();
87
88 bIsInitialized = true;
89}

91void USpeakStageSubsystem::BeginSpeakQuest(int32 InStageIndex)
92{
93 if (!SpeakStageActor)
94 return;
95
96 UWorld* World = GetWorld();
97 if (!World)
98 return;
99
100 // 모든 플레이어 수집
101 TArray<APlayerState*> Players;
102 for (auto It = World->GetPlayerControllerIterator(); It; ++It)
103 {
104 if (APlayerController* PC = It->Get())
105 {
106 if (APlayerState* PS = PC->GetPlayerState<APlayerState>())
107 {
108 Players.Add(PS);
109 }
110 }
111 }
112
113 // PlayStart 호출 (명시적 초기화)
114 SpeakStageActor->PlayStart(Players);
115}

121void USpeakStageSubsystem::CreateSpeakStageActor()
122{
123 UWorld* World = GetWorld();
124 if (!World)
125 {
126 return;
127 }
128
129 // 1. 먼저 월드에서 기존 SpeakStageActor 찾기
130 SpeakStageActor = Cast<ASpeakStageActor>(UGameplayStatics::GetActorOfClass(World, ASpeakStageActor::StaticClass()));
131
132 if (SpeakStageActor)
133 return;
134
135 // 2. 없으면 Config에서 클래스 로드하여 생성
136 if (SpeakStageClass.IsNull())
137 return;
138
139 // 클래스 로드
140 UClass* LoadedClass = SpeakStageClass.LoadSynchronous();
141 if (!LoadedClass)
142 return;
143
144 // SpeakStageActor 생성
145 FActorSpawnParameters SpawnParams;
146 SpawnParams.Owner = nullptr;
147 SpawnParams.SpawnCollisionHandlingOverride = ESpawnActorCollisionHandlingMethod::AdjustIfPossibleButAlwaysSpawn;
148
149 SpeakStageActor = World->SpawnActor<ASpeakStageActor>(
150 LoadedClass,
151 FVector::ZeroVector,
152 FRotator::ZeroRotator,
153 SpawnParams
154 );
155
156 if (SpeakStageActor)
157 {
158 PRINTLOG(TEXT("[SpeakStageSubsystem] SpeakStageActor spawned successfully"));
159 }
160 else
161 {
162 PRINTLOG(TEXT("[SpeakStageSubsystem] Failed to spawn SpeakStageActor"));
163 }
164}

166void USpeakStageSubsystem::CreateNPCExaminer()
167{
168 UWorld* World = GetWorld();
169 if (!World)
170 return;
171
172 // 1. 먼저 월드에서 기존 NPCExaminer 찾기
173 Examiner = Cast<ANPCExaminer>(UGameplayStatics::GetActorOfClass(World, ANPCExaminer::StaticClass()));
174
175 if (Examiner)
176 {
177 // 기존 NPC에 SpeakStage 연결
178 if (SpeakStageActor)
179 Examiner->SetSpeakStage(SpeakStageActor);
180
181 return;
182 }
183
184 // 2. 없으면 Config에서 클래스 로드하여 생성
185 if (ExaminerClass.IsNull())
186 return;
187
188 // 클래스 로드
189 UClass* LoadedClass = ExaminerClass.LoadSynchronous();
190 if (!LoadedClass)
191 return;
192
193 // NPC Examiner 생성
194 FActorSpawnParameters NPCSpawnParams;
195 NPCSpawnParams.Owner = nullptr;
196 NPCSpawnParams.SpawnCollisionHandlingOverride =
197 ESpawnActorCollisionHandlingMethod::AdjustIfPossibleButAlwaysSpawn;
198
199 Examiner = World->SpawnActor<ANPCExaminer>(
200 LoadedClass,
201 ExaminerSpawnTransform,
202 NPCSpawnParams
203 );
204
205 if (Examiner)
206 {
207 // NPC에 SpeakStage 연결
208 if (SpeakStageActor)
209 {
210 Examiner->SetSpeakStage(SpeakStageActor);
211 PRINTLOG(TEXT("[SpeakStageSubsystem] Connected spawned Examiner to SpeakStage"));
212 }
213 }
214 else
215 {
216 PRINTLOG(TEXT("[SpeakStageSubsystem] Failed to spawn Examiner NPC"));
217 }
218}
