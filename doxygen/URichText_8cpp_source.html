41void URichText::ApplyStyle()
42{
43 if (!RichTxt)
44 return;
45
46 // 기존 폰트 정보 가져오기 (없으면 기본)
47 FSlateFontInfo NewFont = RichTxt->GetDefaultTextStyle().Font;
48 NewFont.Size = FontSize;
49
50 // 새로운 스타일 생성
51 FTextBlockStyle NewStyle;
52 NewStyle.SetFont(NewFont);
53 NewStyle.SetColorAndOpacity(FSlateColor(TextColor));
54
55 RichTxt->SetDefaultTextStyle(NewStyle);
56 RichTxt->SetText(FText::FromString(DefaultText));
57
58 // HyperLink Decorator 스타일도 함께 업데이트
59 if (UHyperLinkPluginBPLibrary* LinkDecorator = Cast<UHyperLinkPluginBPLibrary>(
60 RichTxt->GetDecoratorByClass(UHyperLinkPluginBPLibrary::StaticClass())))
61 {
62 FTextBlockStyle LinkTextStyle = LinkDecorator->Style.TextStyle;
63 LinkTextStyle.SetFont(NewFont);
64 LinkTextStyle.SetColorAndOpacity(FSlateColor(TextColor));
65 LinkDecorator->Style.SetTextStyle(LinkTextStyle);
66 }
67}

69void URichText::SetText(const FWordData& WordData)
70{
71 if (!IsValid(RichTxt))
72 return;
73
74 CachedPhonemeData = WordData.GetPhonemeData();
75
76 FString TextString;
77 for (int32 i = 0; i < CachedPhonemeData.Num(); ++i)
78 {
79 const FPhonemeData& Phoneme = CachedPhonemeData[i];
80 TextString += Phoneme.ToRichTextString(i);
81 }
82 RichTxt->SetText(FText::FromString(TextString));
83}

98void URichText::OnClickLink(const FString& LinkID, const FString& Content)
99{
100 const int32 Index = FCString::Atoi(*LinkID);
101
102 if (CachedPhonemeData.IsValidIndex(Index))
103 {
104 const FPhonemeData& Phoneme = CachedPhonemeData[Index];
105 PRINTLOG(TEXT("%s, %s"), *Phoneme.Kor, *Phoneme.Pronunciation);
106 }
107}
