29TArray<FPhonemeData> FWordData::GetPhonemeData() const
30{
31 TArray<FString> KorWords;
32 Kor.ParseIntoArray(KorWords, TEXT(" "), true);
33
34 TArray<FString> PronWords;
35 Pronunciation.ParseIntoArray(PronWords, TEXT(" "), true);
36
37 TArray<FPhonemeData> WordDataArray;
38
39 for (int32 i = 0; i < KorWords.Num(); ++i)
40 {
41 FPhonemeData Data;
42 Data.Kor = i < KorWords.Num() ? KorWords[i] : TEXT("");
43 Data.Pronunciation = i < PronWords.Num() ? PronWords[i] : TEXT("");
44
45 WordDataArray.Add(Data);
46 }
47
48 return WordDataArray;

375void FResponseWriteSubmit::SetFromHttpResponse(const TSharedPtr<IHttpResponse, ESPMode::ThreadSafe>& Response)
376{
377 if (!Response.IsValid())
378 {
379 return;
380 }
381
382 FString JsonString = Response->GetContentAsString();
383
384 TArray<TSharedPtr<FJsonValue>> JsonArray;
385 TSharedRef<TJsonReader<>> Reader = TJsonReaderFactory<>::Create(JsonString);
386
387 if (FJsonSerializer::Deserialize(Reader, JsonArray))
388 {
389 for (const TSharedPtr<FJsonValue>& EntryValue : JsonArray)
390 {
391 if (EntryValue.IsValid())
392 {
393 TSharedPtr<FJsonObject> EntryObject = EntryValue->AsObject();
394 if (!EntryObject.IsValid())
395 {
396 continue;
397 }
398
399 FResponseWriteData Entry; // Display & Record를 포함하는 사용자 정의 구조체
400
401 if (EntryObject->HasTypedField<EJson::Object>(TEXT("display")))
402 {
403 const TSharedPtr<FJsonObject> DisplayObj = EntryObject->GetObjectField(TEXT("display"));
404 Entry.display.is_pass = DisplayObj->GetBoolField(TEXT("is_pass"));
405 Entry.display.message = DisplayObj->GetStringField(TEXT("message"));
406 Entry.display.correction = DisplayObj->GetStringField(TEXT("correction"));
407 }
408
409 if (EntryObject->HasTypedField<EJson::Object>(TEXT("record")))
410 {
411 const TSharedPtr<FJsonObject> RecordObj = EntryObject->GetObjectField(TEXT("record"));
412 Entry.record.score = RecordObj->GetIntegerField(TEXT("score"));
413 Entry.record.target = RecordObj->GetStringField(TEXT("target"));
414 Entry.record.input = RecordObj->GetStringField(TEXT("input"));
415 Entry.record.stage = RecordObj->GetStringField(TEXT("stage"));
416 }
417
418 ResponseWriteDataArray.Add(Entry);
419 }
420 }
421 }
