22void UPopup_ReadQuest::InitPopup(const FResponseScenario& InScenarioData)
23{
24 if (Btn_Exit)
25 {
26 Btn_Exit->OnButtonClickedEvent.RemoveDynamic(this, &UPopup_ReadQuest::OnClickClose);
27 Btn_Exit->OnButtonClickedEvent.AddDynamic(this, &UPopup_ReadQuest::OnClickClose);
28 }
29
30 if (auto BM = UBroadcastManager::Get(GetWorld()))
31 {
32 BM->OnUpdateQuestRole.RemoveDynamic(this, &UPopup_ReadQuest::InitQuestInfo);
33 BM->OnUpdateQuestRole.AddDynamic(this, &UPopup_ReadQuest::InitQuestInfo);
34 }
35
36 this->ScenarioData = InScenarioData;
37
38 if ( const auto PS = ULingoGameHelper::GetLingoPlayerState(GetWorld()) )
39 InitQuestInfo(PS->QuestRole);
40}

42void UPopup_ReadQuest::InitQuestInfo(EQuestRole QuestRole)
43{
44 if ( QuestRole == EQuestRole::Both )
45 {
46 WordWidget->InitWordData(ScenarioData.full_data);
47 ListenAudio(ScenarioData.full_data.Kor);
48 }
49 else if ( QuestRole == EQuestRole::OnlyQuestion1 )
50 {
51 WordWidget->InitWordData(ScenarioData.word_data1);
52 ListenAudio(ScenarioData.word_data1.Kor);
53 }
54 else if ( QuestRole == EQuestRole::OnlyQuestion2 )
55 {
56 WordWidget->InitWordData(ScenarioData.word_data2);
57 ListenAudio(ScenarioData.word_data2.Kor);

69void UPopup_ReadQuest::ListenAudio(const FString& AudioText)
70{
71 if (bIsRequest)
72 return;
73
74 if (auto KLingoNetwork = UKLingoNetworkSystem::Get(GetWorld()))
75 {
76 bIsRequest = true;
77
78 KLingoNetwork->RequestListenAudio(
79 AudioText,
80 FResponseListenAudioDelegate::CreateUObject(this, &UPopup_ReadQuest::OnResponseListenAudio)
81 );
82 }
83}

85void UPopup_ReadQuest::OnResponseListenAudio(FResponseListenAudio& ResponseData, bool bWasSuccessful)
86{
87 bIsRequest = false;
88
89 if (bWasSuccessful)
90 {
91 if (auto PlayerActor = ULingoGameHelper::GetPlayerActor(this))
92 PlayerActor->PlayTTSAudio(ResponseData.audio_base64);
93 }
