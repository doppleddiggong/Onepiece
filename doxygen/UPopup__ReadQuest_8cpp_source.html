18{

60void UPopup_ReadQuest::InitQuestInfo(EQuestRole QuestRole)
61{
62 if ( QuestType == EQuestType::Read )
63 {
64 if ( QuestRole == EQuestRole::Both )
65 {
66 WordWidget->InitWordData(ReadData.full_data);
67 ListenAudio(ReadData.full_data.Kor);
68 }
69 else if ( QuestRole == EQuestRole::OnlyQuestion1 )
70 {
71 WordWidget->InitWordData(ReadData.word_data1);
72 ListenAudio(ReadData.word_data1.Kor);
73 }
74 else if ( QuestRole == EQuestRole::OnlyQuestion2 )
75 {
76 WordWidget->InitWordData(ReadData.word_data2);
77 ListenAudio(ReadData.word_data2.Kor);
78 }
79 }
80 else if ( QuestType == EQuestType::Listen )

108void UPopup_ReadQuest::ListenAudio(const FString& AudioText)
109{
110 if (bIsRequest)
111 return;
112
113 if (auto KLingoNetwork = UKLingoNetworkSystem::Get(GetWorld()))
114 {
115 bIsRequest = true;
116
117 KLingoNetwork->RequestListenAudio(
118 AudioText,
119 FResponseListenAudioDelegate::CreateUObject(this, &UPopup_ReadQuest::OnResponseListenAudio)
120 );
121 }
122}

124void UPopup_ReadQuest::OnResponseListenAudio(FResponseListenAudio& ResponseData, bool bWasSuccessful)
125{
126 bIsRequest = false;
127
128 if (bWasSuccessful)
129 {
130 if (auto PlayerActor = ULingoGameHelper::GetPlayerActor(this))
131 PlayerActor->PlayTTSAudio(ResponseData.audio_base64);
132 }
