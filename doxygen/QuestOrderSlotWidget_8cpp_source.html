28void UQuestOrderSlotWidget::PlayStampAnimation(float InDeltaTime)
29{
30 auto easeOutElastic = [](float x)->float
31 {
32 const float c4 = (2 * PI) / 3;
33
34 if (x == 0)
35 {
36 return 0;
37 }
38 if (x == 1)
39 {
40 return 1;
41 }
42
43 return FMath::Pow(2, -10 * x) * FMath::Sin((x * 10 - 0.75) * c4) + 1;
44 };
45
46 GetWorld()->GetTimerManager().SetTimer(StampTimerHandle, [this, easeOutElastic, InDeltaTime]()
47 {
48 Image_Complete->SetRenderScale(FVector2D(StampStartScale - easeOutElastic(StampDX)));
49 StampDX += InDeltaTime;
50 if (StampDX >= 1.0f)
51 {
52 GetWorld()->GetTimerManager().ClearTimer(StampTimerHandle);
53 StampDX = 0;
54 }
55 }, InDeltaTime, true);
56}
