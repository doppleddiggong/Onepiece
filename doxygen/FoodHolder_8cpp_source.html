38void AFoodHolder::OnFoodBoxOverlapBegin(
39 UPrimitiveComponent* OverlappedComponent,
40 AActor* OtherActor,
41 UPrimitiveComponent* OtherComp,
42 int32 OtherBodyIndex,
43 bool bFromSweep,
44 const FHitResult& SweepResult)
45{
46 if (!OtherActor)
47 return;
48
49 // 서버에서만 실행
50 if (!HasAuthority())
51 return;
52
53 if (bIsActivated)
54 {
55 // 이미 Activate 되었으면 작동 안 함
56 return;
57 }
58
59 // Food인지 확인
60 if (AFood* Food = Cast<AFood>(OtherActor))
61 {
62 bool bSuccess = CheckFood(Food);
63
64 // 블루프린트 이벤트 호출 (부모 클래스의 OnActivate)
65 OnActivate(bSuccess);
66

68 if (auto DM = ANetworkBroadcastActor::Get(this))
69 {
70 DM->SendTutorMessage(FText::FromString(bSuccess ? GameMessage::Holder_Success : GameMessage::Holder_Fail), this);
71 }
72 }
73}

75bool AFoodHolder::CheckFood(AFood* TargetFood)
76{
77 if (!TargetFood)
78 return false;
79
80 // Food Index 비교
81 bool bSuccess = false;
82
83 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
84 if (GS)
85 {
86 const int32 CorrectIdx = GS->GetReadScenarioData().correct_answer_index;
87
88 if (TargetFood->GetFoodIndex() == CorrectIdx) bSuccess = true;
89 }
90
91 PRINTLOG(TEXT("FoodIndex: %d (Answer: %d), Result: %s"),
92 TargetFood->GetFoodIndex(), AnswerFoodIndex,
93 bSuccess ? TEXT("Success") : TEXT("Fail"));
94
95 if (bSuccess)
96 {
97 // Success: Food를 HoldPos 위치보다 살짝 위에 배치
98 if (HoldPos)
99 {
100 FVector ActivatedLocation = HoldPos->GetComponentLocation();
101 ActivatedLocation.Z += ActivatedHeightOffset;
102 TargetFood->SetActorLocation(ActivatedLocation);
103 TargetFood->SetActorRotation(HoldPos->GetComponentRotation());
104 }
105
106 // Food의 물리 및 충돌 비활성화
107 if (TargetFood->Mesh)
108 {
109 TargetFood->Mesh->SetSimulatePhysics(false);
110 TargetFood->Mesh->SetCollisionEnabled(ECollisionEnabled::NoCollision);
111 }
112
113 // Activate 상태로 전환
114 bIsActivated = true;
115 CurTarget = TargetFood;
116
117 // 서버에서도 머티리얼 업데이트 (클라이언트는 OnRep_IsActivated에서 호출됨)
118 UpdateActivateState(true);
119 }
120 else
121 {
122 // Fail: Food 제거
123 bIsActivated = false;
124 TargetFood->Destroy();
125
126 // 서버에서 머티리얼 업데이트 (오답)
127 UpdateActivateState(false);
128 }
129
130 return bSuccess;
131}
