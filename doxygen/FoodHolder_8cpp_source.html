107void AFoodHolder::OnFoodBoxOverlapBegin(
108 UPrimitiveComponent* OverlappedComponent,
109 AActor* OtherActor,
110 UPrimitiveComponent* OtherComp,
111 int32 OtherBodyIndex,
112 bool bFromSweep,
113 const FHitResult& SweepResult)
114{
115 if (!OtherActor)
116 return;
117
118 if (!HasAuthority())
119 return;
120
121 if (bIsActivated)
122 return;
123
124 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
125 if (!GS)
126 return;
127
128 TryIdx++;
129
130 // Food인지 확인
131 if (AFood* Food = Cast<AFood>(OtherActor))
132 {
133 const bool bSuccess = CheckFood(Food);
134
135 FScenarioTargetData TempData;
136 TempData.word1 = Food->CurrentFoodData.word1;
137 TempData.word2 = Food->CurrentFoodData.word2;
138 GS->TryListenAnswerData.target_data.Add(TempData);
139
140 // 블루프린트 이벤트 호출
141 OnActivate(bSuccess);
142
143 if (bSuccess)
144 {
145 UE_LOG(LogTemp, Warning, TEXT("[FoodHolder] Correct"));
146
147 // 정답인 경우
148 FTimerHandle TimerHandle;
149 GetWorldTimerManager().SetTimer(TimerHandle, [this]
150 {
151 ADoor* Door = FindDoorToOpen();
152 if (Door) Door->OpenDoor();
153
154 // 모든 클라이언트에 정답 인덱스와 함께 결과 팝업 표시
155 Multicast_ShowResultPopup(TryIdx);
156 }, 0.5f, false);
157 }
158 else
159 {
160 UE_LOG(LogTemp, Warning, TEXT("[FoodHolder] Wrong"));
161
162 // 오답인 경우
163 FTimerHandle TimerHandle;
164 GetWorldTimerManager().SetTimer(TimerHandle, [this, Food, GS]
165 {
166 GS->WrongListenAnswerList.Add(TryIdx);
167
168 // 모든 클라이언트에 오답 메시지 표시
169 Multicast_ShowWrongPopup(Food->CurrentFoodData.word1.name);
170
171 // Food 소거 (서버에서만, 자동 복제됨)
172 Food->Destroy();
173 }, 0.5f, false);
174 }
175 }
176}

178bool AFoodHolder::CheckFood(AFood* TargetFood)
179{
180 if (!TargetFood) return false;
181
182 // Food의 모든 충돌 비활성화
183 if (TargetFood->Mesh)
184 {
185 TargetFood->Mesh->SetSimulatePhysics(false);
186 TargetFood->Mesh->SetCollisionEnabled(ECollisionEnabled::NoCollision);
187 }
188
189 // ListenQuest 정답 인덱스 가져오기
190 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
191 if (!GS) return false;
192
193 const int32 CorrectIdx = GS->GetListenScenarioData().correct_answer_index;
194 const TArray<FScenarioTargetData>& ScenarioData = GS->GetListenScenarioData().target_data;
195
196 FString CorrectCityName = ScenarioData[CorrectIdx].word1.name;
197 FString CorrectFoodName = ScenarioData[CorrectIdx].word2.name;
198
199 if (CorrectFoodName == TargetFood->CurrentFoodData.word2.name
200 && CorrectCityName == TargetFood->CurrentFoodData.word1.name)
201 {
202 // Success: Food를 HoldPos 위치보다 살짝 위에 배치
203 FVector ActivatedLocation = HoldPos->GetComponentLocation();
204 ActivatedLocation.Z += ActivatedHeightOffset;
205 TargetFood->SetActorLocation(ActivatedLocation);
206 TargetFood->SetActorRotation(HoldPos->GetComponentRotation());
207
208 // Activate 상태로 전환
209 bIsActivated = true;
210 CurTarget = TargetFood;
211
212 // 서버에서도 머티리얼 업데이트 (클라이언트는 OnRep_IsActivated에서 호출됨)
213 UpdateActivateState(true);
214
215 return true;
216 }
217 else
218 {
219 UE_LOG(LogTemp, Warning, TEXT("[CheckFood] Returning FALSE"));
220
221 // Fail: 서버에서 머티리얼 업데이트 (오답)
222 bIsActivated = false;
223 UpdateActivateState(false);
224
225 return false;
226 }
227}

229void AFoodHolder::UpdateActivateState(bool State)
230{
231 // 머티리얼 파라미터 설정
232 if (MeshComponent && MeshComponent->GetNumMaterials() > 0)
233 {
234 UMaterialInstanceDynamic* DynamicMaterial = Cast<UMaterialInstanceDynamic>(MeshComponent->GetMaterial(0));
235 if (!DynamicMaterial)
236 DynamicMaterial = MeshComponent->CreateDynamicMaterialInstance(0);
237
238 if (DynamicMaterial)
239 DynamicMaterial->SetScalarParameterValue(FName("Activate"), State ? 1.0f : 0.0f);
240 }
