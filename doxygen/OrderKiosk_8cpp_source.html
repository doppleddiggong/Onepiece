113class ADropper* AOrderKiosk::FindDropperByIdx(int32 InIdx)
114{
115 TArray<AActor*> Droppers;
116 UGameplayStatics::GetAllActorsOfClass(GetWorld(), ADropper::StaticClass(), Droppers);
117
118 for (auto Dropper : Droppers)
119 {
120 if (ADropper* Dpp = Cast<ADropper>(Dropper))
121 {
122 if (Dpp->DropperIndex == InIdx)
123 {
124 return Dpp;
125 }
126 }
127 }
128 return nullptr;
129}
