69void AOrderKiosk::BeginSubmitOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
70 UPrimitiveComponent* OtherComp, int32 OtherBodyIndex, bool bFromSweep, const FHitResult& SweepResult)
71{
72 UE_LOG(LogTemp, Warning, TEXT("[OrderKiosk::BeginSubmitOverlap] Called with Actor: %s"),
73 OtherActor ? *OtherActor->GetName() : TEXT("NULL"));
74
75 // 답 제출하면 컨베이어 재개
76 if (AListenAnswer* Answer = Cast<AListenAnswer>(OtherActor))
77 {
78 UE_LOG(LogTemp, Warning, TEXT("[OrderKiosk::BeginSubmitOverlap] ListenAnswer detected. AnswerType: %d, Index: %d"),
79 Answer->AnswerData.AnswerType, InAnswerType);
80
81 if (ConveyorsToControl.Num() <= 0)
82 {
83 UE_LOG(LogTemp, Error, TEXT("[OrderKiosk::BeginSubmitOverlap] ConveyorsToControl is empty!"));
84 return;
85 }
86
87 for (auto Conveyor : ConveyorsToControl)
88 {
89 if (AConveyorBelt* CB = Cast<AConveyorBelt>(Conveyor))
90 {
91 if (!CurrentFoodContainer) return;
92
93 // 인덱스 번호와 정답 타입이 일치하면 데이터 전달 후 컨베이어 움직이기
94 if (Answer->AnswerData.AnswerType == InAnswerType)
95 {
96 switch (InAnswerType)
97 {
98 case EAnswerType::Food:
99 CurrentFoodContainer->SetFoodMesh(Answer->AnswerData.word1, Answer->Mesh->GetStaticMesh());
100 break;
101
102 case EAnswerType::City:
103 {
104 CurrentFoodContainer->SetCityName(Answer->AnswerData.word1);
105
106 FTimerHandle TimerHandle;
107 GetWorldTimerManager().SetTimer(TimerHandle, FTimerDelegate::CreateLambda([this]
108 {
109
110 Server_MoveFoodContainer(CurrentFoodContainer);
111
112 }), 3.f, false);
113
114 break;
115 }
116
117 default:
118 break;
119 }
120
121 CB->ChangeConveyorSpeed(200.f);
122
123 Server_DestroyListenAnswer(Answer);
124 //CurrentFoodContainer = nullptr;
125 }
126 }
127 }
128
129 IsOnceStopped = true;
130 }
131 else
132 {
133 UE_LOG(LogTemp, Warning, TEXT("[OrderKiosk::BeginSubmitOverlap] OtherActor is not a ListenAnswer"));
134 }
135}
