56void AOrderKiosk::BeginFoodOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
57 UPrimitiveComponent* OtherComp, int32 OtherBodyIndex, bool bFromSweep, const FHitResult& SweepResult)
58{
59 if (IsOnceStopped) return;
60
61 if (AFood* Temp = Cast<AFood>(OtherActor))
62 {
63 CurrentFoodContainer = Temp;
64
65 if (ConveyorsToControl.Num() <= 0) return;
66
67 // 만약 음식 큐브일 경우, 컨베이어 멈추기
68 for (auto Conveyor : ConveyorsToControl)
69 {
70 if (AConveyorBelt* CB = Cast<AConveyorBelt>(Conveyor))
71 {
72 CB->ChangeConveyorSpeed(0.f);
73 }
74 }
75 }
76}

78void AOrderKiosk::BeginSubmitOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
79 UPrimitiveComponent* OtherComp, int32 OtherBodyIndex, bool bFromSweep, const FHitResult& SweepResult)
80{
81 UE_LOG(LogTemp, Warning, TEXT("[OrderKiosk::BeginSubmitOverlap] Called with Actor: %s"),
82 OtherActor ? *OtherActor->GetName() : TEXT("NULL"));
83
84 // 답 제출하면 컨베이어 재개
85 if (AListenAnswer* Answer = Cast<AListenAnswer>(OtherActor))
86 {
87 UE_LOG(LogTemp, Warning, TEXT("[OrderKiosk::BeginSubmitOverlap] ListenAnswer detected. AnswerType: %d, Index: %d"),
88 Answer->AnswerData.AnswerType, Index);
89
90 if (ConveyorsToControl.Num() <= 0)
91 {
92 UE_LOG(LogTemp, Error, TEXT("[OrderKiosk::BeginSubmitOverlap] ConveyorsToControl is empty!"));
93 return;
94 }
95
96 for (auto Conveyor : ConveyorsToControl)
97 {
98 if (AConveyorBelt* CB = Cast<AConveyorBelt>(Conveyor))
99 {
100 if (!CurrentFoodContainer) return;
101
102 // 인덱스 번호와 정답 타입이 일치하면 데이터 전달 후 컨베이어 움직이기
103 if (Answer->AnswerData.AnswerType == Index)
104 {
105 switch (Index)
106 {
107 case 0:
108 break;
109
110 case 1:
111 CurrentFoodContainer->SetFoodMesh(Answer->AnswerData.word1, Answer->Mesh->GetStaticMesh());
112 break;
113
114 case 2:
115 CurrentFoodContainer->SetCityName(Answer->AnswerData.word1);
116 break;
117
118 default:
119 break;
120 }
121
122 CB->ChangeConveyorSpeed(200.f);
123
124 Server_DestroyListenAnswer(Answer);
125 //CurrentFoodContainer = nullptr;
126 }
127 }
128 }
129
130 IsOnceStopped = true;
131 }
132 else
133 {
134 UE_LOG(LogTemp, Warning, TEXT("[OrderKiosk::BeginSubmitOverlap] OtherActor is not a ListenAnswer"));
135 }
136}
