158void AWheatly::Tick(float DeltaSeconds)
159{
160 Super::Tick(DeltaSeconds);
161
162 if (!SpeakStageActor)
163 return;
164
165 // 퀘스트 진행 중일 때
166 if (auto CurrentSpeaker = SpeakStageActor->GetCurrentSpeaker())
167 {
168 if (HasAuthority())
169 {
170 if (auto SpeakerPawn = CurrentSpeaker->GetPawn())
171 {
172 ReplicatedEyeSightEnd = SpeakerPawn->GetActorLocation();
173 bEyeSightVisible = true;
174
175 ApplyEyeSight();
176
177 // 스피커를 쳐다보도록 회전
178 FRotator LookAtRotation = UKismetMathLibrary::FindLookAtRotation(GetActorLocation(), SpeakerPawn->GetActorLocation());
179 FRotator TargetRotation( LookAtRotation.Pitch, LookAtRotation.Yaw, 0); // Yaw만 사용
180 SetActorRotation(FMath::RInterpTo(GetActorRotation(), TargetRotation, DeltaSeconds, 5.0f));
181 }
182 }
183 }
184 // 퀘스트 진행 중이 아닐 때
185 else
186 {
187 if (HasAuthority())
188 {
189 bEyeSightVisible = false;
190
191 ApplyEyeSight();
192 }
193
194 TArray<AActor*> OverlappingActors;
195 if (PlayerDetectionZone)
196 PlayerDetectionZone->GetOverlappingActors(OverlappingActors, APlayerActor::StaticClass());
197
198 APawn* NearestPawn = nullptr;
199 double MinDistanceSquared = MAX_dbl;
200
201 for (AActor* OverlappingActor : OverlappingActors)
202 {
203 if (APawn* Pawn = Cast<APawn>(OverlappingActor))
204 {
205 double DistanceSquared = FVector::DistSquared(GetActorLocation(), Pawn->GetActorLocation());
206 if (DistanceSquared < MinDistanceSquared)
207 {
208 MinDistanceSquared = DistanceSquared;
209 NearestPawn = Pawn;
210 }
211 }
212 }
213
214 if(NearestPawn)
215 {
216 // LineTrace로 시야 확인
217 FHitResult HitResult;
218 FVector StartLocation = EyeMesh->GetComponentLocation();
219 FVector EndLocation = NearestPawn->GetActorLocation();
220 FCollisionQueryParams CollisionParams;
221 CollisionParams.AddIgnoredActor(this); // 자기 자신은 무시
222
223 bool bHit = GetWorld()->LineTraceSingleByChannel(HitResult, StartLocation, EndLocation, ECC_Visibility, CollisionParams);
224
225 // 아무것도 맞지 않았거나, 맞은 대상이 목표한 폰일 경우에만 시야가 확보된 것으로 간주
226 if (!bHit || (bHit && HitResult.GetActor() == NearestPawn))
227 {
228 // 가장 가까운 플레이어를 쳐다보도록 회전
229 FRotator LookAtRotation = UKismetMathLibrary::FindLookAtRotation(EyeMesh->GetComponentLocation(), NearestPawn->GetActorLocation());
230 FRotator TargetRotation( LookAtRotation.Pitch, LookAtRotation.Yaw, 0); // Yaw만 사용
231 SetActorRotation(FMath::RInterpTo(GetActorRotation(), TargetRotation, DeltaSeconds, 2.0f));
232 }
233 }
234 }
235}

299void AWheatly::SyncSpeakScenarioData(APlayerActor* Player, const FResponseSpeakScenario& Data)
300{
301 if (!HasAuthority())
302 return;
303
304 // Player가 유효한지 확인
305 if (!Player)
306 {
307 PRINTLOG(TEXT("[AWheatly] SyncSpeakScenarioData: Player is null"));
308 return;
309 }
310
311 // PlayerState에 데이터 저장
312 if (auto PS = Player->GetPlayerState<ALingoPlayerState>())
313 {
314 PS->SpeakScenarioData = Data;
315 PS->OnUpdateSpeakScenarioData();
316
317 PRINTLOG(TEXT("[AWheatly] SyncSpeakScenarioData: Successfully synced scenario data for %s"),
318 *ULingoGameHelper::GetPlayerNameFromState(PS));
319 }
320 else
321 {
322 PRINTLOG(TEXT("[AWheatly] SyncSpeakScenarioData: Failed to get PlayerState"));
323
324 // 실패 시 SpeakStage를 종료
325 if (SpeakStageActor && SpeakStageActor->GetCurrentSpeaker())
326 SpeakStageActor->EndStage();
327 }

334void AWheatly::OnInteractionTriggered(AActor* InteractingActor)
335{
336 if (!HasAuthority() || !SpeakStageActor)
337 return;
338
339 APlayerActor* InteractingPlayer = Cast<APlayerActor>(InteractingActor);
340 if (!InteractingPlayer)
341 return;
342
343 auto PC = Cast<APlayerControl>(InteractingPlayer->GetController());
344
345 // 플레이어의 SpeakQuest 완료 여부 확인
346 if (auto PS = InteractingPlayer->GetPlayerState<ALingoPlayerState>())
347 {
348 if (PS->IsSpeakQuestCompleted())
349 {
350 if ( PC )
351 PC->Client_ToastMessage(TEXT("Already Clear SpeakQuest"));
352
353 return;
354 }
355 }
356
357 // SpeakStage의 상태를 직접 확인
358 if (ALingoPlayerState* CurrentSpeaker = SpeakStageActor->GetCurrentSpeaker())
359 {
360 if ( PC )
361 PC->Client_ToastMessage(FString::Printf(TEXT("Current Turn is [%s]"), *ULingoGameHelper::GetPlayerNameFromState(CurrentSpeaker)));
362
363 return;
364 }
365
366 if ( PC )
367 PC->Client_RequestSpeakScenario(this);
368}

378void AWheatly::OnSpeakStageSpeakerChanged(APlayerState* NewSpeaker)
379{
380 // 눈 색상 변경은 서버에서만
381 if (HasAuthority())
382 {
383 ReplicatedEyeColor = (NewSpeaker != nullptr)
384 ? FLinearColor(1.0f, 1.0f, 0.0f, 1.0f) // Yellow: Busy
385 : FLinearColor(0.0f, 0.5f, 1.0f, 1.0f); // Blue: Available
386
387 OnRep_EyeColor();
388 }
389
390 // 위젯 표시 제어는 모든 클라이언트에서 실행
391 if (!InteractableComp)
392 return;
393
394 // 로컬 플레이어 확인
395 if ( auto LocalPawn = ULingoGameHelper::GetLocalPawn(GetWorld()) )
396 {
397 auto LocalPlayerState = LocalPawn->GetPlayerState<ALingoPlayerState>();
398
399 // 로컬 플레이어가 현재 발화자인지 확인
400 if (LocalPlayerState && NewSpeaker && LocalPlayerState == NewSpeaker)
401 {
402 // 발화자이면 위젯 숨김
403 InteractableComp->SetWidgetVisibility(false);
404 PRINTLOG(TEXT("[AWheatly] Local player is speaker - hiding widget"));
405 }
406 else
407 {
408 // 발화자가 아니면, DetectionRange 내에 있을 때만 위젯 표시
409 if (InteractableComp->DetectionRange)
410 {
411 if ( IsInRange(LocalPawn))
412 {
413 InteractableComp->SetWidgetVisibility(true);
414 PRINTLOG(TEXT("[AWheatly] Local player not speaker and in range - showing widget"));
415 }
416 }
417 }
418 }

469void AWheatly::UpdateEyeSight(const FVector& Start, const FVector& End)
470{
471 FVector Delta = End - Start;
472 float Length = Delta.Size();
473 if (Length <= KINDA_SMALL_NUMBER)
474 {
475 EyeSightComp->SetVisibility(false);
476 return;
477 }
478
479 FVector Midpoint = Start + (Delta * 0.5f);
480 EyeSightComp->SetWorldLocation(Midpoint);
481
482 FVector Direction = Delta / Length;
483 FRotator Rotation = FRotationMatrix::MakeFromZ(Direction).Rotator();
484 EyeSightComp->SetWorldRotation(Rotation);
485
486 const float LengthScale = Length / FMath::Max(IndicatorBaseLength, KINDA_SMALL_NUMBER);
487 const float TargetThickness = 10.f; // Desired thickness
488 const float MeshDiameter = FMath::Max(IndicatorBaseRadius * 2.0f, KINDA_SMALL_NUMBER);
489 const float RadiusScale = (TargetThickness / MeshDiameter)*2.5f;
490
491 EyeSightComp->SetWorldScale3D(FVector(RadiusScale, RadiusScale, LengthScale));
492 EyeSightComp->SetVisibility(true);
493}
