73void APopupTesterActor::InterviewPopup()
74{
75 if (const auto PopupMgr = UPopupManager::Get(GetWorld()))
76 {
77 if (const auto Popup = Cast<UPopup_Interview>(PopupMgr->ShowPopup(EPopupType::Interview)))
78 {
79 // 테스트용 더미 데이터 생성

84 Q1.Id = 1;
85 Q1.TypeCode = 0;
86 Q1.Eng = TEXT("What is your current country of residence?");
87 Q1.Kor = TEXT("현재 거주 국가는 어디인가요?");
88 Q1.EngKey = TEXT("country");
89 Q1.KorKey = TEXT("국가");
90 Q1.CreatedAt = TEXT("2025-01-15");
91 TestData.Questions.Add(Q1);
92
93 // 질문 2
94 FInterviewQuestionData Q2;
95 Q2.Id = 2;
96 Q2.TypeCode = 0;
97 Q2.Eng = TEXT("How did you find out about this program?");
98 Q2.Kor = TEXT("이 프로그램을 어떻게 알게 되었나요?");
99 Q2.EngKey = TEXT("discovery");
100 Q2.KorKey = TEXT("발견경로");
101 Q2.CreatedAt = TEXT("2025-01-15");
102 TestData.Questions.Add(Q2);
103
104 // 질문 3
105 FInterviewQuestionData Q3;
106 Q3.Id = 3;
107 Q3.TypeCode = 0;
108 Q3.Eng = TEXT("What are your goals for learning Korean?");
109 Q3.Kor = TEXT("한국어 학습의 목표는 무엇인가요?");
110 Q3.EngKey = TEXT("goals");

112 Q3.CreatedAt = TEXT("2025-01-15");
113 TestData.Questions.Add(Q3);
114
115 // 팝업 초기화
116 Popup->InitPopup(TestData);
117
118 PRINTLOG(TEXT("[PopupTester] Interview popup opened with %d questions"), TestData.Questions.Num());
119 }
120 }
