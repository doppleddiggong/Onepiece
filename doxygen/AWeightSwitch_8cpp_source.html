66void AWeightSwitch::InitSwitch()
67{
68 if (!SwitchBody)
69 {
70 PRINTLOG( TEXT("SwitchBody is null"));
71 return;
72 }
73
74 // 1. 머티리얼 얻기
75 UMaterialInterface* Material = SwitchBody->GetMaterial(0);
76 if (!Material)
77 {
78 PRINTLOG( TEXT("Material is null"));
79 return;
80 }
81
82 // 2. 다이나믹 머티리얼 인스턴스 생성
83 UMaterialInstanceDynamic* DynMaterial = UMaterialInstanceDynamic::Create(Material, this);
84 if (!DynMaterial)
85 {
86 PRINTLOG( TEXT("Failed to create dynamic material"));
87 return;
88 }
89
90 // 3. 변수에 저장
91 EmissiveMaterial = DynMaterial;
92
93 // 4. 메시에 적용 (이 단계가 누락되어 있었음!)
94 SwitchBody->SetMaterial(0, EmissiveMaterial);
95
96 // 5. 파라미터 설정
97 EmissiveMaterial->SetVectorParameterValue(EmissiveParam, DeactivateColor);
98
99 UAnimInstance* AnimInstance = SwitchBody->GetAnimInstance();
100 if (!AnimInstance)
101 {
102 PRINTLOG( TEXT("AnimInstance is null"));
103 return;
104 }
105
106 // 7. 캐스팅 및 변수 설정
107 UTweenAnimInstance* CubeButtonAnim = Cast<UTweenAnimInstance>(AnimInstance);
108 if (CubeButtonAnim)
109 {
110 AnimBlueprint = CubeButtonAnim;
111 }
112 else
113 {
114 PRINTLOG( TEXT("Failed to cast to UABP_CubeButton"));
115 }
116}

118void AWeightSwitch::SetActivate(bool State)
119{
120 if (!EmissiveMaterial)
121 {
122 PRINTLOG( TEXT("EmissiveMaterial is null"));
123 return;
124 }
125
126 // 1. 색상 선택 (State가 true면 ActivateColor, false면 DeactivateColor)
127 FLinearColor SelectedColor = State ? ActivateColor : DeactivateColor;
128
129 // 2. 머티리얼 파라미터 설정
130 EmissiveMaterial->SetVectorParameterValue(EmissiveParam, SelectedColor);
131
132 // 3. 애님 블루프린트의 ChangeState 호출
133 if (AnimBlueprint)
134 {
135 AnimBlueprint->ChangeState(State);
136 }
137 else
138 {
139 PRINTLOG( TEXT("AnimBlueprint is null"));
140 }
141
142 OnActivate(State);
143}

158{

173bool AWeightSwitch::ActivateTrigger()
174{
175 TriggerDelay -= GetWorld()->GetDeltaSeconds();
176
177 if ( TriggerDelay < 0 )
178 {
179 TriggerDelay = Duration;
180 return true;
181 }
182
183 return false;

194void AWeightSwitch::OnBeginOverlap(
195 UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
196 UPrimitiveComponent* OtherComp, int32 OtherBodyIndex, bool bFromSweep, const FHitResult& SweepResult)
197{
198 if (!OtherActor)
199 return;
200
201 // [개선] 서버에서만 정답 판정 수행
202 if (!HasAuthority())
203 {
204 PRINTLOG(TEXT("OnBeginOverlap: Client detected overlap, skipping (server will handle)"));
205 return;
206 }
207
208 // 조건 1 : 플레이어일 경우 오픈
209 if ( IsPlayerDetect )
210 {
211 if (Cast<APlayerActor>(OtherActor))
212 {
213 // 리스트에 추가 (중복 방지)
214 OverlappingActors.AddUnique(OtherActor);
215
216 // 첫 번째 물체가 올라갔을 때만 타이머 시작
217 if (OverlappingActors.Num() == 1)
218 {
219 this->DetectTarget = true;
220 this->ElapsedTime = 0.0;
221 }
222 }
223 }
224
225 // 조건 2 : 정답 캐리어일 경우 오픈
226 if (Aluggage* Luggage = Cast<Aluggage>(OtherActor))
227 {
228 // [개선] GameState nullptr 체크
229 ALingoGameState* GS = Cast<ALingoGameState>(GetWorld()->GetGameState());
230 if (!GS)
231 {
232 PRINTLOG(TEXT("OnBeginOverlap: GameState is null! Cannot validate answer."));
233 return;
234 }
235
236 // [개선] ScenarioData 유효성 체크
237 if (GS)
238 {
239 const int32 CorrectIdx = GS->GetScenarioData().correct_answer_index;
240
241 PRINTLOG(TEXT("[WeightSwitch] Server validating: LuggageIdx=%d, CorrectIdx=%d"),
242 Luggage->GetSpawnIdx(), CorrectIdx);
243
244 if (CorrectIdx == Luggage->GetSpawnIdx())
245 {
246 // 리스트에 추가 (중복 방지)
247 OverlappingActors.AddUnique(OtherActor);
248
249 // 첫 번째 물체가 올라갔을 때만 타이머 시작
250 if (OverlappingActors.Num() == 1)
251 {
252 this->DetectTarget = true;
253 this->ElapsedTime = 0.0;
254 }
255
256 if (AnswerFound) return;
257
258 AnswerFound = true;
259 // 마지막으로 정답 추가
260 GS->WrongLuggageList.Add(Luggage->GetSpawnIdx());
261
262 // [개선] Multicast RPC로 모든 클라이언트에 정답 팝업 표시
263 FTimerHandle TimerHandle;
264 GetWorldTimerManager().SetTimer(TimerHandle, [this, GS, Luggage]
265 {
266 // 모든 클라이언트에 결과 팝업 표시
267 Multicast_ShowResultPopup();
268
269 // 오답 캐리어 로그 (서버에서만)
270 TArray<int32> WrongList = GS->WrongLuggageList;
271 if (WrongList.Num() == 0) return;
272
273 PRINTLOG( TEXT("[AWeightSwitch] Wrong luggage :"));
274 for (auto Wrong : WrongList)
275 {
276 PRINTLOG( TEXT("%d, "), Wrong);
277 }
278 }, 0.5f, false);
279 }
280 else
281 {
282 // [개선] Multicast RPC로 모든 클라이언트에 오답 팝업 표시
283 FString LuggageColor = Luggage->GetColor();
284 FString LuggagePattern = Luggage->GetPattern();
285 int32 LuggageIdx = Luggage->GetSpawnIdx();
286
287 FTimerHandle TimerHandle;
288 GetWorldTimerManager().SetTimer(TimerHandle, [this, LuggageColor, LuggagePattern, LuggageIdx, GS, Luggage]
289 {
290 // 모든 클라이언트에 오답 메시지 표시
291 Multicast_ShowWrongPopup(LuggageColor, LuggagePattern);
292
293 // 오답 목록에 인덱스 추가 (서버에서만)
294 GS->WrongLuggageList.Add(LuggageIdx);
295
296 // 큐브 소거 (서버에서만, 자동 복제됨)
297 Luggage->Destroy();
298 }, 0.5f, false);
299 }
300
301 }
302 }
303}

309void AWeightSwitch::OnEndOverlap(
310 UPrimitiveComponent* OverlappedComponent, AActor* OtherActor,
311 UPrimitiveComponent* OtherComp, int32 OtherBodyIndex)
312{
313 if (!OtherActor)
314 return;
315
316 // [개선] 서버에서만 실행
317 if (!HasAuthority())
318 return;
319
320 // 리스트에서 제거
321 OverlappingActors.Remove(OtherActor);
322
323 // 모든 물체가 내려갔을 때만 비활성화
324 if (OverlappingActors.Num() == 0)
325 {
326 this->DetectTarget = false;
327 this->ElapsedTime = 0.0;
328
329 // 물건이 모두 떨어지면 해제
330 UBroadcastManager::Get(GetWorld())->SendWeightSwitch(ButtonIndex, false);
331 }
332}

371void AWeightSwitch::Multicast_ShowWrongPopup_Implementation(const FString& LuggageColor, const FString& LuggagePattern)
372{
373 // 모든 클라이언트(호스트 포함)에서 오답 메시지 표시
374 if (UPopupManager* PopupMgr = UPopupManager::Get(GetWorld()))
375 {
376 FString Title = TEXT("Wrong Answer!");
377 FString Message = FString::Printf(TEXT("This is not the correct luggage.\n\nColor: %s\nPattern: %s"),
378 *LuggageColor, *LuggagePattern);
379
380 PopupMgr->ShowMsgBox(Title, Message, EMsgBoxType::OK,
381 FOnMsgBoxOkDelegate::CreateLambda([]() {
382 // 확인 버튼 클릭 시 아무 작업도 하지 않음 (팝업만 닫힘)
383 }));
384
385 PRINTLOG(TEXT("[WeightSwitch] Showing wrong popup on %s (Color: %s, Pattern: %s)"),
386 HasAuthority() ? TEXT("Server") : TEXT("Client"), *LuggageColor, *LuggagePattern);
387 }
