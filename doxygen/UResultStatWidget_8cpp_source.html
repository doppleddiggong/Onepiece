152void UResultStatWidget::LoadStyleTable()
153{
154 if (auto DM = UGameDataManager::Get(this))
155 {
156 // 런타임: GameDataManager에서 가져오기
157 StyleTable = DM->GetAllColorStyleData();
158 return;
159 }
160
161#if WITH_EDITOR
162 // 에디터 모드: DataTable 직접 로드
163 if (auto ColorStyleTable = LoadObject<UDataTable>(nullptr, COLORSTYLEDATA_PATH ))
164 {
165 for (const FName& RowName : ColorStyleTable->GetRowNames())
166 {
167 if (FColorStyleData* Row = ColorStyleTable->FindRow<FColorStyleData>(RowName, TEXT("")))
168 {
169 FString EnumString = RowName.ToString();
170 EColorStyleType TempColorType = static_cast<EColorStyleType>(
171 StaticEnum<EColorStyleType>()->GetValueByNameString(EnumString)
172 );
173 StyleTable.Add(TempColorType, *Row);
174 }
175 }
176 }
177#endif
178}

180void UResultStatWidget::ApplyStyle()
181{
182 if (!StyleTable.Contains(ColorType))
183 return;
184
185 const auto& Style = StyleTable[ColorType];
186
187 if (Image_BG)
188 Image_BG->SetColorAndOpacity(Style.BGColor);
189
190 if (Border_Title)
191 Border_Title->SetBrushColor(Style.BorderColor);
192
193 if (Text_Title)
194 Text_Title->SetColorAndOpacity(Style.TitleColor);
195}
