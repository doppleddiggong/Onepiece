22 static FString GetFullUrl(const FString& Endpoint)
23 {
24 const EServerMode Mode = UCustomNetworkSettings::GetCurrentServerMode();
25 const FServerConfig& Config = GetDefault<UCustomNetworkSettings>()->GetConfig(Mode);
26 return Config.GetFullUrl(Endpoint);
27 }
28
31 static FString GetFullUrlWithQuery(
32 const FString& Endpoint,
33 const TMap<FString, FString>& QueryParams)
34 {
35 FString BaseUrl = GetFullUrl(Endpoint);
36
37 if (QueryParams.Num() == 0)
38 return BaseUrl;
39
40 FString QueryString;
41 bool bFirst = true;
42
43 for (const auto& Pair : QueryParams)
44 {
45 if (Pair.Key.IsEmpty() || Pair.Value.IsEmpty())
46 continue;
47
48 FString EncodedKey = FGenericPlatformHttp::UrlEncode(Pair.Key);
49 FString EncodedValue = FGenericPlatformHttp::UrlEncode(Pair.Value);
50
51 if (bFirst)
52 {
53 QueryString += FString::Printf(TEXT("?%s=%s"), *EncodedKey, *EncodedValue);
54 bFirst = false;
55 }
56 else
57 {
58 QueryString += FString::Printf(TEXT("&%s=%s"), *EncodedKey, *EncodedValue);
59 }
60 }
61
62 return BaseUrl + QueryString;
63 }

67 static FString GetSocketURL()
68 {
69 const EServerMode Mode = UCustomNetworkSettings::GetCurrentServerMode();
70 const FServerConfig& Config = GetDefault<UCustomNetworkSettings>()->GetConfig(Mode);
71 return Config.WebSocketUrl;
72 }
