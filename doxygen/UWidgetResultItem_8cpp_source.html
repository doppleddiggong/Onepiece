30void UWidgetResultItem::SetGradeValue(EResourceTextureType TextureType)
31{
32 if (!Image_Grade)
33 return;
34
35 // GameDataManager에서 TextureType에 해당하는 텍스처 가져오기
36 UGameDataManager* DataManager = UGameDataManager::Get(this);
37 if (!DataManager)
38 return;
39
40 UTexture2D* Texture = DataManager->GetTexture(TextureType);
41 if (!Texture)
42 return;
43
44 // Image의 Brush 데이터 변경
45 FSlateBrush Brush = Image_Grade->GetBrush();
46 Brush.SetResourceObject(Texture);

48}

105void UWidgetResultItem::ApplyStyle()

108 return;
109

111
112 if (Image_BG)
113 Image_BG->SetColorAndOpacity(Style.BGColor);
114
115 if (Border_Title)

117
118 if (Text_Title)
119 Text_Title->SetColorAndOpacity(Style.TitleColor);

125void UWidgetResultItem::UpdateActivePanel()

130 int32 Index = 0;
131
132 switch (WidgetType)

134 case EResultItemWidgetType::Grade:

136 break;
137
138 case EResultItemWidgetType::Score:

140 break;
141
142 case EResultItemWidgetType::Rate:
143 Index = 2;
144 break;
145
146 case EResultItemWidgetType::Symbol:
147 Index = 3;
148 break;
149 }
150
151 WidgetSwitcher->SetActiveWidgetIndex(Index);
152}
