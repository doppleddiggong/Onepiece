167void ANetworkTesterActor::OnResponseScenario(FResponseScenario& ResponseData, bool bWasSuccessful)
168{
169 if (bWasSuccessful)
170 {
171 PRINTLOG(TEXT("--- Scenario SUCCESS ---"));
172 ResponseData.PrintData();
173 PRINTLOG(TEXT("Index: %d, Difficulty: %d"), ResponseData.index, ResponseData.dificulity);
174 PRINTLOG(TEXT("Correct Answer Index: %d"), ResponseData.correct_answer_index);
175 PRINTLOG(TEXT("Target Data Count: %d"), ResponseData.target_data.Num());
176
177 for (int32 i = 0; i < ResponseData.target_data.Num(); i++)
178 {
179 PRINTLOG(TEXT(" [%d] Word1: %s (Code: %s), Word2: %s (Code: %s)"),
180 i,
181 *ResponseData.target_data[i].word1.name,
182 *ResponseData.target_data[i].word1.code,
183 *ResponseData.target_data[i].word2.name,
184 *ResponseData.target_data[i].word2.code);
185 }
186
187 PRINTLOG(TEXT("Word Data 1 - Kor: %s, Eng: %s, Pronunciation: %s"),
188 *ResponseData.word_data1.Kor, *ResponseData.word_data1.Eng, *ResponseData.word_data1.Pronunciation);
189 PRINTLOG(TEXT("Word Data 2 - Kor: %s, Eng: %s, Pronunciation: %s"),
190 *ResponseData.word_data2.Kor, *ResponseData.word_data2.Eng, *ResponseData.word_data2.Pronunciation);
191 PRINTLOG(TEXT("Full Data - Kor: %s, Eng: %s, Pronunciation: %s"),
192 *ResponseData.full_data.Kor, *ResponseData.full_data.Eng, *ResponseData.full_data.Pronunciation);
193 }
194 else
195 {
196 PRINTLOG(TEXT("--- Scenario FAILED ---"));
197 }
198}
