198void ANetworkTesterActor::OnResponseScenario(FResponseScenario& ResponseData, bool bWasSuccessful)
199{
200 if (bWasSuccessful)
201 {
202 PRINTLOG(TEXT("--- Scenario SUCCESS ---"));
203 ResponseData.PrintData();
204 PRINTLOG(TEXT("Index: %d, Difficulty: %d"), ResponseData.index, ResponseData.dificulity);
205 PRINTLOG(TEXT("Correct Answer Index: %d"), ResponseData.correct_answer_index);
206 PRINTLOG(TEXT("Target Data Count: %d"), ResponseData.target_data.Num());
207
208 for (int32 i = 0; i < ResponseData.target_data.Num(); i++)
209 {
210 PRINTLOG(TEXT(" [%d] Word1: %s (Code: %s), Word2: %s (Code: %s)"),
211 i,
212 *ResponseData.target_data[i].word1.name,
213 *ResponseData.target_data[i].word1.code,
214 *ResponseData.target_data[i].word2.name,
215 *ResponseData.target_data[i].word2.code);
216 }
217
218 PRINTLOG(TEXT("Word Data 1 - Kor: %s, Eng: %s, Pronunciation: %s"),
219 *ResponseData.word_data1.Kor, *ResponseData.word_data1.Eng, *ResponseData.word_data1.Pronunciation);
220 PRINTLOG(TEXT("Word Data 2 - Kor: %s, Eng: %s, Pronunciation: %s"),
221 *ResponseData.word_data2.Kor, *ResponseData.word_data2.Eng, *ResponseData.word_data2.Pronunciation);
222 PRINTLOG(TEXT("Full Data - Kor: %s, Eng: %s, Pronunciation: %s"),
223 *ResponseData.full_data.Kor, *ResponseData.full_data.Eng, *ResponseData.full_data.Pronunciation);
224 }
225 else
226 {
227 PRINTLOG(TEXT("--- Scenario FAILED ---"));
228 }
229}
