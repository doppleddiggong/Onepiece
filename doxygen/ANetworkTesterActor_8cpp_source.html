289void ANetworkTesterActor::OnResponseChatAnswers(FResponseChatAnswers& ResponseData, bool bWasSuccessful)
290{
291 if (bWasSuccessful)
292 {
293 PRINTLOG(TEXT("--- Chat Answers SUCCESS ---"));
294
295 // AI 응답을 Bot 정보로 채팅에 표시
296 if (auto* GS = GetWorld()->GetGameState<ALingoGameState>())
297 {
298 FText AIAnswer = FText::FromString(ResponseData.answer);
299 // Bot은 PlayerIndex -1 사용
300 GS->MulticastRPC_SendChat(GS->GetBotInfo(), AIAnswer, -1);
301
302 PRINTLOG(TEXT("[AI Chat] AI Answer: %s"), *ResponseData.answer);
303 }
304 }
305 else
306 {
307 PRINTLOG(TEXT("--- Chat Answers FAILED ---"));
308 }
