112void ARobotPlayer::OnMove(const FInputActionValue& Value)
113{
114 // input is a Vector2D
115 FVector2D MovementVector = Value.Get<FVector2D>();
116
117 if (GetController() != nullptr)
118 {
119 // find out which way is forward
120 const FRotator Rotation = GetController()->GetControlRotation();
121 const FRotator YawRotation(0, Rotation.Yaw, 0);
122
123 // get forward vector
124 const FVector ForwardDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::X);
125
126 // get right vector
127 const FVector RightDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::Y);
128
129 // add movement
130 AddMovementInput(ForwardDirection, MovementVector.Y);
131 AddMovementInput(RightDirection, MovementVector.X);
132 }
133}
