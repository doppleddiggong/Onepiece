129void AOwlPlayer::OnMove(const FInputActionValue& Value)
130{
131 // input is a Vector2D
132 FVector2D MovementVector = Value.Get<FVector2D>();
133
134 if (GetController() != nullptr)
135 {
136 // find out which way is forward
137 const FRotator Rotation = GetController()->GetControlRotation();
138 const FRotator YawRotation(0, Rotation.Yaw, 0);
139
140 // get forward vector
141 const FVector ForwardDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::X);
142
143 // get right vector
144 const FVector RightDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::Y);
145
146 // add movement
147 AddMovementInput(ForwardDirection, MovementVector.Y);
148 AddMovementInput(RightDirection, MovementVector.X);
149 }
150}
